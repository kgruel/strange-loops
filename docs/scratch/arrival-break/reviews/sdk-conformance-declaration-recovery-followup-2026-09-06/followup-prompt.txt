480: def _publish_cache(
481:     plan: DeclarationEditPlan,
482:     *,
483:     intent: Path,
484:     intent_data: dict[str, Any],
485:     failure_hook: Callable[[str], None] | None,
486: ) -> None:
487:     """Detach the verified old cache, then exclusively link a full new file."""
488:     target = plan.target_path
489:     temp = (
490:         Path(intent_data["cache_temp"])
491:         if intent_data.get("cache_temp")
492:         else _unique_sibling(target, "cache")
493:     )
494:     backup = (
495:         Path(intent_data["cache_backup"])
496:         if intent_data.get("cache_backup")
497:         else _unique_sibling(target, "backup")
498:     )
499:     intent_data["cache_temp"] = str(temp)
500:     intent_data["cache_backup"] = str(backup)
501:     _replace_json(intent, intent_data)
502:     try:
503:         proposed = plan.proposed_text.encode("utf-8")
504:         if temp.exists():
505:             _verify_cache_temp(temp, proposed)
506:         else:
507:             _create_cache_temp(temp, proposed)
508:             if failure_hook:
509:                 failure_hook("after-cache-temp")
510:         if target.exists():
511:             current = target.read_bytes()
512:             if current != plan.old_cache_bytes:
513:                 raise DeclarationStale(
514:                     "declaration cache changed after preparation; refusing publication"
515:                 )
516:             if backup.exists():
517:                 raise DeclarationApplyError(
518:                     "a prior cache publication left a backup beside the live declaration"
519:                 )
520:             target.rename(backup)
521:             _fsync_parent(target)
522:         elif not backup.exists():
523:             raise DeclarationApplyError(
524:                 "declaration cache is absent without the verified detached backup"
525:             )
526:         if failure_hook:
527:             failure_hook("after-cache-detach")
528:         if backup.read_bytes() != plan.old_cache_bytes:
529:             raise DeclarationApplyError(
530:                 "detached declaration cache bytes differ from prepared evidence"
531:             )
532:         os.link(temp, target)
533:         temp.unlink()
534:         _fsync_parent(target)
535:         if failure_hook:
536:             failure_hook("after-cache-link")
537:         with suppress(FileNotFoundError):
538:             backup.unlink()
539:         _fsync_parent(backup)
540:     except BaseException:
541:         raise

711: def _signed_with_existing_key(
712:     *,
713:     credentials: WriteCredentials,
714:     signer: Callable[[str, str], str | None] | None,
715:     verifier: Callable[[str, str, str], bool],
716:     registry: Any,
717:     author: str,
718:     head: Head,
719:     digest: str,
720:     label: str,
721:     domain: SigningDomain,
722:     purpose: CredentialPurpose,
723:     context: CapturedSigningContext,
724:     session: CredentialResolutionSession,
725: ) -> str:
726:     if credentials.mapped:
727:         request = CredentialRequest(
728:             namespace=credentials.binding_namespace,
729:             observer=author,
730:             domain=domain,
731:             purpose=purpose,
732:         )
733:         try:
734:             signature = session.sign(
735:                 credentials,
736:                 request,
737:                 digest,
738:                 context=context,
739:                 authorized_keys=context.keys_for_author(author),
740:                 required=True,
741:             )
742:         except CredentialBindingRefused as exc:
743:             raise _preparation_refused(
744:                 f"{label} mapped credential refused: {exc}",
745:                 captured_head=head,
746:             ) from exc
747:         if signature is None:  # required=True makes this unreachable
748:             raise _preparation_refused(f"missing {label} signature")
749:         return signature
750:     if not callable(signer):
751:         raise _preparation_refused(f"missing {label} signer")
752:     keys = _valid_author_keys(registry, author, head)
753:     if not keys:
754:         raise _preparation_refused(
755:             f"edit author {author!r} has no key valid at the captured Arrival head"
756:         )
757:     signature = signer(author, digest)
758:     if not signature:
759:         raise _preparation_refused(f"missing {label} signature")

761:         raise _preparation_refused(
762:             f"{label} signature does not verify under a key valid for edit author "
763:             f"{author!r} at the captured Arrival head"
764:         )
765:     return signature

858: def prepare_declaration_edit(
859:     registry: BackendRegistry,
860:     descriptor: StoreDescriptor,
861:     *,
862:     target: Path | str,
863:     proposed_text: str,
864:     observer: str,
865:     credentials: WriteCredentials | None,
866:     fact_verify: Callable[[str, str, str], bool],
867:     arrival_verify: Callable[[str, str, str], bool],
868:     source_cache_bytes: bytes | None = None,
869:     require_cache_basis: bool = False,
870: ) -> DeclarationEditPlan:
871:     """Prepare one exact declaration transition against an attested CURRENT head.
872: 
873:     This function only reads and signs in memory. It creates no intent, does
874:     not append, and does not touch the declaration cache. The registry opener
875:     is expected to return an attested custody handle; its ``scan`` is the
876:     structurally verified genesis-through-head sequence required by the
877:     iterable authorship seam.
878:     """
879:     target_path = Path(target).resolve()
880:     if not isinstance(proposed_text, str) or not proposed_text:
881:         raise _preparation_refused("proposed declaration text is empty")
882:     try:
883:         old_cache_bytes = target_path.read_bytes()
884:     except OSError as exc:
885:         raise _preparation_refused(
886:             f"cannot read declaration cache {target_path}: {exc}"
887:         ) from exc
888:     if source_cache_bytes is not None and old_cache_bytes != source_cache_bytes:
889:         raise _preparation_refused(
890:             "declaration cache changed while preparing the requested splice"
891:         )
892:     try:
893:         ast = parse_vertex(proposed_text, path=target_path)
894:         validate_vertex(ast)
895:     except Exception as exc:
896:         raise _preparation_refused(
897:             f"proposed declaration is not valid Vertex grammar/semantics: {exc}"
898:         ) from exc
899:     validate_arrival_runtime_identity(
900:         ast.name,
901:         ast.loops,
902:         refusal=_preparation_refused,
903:     )
904:     _validate_residence(target_path, ast, descriptor)
905:     proposed_documents = tuple(
906:         document.as_json() for document in vertex_to_documents(ast)
907:     )
908: 
909:     ledger: ArrivalLedger | None = None
910:     query: ArrivalQuery | None = None
911:     snapshot: Any | None = None
912:     atomic_limit: int | None = None
913:     captured_head: Head | None = None
914:     try:
915:         ledger, query = registry.open(descriptor)
916:         report = getattr(ledger, "opened", None)
917:         comparison = getattr(report, "comparison", None)
918:         if isinstance(comparison, PreGenesis):
919:             if comparison.ledger_refusal is not None:
920:                 raise _preparation_refused(
921:                     f"Arrival ledger is not opened: {comparison.ledger_refusal}"
922:                 )
923:             raise _preparation_refused(
924:                 "Arrival ledger has no captured head for declaration editing"
925:             )
926:         if isinstance(comparison, Indeterminate):
927:             raise _preparation_refused(
928:                 f"Arrival head attestation is indeterminate: {comparison.refusal}"
929:             )
930:         if not isinstance(comparison, Compared):
931:             raise _preparation_refused(
932:                 "registry opener did not provide a compared Arrival head"
933:             )
934:         captured_head = comparison.presented
935:         snapshot = query.open_snapshot(
936:             captured_head=captured_head,
937:             requirement=ProjectionRequirement.CURRENT,
938:             continuation=None,
939:         )
940:         validate_arrival_declaration_anchor(
941:             snapshot.declaration_anchor, captured_head
942:         )
943:         facts = snapshot.facts(
944:             FactRequest(
945:                 limit=None,
946:                 kind="_decl",
947:                 include_internal=True,
948:                 order="oldest",
949:             )
950:         ).items
951:         documents = resolve_declaration_documents_from_snapshot(
952:             snapshot.declaration_anchor, facts
953:         )
954:         if documents is None or isinstance(documents, Unhistorized):
955:             raise _preparation_refused(
956:                 "the CURRENT Arrival snapshot has no historized declaration"
957:             )
958:         represented = snapshot.represented
959:         if represented is None:
960:             raise _preparation_refused(
961:                 "the CURRENT declaration snapshot has no matching lineage basis"
962:             )
963:         projected = complete_projection_custody(
964:             ledger,
965:             captured=captured_head,
966:             represented=represented,
967:             lineage_refusal=lambda message: _preparation_refused(message),
968:             conflict_refusal=lambda message: _preparation_refused(message),
969:         )
970:         if projected.ordinal < captured_head.ordinal:
971:             behind = ProjectionBehind(
972:                 "the CURRENT declaration snapshot is behind the captured Arrival head"
973:             )
974:             behind.captured_head = captured_head
975:             behind.projected_head = projected
976:             raise behind
977:         basis = ReadBasis(
978:             lineage=captured_head.lineage,
979:             captured_head=captured_head,
980:             projected_through=projected,
981:             view_generation=getattr(snapshot, "view_generation", None),
982:         )
983:         before_documents = tuple(dict(document) for document in documents)
984:         if require_cache_basis:
985:             try:
986:                 local_ast = parse_vertex(
987:                     old_cache_bytes.decode("utf-8"), path=target_path
988:                 )
989:                 local_documents = tuple(
990:                     document.as_json() for document in vertex_to_documents(local_ast)
991:                 )
992:             except Exception as exc:
993:                 raise _preparation_refused(
994:                     f"current declaration cache is not valid for this splice: {exc}"
995:                 ) from exc
996:             if local_documents != before_documents:
997:                 raise _preparation_refused(
998:                     "declaration cache differs from CURRENT Arrival history; "
999:                     "convenience mutation refused"
1000:                 )
1001:         changes = tuple(diff_documents(before_documents, proposed_documents))
1002:         continuity_ticks = tuple(
1003:             snapshot.ticks(TickRequest(since=float("-inf")))
1004:         )
1005:         verified_params = collect_verified_parameter_rows(
1006:             snapshot.declaration_anchor,
1007:             facts,
1008:             target_documents=proposed_documents,
1009:             base_dir=target_path.parent,
1010:         )
1011:         analyze_boundary_continuity(
1012:             snapshot.declaration_anchor,
1013:             facts,
1014:             continuity_ticks,
1015:             target_documents=proposed_documents,
1016:             verified_params=verified_params,
1017:         )
1018:         registry_verifier = arrival_verify
1019:         if credentials is not None and credentials.mapped:
1020:             mapped_verifier = credentials.signature_verifier
1021:             if mapped_verifier is None:  # guarded by WriteCredentials construction
1022:                 raise _preparation_refused(
1023:                     "mapped declaration credentials have no signature verifier",
1024:                     captured_head=captured_head,
1025:                 )
1026:             def registry_verifier(key: str, signature: str, digest: str) -> bool:
1027:                 return mapped_verifier(
1028:                     SigningDomain.ARRIVAL, key, signature, digest
1029:                 )
1030:         key_registry, _key_evidence = key_registry_from_records(
1031:             ledger.scan(through=captured_head), registry_verifier
1032:         )
1033:         atomic_limit = ledger.capabilities().max_atomic_records
1034:     except DeclarationPreparationError:
1035:         raise
1036:     except BoundaryContinuityConflict as exc:
1037:         raise _preparation_refused(
1038:             f"boundary continuity refuses declaration preparation: {exc}",
1039:             captured_head=captured_head,
1040:         ) from exc
1041:     except Exception as exc:
1042:         raise _preparation_refused(
1043:             f"cannot establish declaration preparation basis: {exc}",
1044:             captured_head=captured_head,
1045:             projected_head=getattr(exc, "projected_head", None),
1046:         ) from exc
1047:     finally:
1048:         _close_quietly(snapshot)
1049:         _close_quietly(query)
1050:         _close_quietly(ledger)
1051: 
1052:     if not changes:
1053:         return DeclarationEditPlan(
1054:             status="noop",
1055:             target_path=target_path,
1056:             descriptor=descriptor,
1057:             lineage=captured_head.lineage,
1058:             basis=basis,
1059:             captured_head=captured_head,
1060:             before_documents=before_documents,
1061:             proposed_documents=proposed_documents,
1062:             changes=(),
1063:             authored_at=None,
1064:             observer=observer,
1065:             drafts=(),
1066:             fact_ids=(),
1067:             key_introductions=(),
1068:             old_cache_bytes=old_cache_bytes,
1069:             old_cache_sha256=hashlib.sha256(old_cache_bytes).hexdigest(),
1070:             proposed_text=proposed_text,
1071:             atomic_limit=atomic_limit,
1072:             exact_drafts=_exact_drafts(()),
1073:         )
1074: 
1075:     if credentials is None:
1076:         raise _preparation_refused(
1077:             "declaration edits require fact and Arrival signers"
1078:         )
1079:     signing_context = CapturedSigningContext(
1080:         captured_head=captured_head,
1081:         author_keys=tuple(
1082:             (
1083:                 named_observer,
1084:                 tuple(
1085:                     key
1086:                     for key, _introduced in key_registry.keys_valid_at(
1087:                         named_observer, captured_head.ordinal + 1
1088:                     )
1089:                 ),
1090:             )
1091:             for named_observer in sorted(key_registry.introductions)
1092:         ),
1093:         receipt_keys=(),
1094:     )
1095:     credential_session = CredentialResolutionSession()
1096:     authored_at = time.time()
1097:     rows: list[tuple[Any, ...]] = []
1098:     fact_ids: list[str] = []
1099:     for change in changes:
1100:         fact_id, row = _declaration_row(
1101:             change,
1102:             lineage=captured_head.lineage,
1103:             authored_at=authored_at,
1104:             observer=observer,
1105:             credentials=credentials,
1106:             registry=key_registry,
1107:             captured_head=captured_head,
1108:             fact_verify=fact_verify,
1109:             context=signing_context,
1110:             session=credential_session,
1111:         )
1112:         fact_ids.append(fact_id)
1113:         rows.append(row)
1114:     body = body_of_fact_row(rows[0]) if len(rows) == 1 else body_of_batch(rows)
1115:     envelope_kind = "fact" if len(rows) == 1 else "batch"
1116:     envelope_digest = content_commitment(
1117:         envelope_kind, authored_at, observer, "", body
1118:     )
1119:     outer_signature = _signed_with_existing_key(
1120:         credentials=credentials,
1121:         signer=credentials.arrival_signer,
1122:         verifier=arrival_verify,
1123:         registry=key_registry,
1124:         author=observer,
1125:         head=captured_head,
1126:         digest=envelope_digest,
1127:         label="Arrival declaration envelope",
1128:         domain=SigningDomain.ARRIVAL,
1129:         purpose=CredentialPurpose.AUTHORSHIP,
1130:         context=signing_context,
1131:         session=credential_session,
1132:     )
1133:     declaration_draft = RecordDraft(
1134:         kind=envelope_kind,
1135:         authored_at=authored_at,
1136:         observer=observer,
1137:         origin="",
1138:         body=body,
1139:         signature=outer_signature,
1140:     )
1141: 
1142:     proposed_keys: dict[str, str] = {}
1143:     for document in proposed_documents:
1144:         if document["kind"] != DECL_OBSERVER_DEFINED:
1145:             continue
1146:         key = document["payload"].get("key")
1147:         if key is None:
1148:             continue
1149:         if not _key_is_well_formed(key):
1150:             raise _preparation_refused(
1151:                 f"observer {document['subject']!r} carries a malformed public key"
1152:             )
1153:         proposed_keys[str(document["subject"])] = str(key)
1154:     existing_keys = {
1155:         named_observer: {key for key, _ordinal in keys}
1156:         for named_observer, keys in key_registry.introductions.items()
1157:     }
1158:     key_drafts: list[RecordDraft] = []
1159:     for named_observer, key in proposed_keys.items():
1160:         if key in existing_keys.get(named_observer, set()):
1161:             continue
1162:         key_drafts.append(
1163:             _key_draft(
1164:                 observer,
1165:                 named_observer,
1166:                 key,
1167:                 authored_at=authored_at,
1168:                 credentials=credentials,
1169:                 registry=key_registry,
1170:                 captured_head=captured_head,
1171:                 arrival_verify=arrival_verify,
1172:                 context=signing_context,
1173:                 session=credential_session,
1174:             )
1175:         )
1176: 
1177:     drafts = tuple((*key_drafts, declaration_draft))
1178:     if atomic_limit is not None and len(drafts) > atomic_limit:
1179:         raise AtomicLimitExceeded(
1180:             f"declaration edit needs {len(drafts)} Arrival records, but the "
1181:             f"backend atomic limit is {atomic_limit}; refused before append"
1182:         )
1183:     return DeclarationEditPlan(
1184:         status="planned",
1185:         target_path=target_path,
1186:         descriptor=descriptor,
1187:         lineage=captured_head.lineage,
1188:         basis=basis,
1189:         captured_head=captured_head,
1190:         before_documents=before_documents,
1191:         proposed_documents=proposed_documents,
1192:         changes=changes,
1193:         authored_at=authored_at,
1194:         observer=observer,
1195:         drafts=drafts,
1196:         fact_ids=tuple(fact_ids),
1197:         key_introductions=tuple(draft.body["observer"] for draft in key_drafts),
1198:         old_cache_bytes=old_cache_bytes,
1199:         old_cache_sha256=hashlib.sha256(old_cache_bytes).hexdigest(),
1200:         proposed_text=proposed_text,
1201:         atomic_limit=atomic_limit,
1202:         exact_drafts=_exact_drafts(drafts),
1203:         credential_bindings=credential_session.evidence,
1204:     )

1261: def _apply_declaration_edit(
1262:     registry: BackendRegistry,
1263:     plan: DeclarationEditPlan,
1264:     *,
1265:     failure_hook: Callable[[str], None] | None = None,
1266: ) -> DeclarationEditResult:
1267:     """Append one prepared declaration transition and publish its cache.
1268: 
1269:     The prepared drafts are submitted exactly once under the captured full
1270:     head. Contract refusals are zero-prefix stale refusals; transport and
1271:     witness failures retain the intent and their identity for recovery.
1272:     """
1273:     _validate_plan(plan)
1274:     if plan.status == "noop":
1275:         return _result(
1276:             plan, status="noop", head=plan.captured_head, commit=None, intent=None,
1277:             phase="noop", file_written=False, changes=plan.changes,
1278:         )
1279:     intent = declaration_intent_path(plan.target_path)
1280:     try:
1281:         if plan.target_path.read_bytes() != plan.old_cache_bytes:
1282:             raise DeclarationStale(
1283:                 "declaration cache changed after preparation; refusing append"
1284:             )
1285:     except OSError as exc:
1286:         raise DeclarationApplyError(
1287:             f"cannot recheck declaration cache before append: {exc}"
1288:         ) from exc
1289:     data = _intent_data(plan, phase="prepared")
1290:     append_drafts = _drafts_from_exact(
1291:         json.loads(plan.exact_drafts), plan.exact_drafts
1292:     )
1293:     try:
1294:         _write_exclusive_json(intent, data)
1295:     except FileExistsError as exc:
1296:         raise DeclarationApplyError(
1297:             f"unfinished declaration intent already exists: {intent}"
1298:         ) from exc
1299:     if failure_hook:
1300:         try:
1301:             failure_hook("after-intent")
1302:         except Exception as exc:
1303:             raise DeclarationApplyError(
1304:                 "declaration intent was durable but append did not start"
1305:             ) from exc
1306: 
1307:     ledger: Any | None = None
1308:     query: Any | None = None
1309:     commit: Any | None = None
1310:     head: Head | None = None
1311:     append_entered = False
1312:     try:
1313:         ledger, query = registry.open(plan.descriptor)
1314:         try:
1315:             append_entered = True
1316:             commit = ledger.append(plan.captured_head, append_drafts)
1317:         except Exception as exc:
1318:             from .arrival_contract import ContractRefusal
1319:             from .arrival_head_seam import NotWitnessed
1320: 
1321:             if isinstance(exc, ContractRefusal):
1322:                 _remove_intent(intent)
1323:                 raise DeclarationStale(
1324:                     f"declaration append refused before mutation: {exc}"
1325:                 ) from exc
1326:             if isinstance(exc, NotWitnessed):
1327:                 head = exc.head
1328:                 data["phase"] = "append-unwitnessed"
1329:                 with suppress(Exception):
1330:                     _replace_json(intent, data)
1331:                 raise DeclarationUnwitnessed(
1332:                     "declaration records committed but their witness is unknown",
1333:                     head=exc.head, commit=exc.commit, intent_path=intent,
1334:                     fact_ids=plan.fact_ids, cause=exc,
1335:                 ) from exc
1336:             data["phase"] = "append-unknown"
1337:             with suppress(Exception):
1338:                 _replace_json(intent, data)
1339:             raise DeclarationOutcomeUnknown(
1340:                 "declaration append outcome is unknown; recover the intent",
1341:                 head=None, intent_path=intent, fact_ids=plan.fact_ids, cause=exc,
1342:             ) from exc
1343:         head = commit.after
1344:         try:
1345:             data["phase"] = "appended"
1346:             _replace_json(intent, data)
1347:             if failure_hook:
1348:                 failure_hook("after-append")
1349:         except Exception as exc:
1350:             raise DeclarationCommittedIncomplete(
1351:                 "declaration append committed but append phase recording failed",
1352:                 phase="appended", head=head, commit=commit, intent_path=intent,
1353:                 fact_ids=plan.fact_ids, cause=exc,
1354:             ) from exc
1355:     except (DeclarationStale, DeclarationUnwitnessed, DeclarationOutcomeUnknown,
1356:             DeclarationCommittedIncomplete):
1357:         raise
1358:     except Exception as exc:
1359:         if not append_entered:
1360:             raise DeclarationApplyError(
1361:                 "declaration append was refused before entering the append call; "
1362:                 "prepared intent retained for inspection",
1363:             ) from exc
1364:         raise DeclarationOutcomeUnknown(
1365:             "declaration append path failed; recover the intent",

1367:         ) from exc

1370:         _close_quietly(ledger)
1371:

1373:         from .arrival_maintenance import sync_projection
1374: 
1375:         sync_projection(registry, plan.descriptor, through=head)
1376:         data["phase"] = "synced"
1377:         _replace_json(intent, data)
1378:         if failure_hook:
1379:             failure_hook("after-sync")
1380:         data["phase"] = "publishing"
1381:         _publish_cache(

1394:         ) from exc

1415: def _recover_declaration_edit(
1416:     registry: BackendRegistry,
1417:     intent: Path | str,
1418:     *,
1419:     failure_hook: Callable[[str], None] | None = None,
1420: ) -> DeclarationEditResult:
1421:     """Reconcile an edit intent by exact suffix membership, then finish cache."""
1422:     intent_path = Path(intent).resolve()
1423:     data = _read_intent(intent_path)
1424:     drafts = _drafts_from_intent(data)
1425:     plan = _plan_from_intent(data, drafts)
1426:     captured = plan.captured_head
1427:     ledger: Any | None = None
1428:     query: Any | None = None
1429:     try:
1430:         ledger, query = registry.open(plan.descriptor)
1431:         captured_at_ledger = ledger.head_at(
1432:             Watermark(lineage=captured.lineage, ordinal=captured.ordinal)
1433:         )
1434:         if captured_at_ledger != captured:
1435:             raise DeclarationStale(
1436:                 "declaration intent predecessor head hash no longer matches"
1437:             )
1438:         actual = ledger.head()
1439:         if actual == captured:
1440:             if data.get("cache_temp") or data.get("cache_backup"):
1441:                 raise DeclarationApplyError(
1442:                     "unapplied declaration intent has cache publication artifacts"
1443:                 )
1444:             if plan.target_path.read_bytes() != plan.old_cache_bytes:
1445:                 raise DeclarationStale(
1446:                     "unapplied declaration intent has competing cache bytes"
1447:                 )
1448:             _remove_intent(intent_path)
1449:             return _result(
1450:                 plan, status="not-applied", head=actual, commit=None, intent=None,
1451:                 phase="not-applied", file_written=False, changes=None,
1452:             )
1453:         first = captured.ordinal + 1
1454:         for index, draft in enumerate(drafts):
1455:             ordinal = first + index
1456:             if actual.ordinal < ordinal or not _record_matches_draft(
1457:                 ledger.read(ordinal), draft
1458:             ):
1459:                 raise DeclarationStale(
1460:                     "declaration intent does not occupy its exact expected suffix"
1461:                 )
1462:     except DeclarationStale:
1463:         raise
1464:     except Exception as exc:
1465:         raise DeclarationApplyError(
1466:             f"cannot reconcile declaration intent against Arrival custody: {exc}"
1467:         ) from exc
1468:     finally:
1469:         _close_quietly(query)
1470:         _close_quietly(ledger)
1471: 
1472:     file_written = False
1473:     try:
1474:         from .arrival_maintenance import sync_projection
1475: 
1476:         sync_projection(registry, plan.descriptor, through=actual)
1477:         data["phase"] = "synced"
1478:         _replace_json(intent_path, data)
1479:         if failure_hook:
1480:             failure_hook("after-sync")
1481:         if plan.target_path.exists():
1482:             current = plan.target_path.read_bytes()
1483:             if current == plan.proposed_text.encode("utf-8"):
1484:                 _cleanup_cache_artifacts(plan, data)
1485:                 data["phase"] = "published"
1486:                 _replace_json(intent_path, data)
1487:             elif current == plan.old_cache_bytes:
1488:                 data["phase"] = "publishing"
1489:                 _publish_cache(
1490:                     plan, intent=intent_path, intent_data=data,
1491:                     failure_hook=failure_hook,
1492:                 )
1493:                 file_written = True
1494:                 data["phase"] = "published"
1495:                 _replace_json(intent_path, data)
1496:             else:
1497:                 raise DeclarationStale(
1498:                     "competing declaration cache bytes are present; refusing overwrite"
1499:                 )
1500:         elif data.get("cache_backup"):
1501:             data["phase"] = "publishing"
1502:             _publish_cache(
1503:                 plan, intent=intent_path, intent_data=data,
1504:                 failure_hook=failure_hook,
1505:             )
1506:             file_written = True
1507:             data["phase"] = "published"
1508:             _replace_json(intent_path, data)
1509:         else:
1510:             raise DeclarationApplyError(
1511:                 "declaration cache is missing and has no detached recovery backup"
1512:             )
1513:         _remove_intent(intent_path)
1514:     except Exception as exc:
1515:         raise DeclarationCommittedIncomplete(
1516:             "declaration custody is committed but cache recovery remains incomplete",
1517:             phase=str(data.get("phase", "synced")), head=actual, commit=None,
1518:             intent_path=intent_path, fact_ids=plan.fact_ids, cause=exc,
1519:         ) from exc
1520:     return _result(
1521:         plan, status="recovered", head=actual, commit=None, intent=None,
1522:         phase="published", file_written=file_written, changes=None,
1523:     )
