40: def _sha256(path: Path) -> str:
41:     digest = hashlib.sha256()
42:     with path.open("rb") as stream:
43:         for chunk in iter(lambda: stream.read(1024 * 1024), b""):
44:             digest.update(chunk)
45:     return digest.hexdigest()

48: def _sha256_prefix(path: Path, size: int) -> str:
49:     digest = hashlib.sha256()
50:     remaining = size
51:     with path.open("rb") as stream:
52:         while remaining:

56:             digest.update(chunk)

=== libs/engine/src/engine/row_commitment.py ===
1: """Pure commitments for persisted fact and tick rows.
2: 
3: These functions have no store or connection dependency. The SQLite writer,
4: Arrival runtime planner, and readers that verify chain evidence use the same
5: row bytes so a new write path cannot quietly fork an existing chain.
6: """
7: 
8: from __future__ import annotations
9: 
10: import hashlib
11: from collections.abc import Callable
12: 
13: from .admission import _canonical_bytes
14: 
15: __all__ = ["fact_row_hash", "tick_commitment_hash", "tick_envelope", "tick_row_hash"]
16: 
17: 
18: def tick_envelope(row: tuple) -> dict:
19:     """The ten tick fields that its inner signature commits to."""
20:     return {
21:         "id": row[0], "name": row[1], "ts": row[2], "since": row[3],
22:         "origin": row[4], "payload": row[5], "prev_hash": row[6],
23:         "window_start": row[7], "fact_cursor": row[8], "window_hash": row[9],
24:     }
25: 
26: 
27: def tick_commitment_hash(
28:     row: tuple, *, canonical_bytes: Callable[[object], bytes] = _canonical_bytes
29: ) -> str:
30:     """Hash one tick's signed content, excluding its signature field."""
31:     return hashlib.sha256(canonical_bytes(tick_envelope(row))).hexdigest()
32: 
33: 
34: def tick_row_hash(
35:     row: tuple, *, canonical_bytes: Callable[[object], bytes] = _canonical_bytes
36: ) -> str:
37:     """Hash one tick row as its successor's ``prev_hash`` value."""
38:     envelope = tick_envelope(row)
39:     if len(row) > 10 and row[10] is not None:
40:         envelope["signature"] = row[10]
41:     return hashlib.sha256(canonical_bytes(envelope)).hexdigest()
42: 
43: 
44: def fact_row_hash(
45:     row: tuple, *, canonical_bytes: Callable[[object], bytes] = _canonical_bytes
46: ) -> str:
47:     """Hash one fact row for a tick window, preserving exact payload text."""
48:     envelope = {
49:         "id": row[0], "kind": row[1], "ts": row[2],
50:         "observer": row[3], "origin": row[4], "payload": row[5],
51:     }
52:     if len(row) > 6 and row[6] is not None:
53:         envelope["signature"] = row[6]
54:     return hashlib.sha256(canonical_bytes(envelope)).hexdigest()

=== libs/engine/src/engine/arrival_projection.py ===
1: """arrival_projection — the two projections of an arrival log, re-derived.
2: 
3: Cut B of the arrival substrate (decision:design/arrival-sliceB-projections):
4: an arrival-canonical store has exactly two projections — the sqlite index at
5: the sibling ``.db`` and the derived ``.jsonl`` log beside it. Both are
6: functions of the arrival log, both are rebuildable at will, and neither is
7: ever written by anything that is not deriving it.
8: 
9: Why a module and not a store method
10: -----------------------------------
11: The states re-derivation resolves are exactly the states
12: :class:`~engine.arrival_store.ArrivalStore` refuses to open on: catch-up runs
13: inside the constructor, so a method on the class would be unreachable on the
14: very stores that need it. Making the constructor tolerate the refusal would
15: mean a store that opens in a broken state, which is what the consume-or-refuse
16: posture forbids. The precedent is :mod:`engine.canonical_audit`'s "pure
17: reader, by contract" — the module that must not repair never constructs the
18: store that does. Here it is the inverse and the same rule: the module that
19: repairs never constructs the store that refuses.
20: 
21: The second consequence is the reason it is worth a module: **no write path
22: can call re-derivation as a side effect.** It is not on the object the write
23: path holds.
24: 
25: What survives a re-derivation
26: -----------------------------
27: ``facts`` and ``ticks`` are cleared and rebuilt from ordinal 0 forward —
28: never partially, because reproducing the original rowids is what keeps
29: outstanding witness positions and seals valid, and that only holds for a
30: replay from empty. The FTS tables are DROPPED for the reason
31: :meth:`engine.jsonl_store.JsonlStore._rebuild` drops them: ``DELETE FROM
32: facts`` resets sqlite's rowid counter, so a surviving FTS index would resolve
33: stale text to new facts. Search then reports ``missing``, the honest "run
34: reindex" state. Every ``store_meta`` row survives except the two this module
35: owns as projections — the resume mark and ``own_lineage``, both restamped
36: from the log.
37: 
38: Locking
39: -------
40: Re-derivation takes **the sqlite write lock only** (``BEGIN IMMEDIATE`` for
41: the whole clear-and-replay), and never the arrival append lock. Taking the
42: sqlite lock serializes it against every ``ArrivalStore._write``, so no
43: appender can interleave an index write into a half-replayed index. NOT
44: taking the append lock means an appender may land a record mid-walk; that is
45: harmless, because the walk stops at the last complete record and the stamped
46: mark names what was actually consumed. The result is the ordinary "index
47: behind the log" state, which the next catch-up tails forward. Holding the
48: append lock for an O(n) replay would block every writer in the system to buy
49: a property the mark already provides.
50: """

127: def _projects(record: dict) -> bool:
128:     """Does this record project at all — and is it a record we may project?
129: 
130:     The classification BOTH projections make, spelled once. It is the same
131:     question in both directions (an index row and a derived-log line are two
132:     encodings of one row-class body), and the two projections answering it
133:     separately is how they would come to disagree about a store.
134:

136:     bodies are not codec objects at all (the codec refuses them on unknown
137:     fields), and skipping them loses nothing: their content lives in the
138:     arrival log, which is the store. An unrecognized kind is neither class
139:     and refuses here, once, for both callers.

143:         return False

145:         _unsupported(
146:             f"record kind {kind!r} at ordinal {record['ord']} is not one "
147:             "this store's projections know how to derive — refusing rather "
148:             "than silently dropping it"

191: def line_of_record(record: dict) -> str | None:
192:     """The derived log's line for one arrival record, or None for no line.
193: 
194:     Row-class records (fact/tick/batch) each emit ONE line — the codec
195:     encoding of that record's ``body`` and of nothing else. The derived log

201:     stored TEXT and the emitted bytes are exactly what they were before the

231:     """The derived log's ordering: byte-lexicographic over the encoded line.
232: 
233:     Spelled as a function because the ORDER IS A RULE, not an incidental

240: def derived_lines(log: ArrivalLog) -> list[str]:
241:     """Every derived-log line for an arrival log, byte-lexicographically sorted.
242: 
243:     **NON-NEGOTIABLE: line order carries no meaning, and no reader may
244:     attribute one to it.** Byte sort is the only order that is a pure
245:     function of the SET: it needs no field semantics, is reproducible by
246:     ``sort(1)`` in any language, and is what lets the git merge driver
247:     produce the same bytes a fresh derivation produces without knowing any
248:     ordinal.
249: 
250:     Two consequences, stated here rather than left to be found:
251: 
252:     1. **The derived log is not a chain-verification surface.** Tick chain
253:        order is not file order — a chain walk over a byte-sorted projection
254:        would report breaks on a healthy store. Chain verification reads the
255:        arrival log.
256:     2. **A last-0.x reader rebuilding from this file folds in byte order.**
257:        That is a real interim hazard for the migration sidecar, and it is
258:        named here for that design; the sidecar's own answer is to consume
259:        the ``.arrival``, which carries the ordinal.
260: 
261:     **NON-NEGOTIABLE: derived from the arrival log alone, never from the
262:     index.** Deriving it from sqlite would make it a projection of a
263:     projection, and a poisoned index would launder itself into a second
264:     artifact.
265:     """
266:     lines = set()
267:     for record in log.walk():
268:         line = line_of_record(record)
269:         if line is not None:
270:             lines.add(line)
271:     return sorted(lines, key=sort_key)

274: def write_derived_log(canonical: Path) -> int:
275:     """Materialize the derived ``.jsonl`` projection. Returns the line count.
276: 
277:     **NON-NEGOTIABLE: on demand only, never on the append path.** Appending
278:     to a byte-sorted file is not an append, and regenerating it per emit is
279:     O(n) per write — the O(n^2) ingest shape this arc already paid for once.
280:     A lagging derived log is NOT an error: nothing in ``libs/`` reads it,
281:     because reads execute through the index in every mode (arrival law 1).
282: 
283:     Written through a temp file and renamed, so a crash mid-derivation
284:     leaves the previous projection rather than a truncated one.
285:     """
286:     canonical = Path(canonical)
287:     target = derived_log_path_for(canonical)
288:     lines = derived_lines(ArrivalLog(canonical))
289:     tmp = target.with_name(target.name + ".tmp")
290:     payload = "".join(line + "\n" for line in lines)
291:     with tmp.open("w", encoding="utf-8") as fh:
292:         fh.write(payload)
293:         fh.flush()
294:         os.fsync(fh.fileno())
295:     tmp.replace(target)
296:     return len(lines)

299: @dataclass(frozen=True)
300: class DerivedLogAgreement:
301:     """Whether the derived log and the arrival log hold the same set.
302: 
303:     There is **no staleness marker**. The offset/count triple does not get a
304:     third sibling: the derived log's currency is a SET question, and the
305:     honest answer to a set question is a set comparison, not a stamp. A
306:     marker would also reintroduce offset custody inside the mutable artifact
307:     being judged.
308:     """
309: 
310:     ok: bool
311:     missing: int
312:     """Row-class records the arrival log carries and the file does not.
313: 
314:     Structural records are outside the projection, so a healthy store is
315:     never reported as missing its genesis.
316:     """
317:     extra: int
318:     """Lines the arrival log never carried."""
319:     detail: str
320: 
321: 
322: def _derived_digests(canonical: Path) -> Counter[bytes]:
323:     """Digest every line the arrival log projects, retaining none of them.
324: 
325:     Deliberately NOT ``{_digest(line) for line in derived_lines(...)}``:
326:     :func:`derived_lines` must materialize every line because it SORTS them,
327:     and an audit that borrowed it inherited a cost it does not need. Here
328:     each line is produced, hashed, and counted in a multiset (:class:`Counter`),
329:     so what survives the walk is 32 bytes per unique digest.
330: 
331:     The multiset structure makes the audit order-insensitive while preserving
332:     multiplicity: duplicated lines in either artifact are caught as extra
333:     lines, rather than collapsing into an innocence-granting set union.
334: 
335:     The bound, stated exactly rather than loosely: peak is **flat in record
336:     count** and **linear in the size of the LARGEST single record**. What
337:     survives the walk is a 32-byte digest count per record; what is in flight is
338:     one record, which costs a small multiple of its own size because it is
339:     decoded, re-encoded and hashed. Measured: 5, 10, 20 and 40 records of
340:     1 MB each all peak at 7.02 MB, while one 4 MB record peaks at 28 MB.
341: 
342:     So "bounded by record count, never payload size" holds in the sense
343:     that matters — an audit over a big store does not grow with the store —
344:     but a single enormous record is still transiently expensive, and saying
345:     otherwise would be the kind of unbounded promise this module refuses to
346:     make elsewhere.
347:     """
348:     digests: Counter[bytes] = Counter()
349:     for record in ArrivalLog(canonical).walk():
350:         line = line_of_record(record)
351:         if line is not None:
352:             digests[_digest(line)] += 1
353:     return digests
354:

356: def audit_derived_log(canonical: Path) -> DerivedLogAgreement:
357:     """Re-derive and diff. An operation, not a promise.
358: 
359:     The repo's established contract for derived artifacts (``verify_rebirth``
360:     is the precedent). Each side's lines are hashed to 32 bytes as they are
361:     produced and two multiset differences taken, so memory is bounded by RECORD
362:     COUNT and never by total payload size — see :func:`_derived_digests` for
363:     the exact bound.
364: 
365:     The reported counts survive digest-only multisets, because a multiset difference
366:     over digests has the same cardinality as one over the lines they stand
367:     for. Nothing in :class:`DerivedLogAgreement` echoes line CONTENT — the
368:     detail names paths and counts — so the result is bounded too.
369:     """
370:     canonical = Path(canonical)
371:     target = derived_log_path_for(canonical)
372:     derived = _derived_digests(canonical)
373: 
374:     if not target.exists():
375:         missing_count = derived.total()
376:         return DerivedLogAgreement(
377:             ok=not derived,
378:             missing=missing_count,
379:             extra=0,
380:             detail=(
381:                 f"no derived log at {target}"
382:                 + ("" if not derived else f"; the arrival log projects {missing_count} line(s)")
383:             ),
384:         )
385: 
386:     present: Counter[bytes] = Counter()
387:     with target.open("rb") as fh:
388:         for raw in fh:
389:             if raw.endswith(b"\n"):
390:                 present[hashlib.sha256(raw[:-1]).digest()] += 1
391:             elif raw:
392:                 # A torn tail is not a line: it was never terminated, so it
393:                 # never claimed to be a record. Counting it as `extra` would
394:                 # report a crashed derivation as a disagreement about
395:                 # content; the honest reading is that the file is short.
396:                 pass
397: 
398:     missing_counter = derived - present
399:     extra_counter = present - derived
400:     missing = missing_counter.total()
401:     extra = extra_counter.total()
402:     ok = missing == 0 and extra == 0
403:     return DerivedLogAgreement(
404:         ok=ok,
405:         missing=missing,
406:         extra=extra,
407:         detail=(
408:             f"{target} agrees with {canonical}"
409:             if ok
410:             else (
411:                 f"{target} disagrees with {canonical}: {missing} record(s) the "
412:                 f"arrival log carries are absent, {extra} line(s) it never "
413:                 "carried are present"
414:             )
415:         ),
416:     )
417: 
418: 
419: def _digest(line: str) -> bytes:
420:     return hashlib.sha256(line.encode("utf-8")).digest()

423: def licensed_own_lineage(conn: sqlite3.Connection, lineage: str) -> str | None:
424:     """The ``own_lineage`` value this index's rows license, or None.
425: 
426:     The stamp is licensed by a **consumed ``_decl.genesis`` fact row whose
427:     id is the log's lineage** — movement 2, the declaration absorb, which is
428:     what stamps the marker on the live path. Two narrowings, both
429:     load-bearing:
430: 
431:     * NOT the mere existence of the arrival genesis at ordinal 0 (movement
432:       1). Stamping on that would flip a minted-but-never-absorbed store to
433:       "adopted", and ``witness.durable_handle`` would start emitting
434:       portable handles for stores that never opened a declaration lineage —
435:       a witness-semantics widening that is not this cut's.
436:     * NOT any ``_decl.genesis`` row. A merged store's log can carry a
437:       FOREIGN genesis row verbatim, and stamping that would mint the
438:       "marker without its genesis" corruption the declaration resolver
439:       already names.
440: 
441:     Restricted this way the stamp is a pure projection restore with zero
442:     semantic change.
443:     """
444:     from lang.document import DECL_GENESIS
445: 
446:     row = conn.execute(
447:         "SELECT id FROM facts WHERE id = ? AND kind = ?", (lineage, DECL_GENESIS)
448:     ).fetchone()
449:     return None if row is None else row[0]

452: @dataclass(frozen=True)
453: class Rederivation:
454:     """What a re-derivation did — counts, not a promise."""
455: 
456:     lineage: str
457:     ordinal: int
458:     """The last record consumed; -1 when the log carried none."""
459:     records: int
460:     facts: int
461:     ticks: int
462:     projections: tuple[str, ...]

465: def rederive_projections(
466:     canonical: Path, *, derived_log: bool = False
467: ) -> Rederivation:
468:     """Discard an arrival store's projections and rebuild them from the log.
469: 
470:     The explicit operator verb. Building an ABSENT projection stays
471:     automatic (an absent projection destroys nothing when built); an
472:     EXISTING projection carrying rows is evidence, and discarding evidence
473:     is an operator's decision — so it is spelled here, off the object the
474:     write path holds.
475: 
476:     Replays from **ordinal 0, never partially**: the rowids handed out by a
477:     replay from empty reproduce the original assignment exactly, which is
478:     what keeps outstanding witness positions and seals valid across a
479:     re-derivation. A partial rebuild would renumber.
480: 
481:     Refuses (:class:`~engine.arrival_store.ArrivalCanonicalUnsupported`)
482:     when ``own_lineage`` is present and names a lineage other than the
483:     log's — that index is some other log's projection, and silently
484:     repointing it would be the strongest lie available in this design. The
485:     check runs BEFORE anything is cleared.
486:     """
487:     canonical = Path(canonical)
488:     log = ArrivalLog(canonical)
489:     lineage = log.lineage()  # refuses when this is not an arrival log
490:     index = index_path_for(canonical)
491: 
492:     conn = sqlite3.connect(str(index))
493:     conn.isolation_level = None  # explicit transaction control
494:     try:
495:         _ensure_index_schema(conn, log)
496:         conn.execute("BEGIN IMMEDIATE")
497:         try:
498:             marker = _meta_get(conn, _OWN_LINEAGE_KEY)
499:             if marker is not None and marker != lineage:
500:                 _unsupported(
501:                     f"{index} carries an own_lineage marker of {marker!r}, "
502:                     f"but the log at {canonical} opens lineage {lineage!r} — "
503:                     "this index is some other log's projection; re-deriving "
504:                     "it would repoint an identity claim rather than rebuild "
505:                     "one"
506:                 )
507: 
508:             conn.execute("DELETE FROM facts")
509:             conn.execute("DELETE FROM ticks")
510:             conn.execute("DROP TABLE IF EXISTS facts_fts")
511:             conn.execute("DROP TABLE IF EXISTS fts_state")
512:             # Arrival exact-prefix FTS coverage is derived from these rows too.
513:             conn.execute("DROP TABLE IF EXISTS arrival_facts_fts")
514:             conn.execute("DROP TABLE IF EXISTS arrival_fts_state")
515: 
516:             records = facts = ticks = 0
517:             last: ResumeMark | None = None
518:             _, walked = log.walk_marked(None)
519:             for record, mark in walked:
520:                 ord_val = record["ord"]
521:                 for seq, (t, row) in enumerate(rows_of_record(record)):
522:                     conn.execute(
523:                         FACT_INSERT_SQL if t == "fact" else TICK_INSERT_SQL,
524:                         (*row, ord_val, seq),
525:                     )
526:                     if t == "fact":
527:                         facts += 1
528:                     else:
529:                         ticks += 1
530:                 records += 1
531:                 last = mark
532: 
533:             own = licensed_own_lineage(conn, lineage)
534:             if own is not None:
535:                 _meta_set(conn, _OWN_LINEAGE_KEY, own)
536:             if last is not None:
537:                 _stamp_mark(conn, last)
538:             conn.execute("COMMIT")
539:         except BaseException:
540:             conn.execute("ROLLBACK")
541:             raise
542:     finally:
543:         conn.close()
544: 
545:     if derived_log:
546:         # Derived from the arrival log, in its own pass, AFTER the index
547:         # transaction closed — never from the rows just written. A
548:         # projection of a projection would let a poisoned index launder
549:         # itself into a second artifact.
550:         write_derived_log(canonical)
551:     projections = ("index", "derived-log") if derived_log else ("index",)
552:     return Rederivation(
553:         lineage=lineage,
554:         ordinal=-1 if last is None else last.arrival_ordinal,
555:         records=records,
556:         facts=facts,
557:         ticks=ticks,
558:         projections=projections,
559:     )

564: 
565: def _ensure_index_schema(
566:     conn: sqlite3.Connection, log: ArrivalLog | None = None
567: ) -> None:
568:     """Make the index connectable as an index, whatever era it is from.
569: 
570:     Every statement is idempotent. The schema statements are the store
571:     class's own — imported rather than restated, so a schema column and a
572:     re-derived row can never drift apart — and the two column migrations
573:     mirror the store's, so an index written before the chain columns or the
574:     fact signature existed re-derives instead of failing on a missing
575:     column.
576:     """
577:     for stmt in _SCHEMA_STMTS:
578:         conn.execute(stmt)
579:     conn.execute(
580:         "CREATE TABLE IF NOT EXISTS store_meta (key TEXT PRIMARY KEY, value TEXT)"
581:     )
582:     fact_cols = {r[1] for r in conn.execute("PRAGMA table_info(facts)")}
583:     if "signature" not in fact_cols:
584:         conn.execute("ALTER TABLE facts ADD COLUMN signature TEXT")
585:     tick_cols = {r[1] for r in conn.execute("PRAGMA table_info(ticks)")}
586:     for col in ("prev_hash", "window_start", "fact_cursor", "window_hash", "signature"):
587:         if col not in tick_cols:
588:             conn.execute(f"ALTER TABLE ticks ADD COLUMN {col} TEXT")
589:     conn.commit()

1: """arrival_contract — the storage-neutral authority protocol.
2: 
3: Slice 2 of the arrival break (``design:arrival-break-slice2-backend-contract``).
4: The contract of record is ``docs/architecture/arrival/backend-contract.html``
5: (Draft v1); this module states its §03 core types and interfaces in Python so
6: adapters can be written against them. It designs nothing — where the two
7: disagree, the HTML is right and this file is a bug.
8:

11: separate module from the adapter: ``import engine.arrival_contract`` is what

607: def _window_hash(
608:     facts: tuple[SnapshotFact, ...],
609:     *,
610:     start: str,
611:     appended: tuple[object, ...] | None = None,
612: ) -> str:
613:     """The live tick window hash over exact snapshot rows plus this fact."""
614:     import hashlib
615: 
616:     digest = hashlib.sha256()
617:     if appended is not None and start == appended[0]:
618:         return digest.hexdigest()
619:     start_at = -1
620:     if start:
621:         for index, fact in enumerate(facts):
622:             if fact.id == start:
623:                 start_at = index
624:                 break
625:         else:
626:             raise RuntimeWriteRefused(f"tick window start {start!r} is absent from the snapshot")
627:     for fact in facts[start_at + 1 :]:
628:         if fact.payload_text is None:
629:             raise RuntimeWriteRefused("snapshot omitted exact fact payload text")
630:         digest.update(
631:             fact_row_hash(
632:                 (
633:                     fact.id,
634:                     fact.kind,
635:                     fact.ts,
636:                     fact.observer,
637:                     fact.origin,
638:                     fact.payload_text,
639:                     fact.signature,
640:                 )
641:             ).encode()
642:         )
643:     if appended is not None:
644:         digest.update(fact_row_hash(appended).encode())
645:     return digest.hexdigest()

648: def _current_tick_context(
649:     ticks: tuple[SnapshotTick, ...], *, current_fact_id: str
650: ) -> tuple[str | None, str]:
651:     """Return the predecessor identity and next tick's window start."""
652:     if not ticks:
653:         return None, ""
654:     previous = max(ticks, key=lambda tick: (tick.arrival_ordinal, tick.arrival_seq))
655:     if previous.payload_text is None:
656:         raise RuntimeWriteRefused("snapshot omitted exact tick payload text")
657:     previous_hash = tick_row_hash(
658:         (
659:             previous.id,
660:             previous.name,
661:             previous.ts,
662:             previous.since,
663:             previous.origin,
664:             previous.payload_text,
665:             previous.prev_hash,
666:             previous.window_start,
667:             previous.fact_cursor,
668:             previous.window_hash,
669:             previous.signature,
670:         )
671:     )
672:     if previous.window_hash is None:
673:         # The live store starts a new chain epoch at its current fact edge,
674:         # which is this just-planned fact; its first window is empty.
675:         return previous_hash, current_fact_id
676:     if previous.fact_cursor is None:
677:         raise RuntimeWriteRefused("chained tick has no fact cursor")
678:     return previous_hash, previous.fact_cursor

793: def _author_signature(
794:     credentials: WriteCredentials,
795:     session: CredentialResolutionSession,
796:     context: CapturedSigningContext | None,
797:     *,
798:     captured_head: Head,
799:     observer: str,
800:     domain: SigningDomain,
801:     purpose: CredentialPurpose,
802:     digest: str,
803:     legacy_signer: Callable[[str, str], str | None] | None,
804:     required: bool = False,
805: ) -> str | None:
806:     if not credentials.mapped:
807:         if legacy_signer is None:
808:             if required:
809:                 raise RuntimeWriteRefused(f"missing {domain.value} signer")

812:         if required and not signature:
813:             raise RuntimeWriteRefused(f"missing {domain.value} signature")

816:         credentials, context, captured_head=captured_head
817:     )
818:     request = CredentialRequest(
819:         namespace=credentials.binding_namespace,
820:         observer=observer,
821:         domain=domain,
822:         purpose=purpose,
823:     )
824:     return session.sign(
825:         credentials,
826:         request,
827:         digest,
828:         context=resolved_context,
829:         authorized_keys=resolved_context.keys_for_author(observer),
830:         required=required,
831:     )

1098:         vertex.kinds,
1099:         refusal=RuntimeWriteRefused,
1100:     )

1102:     if anchor.own_lineage is None or anchor.genesis is None:
1103:         raise RuntimeWriteRefused("ordinary emission requires a declaration anchor")
1104:     from .declaration import validate_arrival_declaration_anchor
1105:

1109:     existing_plan = _existing_fact_plan(

1111:         basis,

1115:         signing_context=_signing_context,
1116:         credential_session=credential_session,
1117:     )
1118:     if existing_plan is not None:
1119:         return existing_plan
1120:     facts = snapshot.facts(FactRequest(limit=None, include_internal=True, order="oldest")).items
1121:     # The neutral request is receipt-ordered, but keep this safety boundary at
1122:     # the chain calculation: window membership is coordinates, never a
1123:     # backend's event-time or identifier presentation order.
1124:     facts = tuple(sorted(facts, key=lambda row: (row.arrival_ordinal, row.arrival_seq)))
1125:     ticks = snapshot.ticks(TickRequest(since=float("-inf")))

1464: def execute_ordinary_write(
1465:     registry: BackendRegistry,
1466:     descriptor: StoreDescriptor,
1467:     plan: OrdinaryWritePlan,
1468:     *,
1469:     after_commit: Callable[[Head], object] | None = None,
1470: ) -> CommittedWrite:
1471:     """Append a previously prepared plan once against its exact full head.
1472: 
1473:     Projection publication is deliberately not attempted in 3A.  A successful
1474:     result says only that custody committed; maintenance is an explicit later
1475:     operation.  ``after_commit`` is the optional injected maintenance seam:
1476:     it receives the durable ``Commit.after`` and cannot change the append
1477:     outcome.  ``NotWitnessed`` already carries the durable commit and is
1478:     intentionally propagated rather than relabeled as a refusal.
1479:     """
1480:     from .arrival_contract import Profile
1481: 
1482:     if descriptor.role is not Profile.AUTHORITY:
1483:         raise NotAuthority("ordinary writes require an Authority descriptor role")
1484:     ledger, query = registry.open(descriptor)
1485:     try:
1486:         if not isinstance(ledger, AttestedLedger):
1487:             raise TypeError("BackendRegistry.open must return an AttestedLedger")
1488:         comparison = ledger.opened.comparison
1489:         if isinstance(comparison, PreGenesis):
1490:             if comparison.ledger_refusal is not None:
1491:                 raise comparison.ledger_refusal
1492:             raise RuntimeWriteRefused("ordinary emission requires a minted ledger")
1493:         if isinstance(comparison, Indeterminate):
1494:             raise comparison.refusal
1495:         if not isinstance(comparison, Compared):
1496:             raise TypeError(f"unsupported head comparison {comparison!r}")
1497:         if plan.already_present:
1498:             return CommittedWrite(
1499:                 commit=None,
1500:                 fact_id=plan.fact_id,
1501:                 tick_id=None,
1502:                 projection=ProjectionOutcome.NOT_REQUESTED,
1503:             )
1504:         if comparison.presented != plan.captured_head:
1505:             raise RuntimeWriteRefused("captured head changed before append")
1506:         try:
1507:             commit = ledger.append(plan.captured_head, plan.drafts)
1508:         except NotWitnessed:
1509:             raise
1510:         except ContractRefusal:
1511:             raise
1512:         except Exception as exc:
1513:             raise WriteCommitUnknown(plan, exc) from exc
1514:         if after_commit is not None:
1515:             try:
1516:                 after_commit(commit.after)
1517:             except Exception as exc:
1518:                 raise PostCommitProjectionFailed(commit, plan, exc) from exc
1519:             projection = ProjectionOutcome.SYNCED
1520:         else:
1521:             projection = ProjectionOutcome.NOT_REQUESTED
1522:         return CommittedWrite(
1523:             commit=commit,
1524:             fact_id=plan.fact_id,
1525:             tick_id=plan.tick_id,
1526:             projection=projection,
1527:         )
1528:     finally:
1529:         _close_quietly(query)
1530:         _close_quietly(ledger)

1533: def _pack_batch_drafts(
1534:     drafts: Sequence[RecordDraft],
1535:     credentials: WriteCredentials,
1536:     *,
1537:     captured_head: Head | None = None,
1538:     signing_context: CapturedSigningContext | None = None,
1539:     credential_session: CredentialResolutionSession | None = None,
1540: ) -> tuple[RecordDraft, ...]:
1541:     """Turn adjacent ordinary fact drafts into wire-valid same-author batches."""
1542:     packed: list[RecordDraft] = []
1543:     held: list[RecordDraft] = []
1544:     session = credential_session or CredentialResolutionSession()
1545: 
1546:     def flush() -> None:
1547:         if not held:
1548:             return
1549:         if len(held) == 1:
1550:             packed.append(held[0])
1551:         else:
1552:             first = held[0]
1553:             rows = [
1554:                 (
1555:                     row.body["id"],
1556:                     row.body["kind"],
1557:                     row.body["ts"],
1558:                     row.body["observer"],
1559:                     row.body["origin"],
1560:                     row.body["payload"],
1561:                     row.body.get("signature"),
1562:                 )
1563:                 for row in held
1564:             ]
1565:             body = body_of_batch(rows)
1566:             digest = content_commitment(
1567:                 "batch", first.authored_at, first.observer, first.origin, body
1568:             )
1569:             if credentials.mapped:
1570:                 if captured_head is None:
1571:                     raise CredentialBindingRefused(
1572:                         "mapped batch signing requires a captured head"
1573:                     )
1574:                 signature = _author_signature(
1575:                     credentials,
1576:                     session,
1577:                     signing_context,
1578:                     captured_head=captured_head,
1579:                     observer=first.observer,
1580:                     domain=SigningDomain.ARRIVAL,
1581:                     purpose=CredentialPurpose.AUTHORSHIP,
1582:                     digest=digest,
1583:                     legacy_signer=None,
1584:                 )
1585:             else:
1586:                 signature = (
1587:                     credentials.arrival_signer(first.observer, digest)
1588:                     if credentials.arrival_signer is not None
1589:                     else None
1590:                 )
1591:             packed.append(
1592:                 RecordDraft(
1593:                     kind="batch",
1594:                     authored_at=first.authored_at,
1595:                     observer=first.observer,
1596:                     origin=first.origin,
1597:                     body=body,
1598:                     signature=signature,
1599:                 )
1600:             )
1601:         held.clear()
1602: 
1603:     for draft in drafts:
1604:         if draft.kind == "fact" and (not held or held[0].observer == draft.observer):
1605:             held.append(draft)
1606:             continue
1607:         flush()
1608:         if draft.kind == "fact":
1609:             held.append(draft)
1610:         else:
1611:             packed.append(draft)
1612:     flush()
1613:     return tuple(packed)
