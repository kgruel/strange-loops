384:                 raise ValueError("published mapped binding disagrees with its key object")

400:     def resolve(self, request: CredentialRequest) -> ResolvedCredential | None:
401:         """Resolve one credential without locks, mkdirs, repair, or fallback."""
402:         if request.namespace != self.namespace:
403:             raise CredentialBindingRefused("request-mismatch", request=request)
404:         try:
405:             record = self._validated_binding(request.namespace, request.observer)
406:             if record is None:
407:                 pending_dir = self._subdir("pending-v1", create=False)
408:                 pending = (
409:                     None
410:                     if pending_dir is None
411:                     else self._read_json(pending_dir / f"{self._slot(request.namespace, request.observer)}.json")
412:                 )
413:                 if pending is not None:
414:                     raise CredentialBindingRefused("recovery-required", request=request)
415:                 return None
416:             pair = self._load_keypair(str(record["key_ref"]))
417:             evidence = self._evidence(request, record)
418:         except CredentialBindingRefused:
419:             raise

421:             raise CredentialBindingRefused("corrupt-binding", request=request) from exc
422:         return ResolvedCredential(
423:             evidence=evidence,
424:             sign_digest=lambda digest: ed25519.sign(
425:                 pair, digest.encode(), domain=_domain_prefix(request.domain)

463:     def _pending_by_token(self, token: str) -> dict[str, object] | None:
464:         """Find a durable token reservation while holding the provider lock."""
465:         pending_dir = self._subdir("pending-v1", create=False)
466:         if pending_dir is None:
467:             return None
468:         found: dict[str, object] | None = None
469:         for path in pending_dir.iterdir():
470:             if path.suffix != ".json":
471:                 continue
472:             record = self._read_json(path)
473:             if record is None or record.get("token") != token:
474:                 continue
475:             namespace = record.get("namespace")
476:             observer = record.get("observer")
477:             if not isinstance(namespace, str) or not isinstance(observer, str):
478:                 raise TypeError("mapped pending intent lacks exact slot identity")
479:             expected_name = f"{self._slot(namespace, observer)}.json"
480:             if path.name != expected_name:
481:                 raise ValueError("mapped pending intent is stored in the wrong slot")
482:             if found is not None:
483:                 raise BindingConflict("mapped binding token owns multiple pending slots")
484:             found = record
485:         return found

487:     def _ensure_token_index(self, intent: dict[str, object]) -> None:
488:         """Publish or validate the token index for an already durable intent."""
489:         token = str(intent["token"])
490:         intents = self._subdir("intents-v1", create=True)
491:         assert intents is not None
492:         path = intents / f"{_digest(token)}.json"
493:         if _publish_no_clobber(path, _json_bytes(intent)):
494:             return
495:         existing = self._read_json(path)
496:         if existing != intent:
497:             raise BindingConflict("mapped binding token already names another intent")

499:     def _publish_completion(self, intent: dict[str, object]) -> None:
500:         """Record a validated completed intent without deleting recovery evidence."""
501:         complete = dict(intent)
502:         complete["stage"] = "complete"
503:         intents = self._subdir("intents-v1", create=True)
504:         assert intents is not None
505:         completion = intents / f"{_digest(str(intent['token']))}.complete.json"
506:         if not _publish_no_clobber(completion, _json_bytes(complete)):
507:             existing = self._read_json(completion)
508:             if existing != complete:
509:                 raise BindingConflict("mapped binding completion receipt conflicts with its intent")

511:     def _assert_published_intent(
512:         self,
513:         published: dict[str, object],
514:         intent: dict[str, object],
515:         *,
516:         require_token_index: bool = True,
517:     ) -> None:
518:         """Ensure a material-valid slot still names this exact retained intent."""
519:         for name in ("namespace", "observer", "key_ref", "provenance"):
520:             if published.get(name) != intent.get(name):
521:                 raise BindingConflict("published mapped binding contradicts its retained intent")
522:         expected_public = intent.get("expected_public_key")
523:         if expected_public is not None and published.get("public_key") != expected_public:
524:             raise BindingConflict("published mapped binding contradicts its expected public key")
525:         intents = self._subdir("intents-v1", create=False)
526:         if intents is None:
527:             if require_token_index:
528:                 raise BindingConflict("published mapped binding has no token-index evidence")
529:             return
530:         indexed = self._read_json(intents / f"{_digest(str(intent['token']))}.json")
531:         if require_token_index and indexed != intent:
532:             raise BindingConflict("published mapped binding contradicts its token index")
533:         completion = self._read_json(
534:             intents / f"{_digest(str(intent['token']))}.complete.json"
535:         )
536:         if completion is not None:
537:             complete = dict(intent)
538:             complete["stage"] = "complete"
539:             if completion != complete:
540:                 raise BindingConflict("published mapped binding contradicts its completion receipt")

542:     def _intent_for(
543:         self, observer: str, *, token: str, provenance: str, key_ref: str | None = None,
544:         expected_public_key: str | None = None,
545:     ) -> dict[str, object]:
546:         intents = self._subdir("intents-v1", create=False)
547:         token_record = (
548:             None if intents is None else self._read_json(intents / f"{_digest(token)}.json")
549:         )
550:         if token_record is not None and (
551:             token_record.get("namespace") != self.namespace
552:             or token_record.get("observer") != observer
553:         ):
554:             raise BindingConflict("mapped binding token already names another intent")
555:         pending_for_token = self._pending_by_token(token)
556:         if pending_for_token is not None:
557:             if token_record is not None and token_record != pending_for_token:
558:                 raise BindingConflict("mapped binding pending slot contradicts its token index")
559:             if (
560:                 pending_for_token.get("namespace") != self.namespace
561:                 or pending_for_token.get("observer") != observer
562:             ):
563:                 raise BindingConflict("mapped binding token already owns another pending slot")
564:         completed = self._validated_binding(self.namespace, observer)
565:         if completed is not None:
566:             if pending_for_token is not None:
567:                 # The pending marker may be the only durable token evidence
568:                 # after interruption between its publication and token-index
569:                 # publication. Never restore that index until the published
570:                 # slot proves it still names this exact intent.
571:                 self._assert_published_intent(
572:                     completed, pending_for_token, require_token_index=False
573:                 )
574:                 self._ensure_token_index(pending_for_token)
575:                 self._assert_published_intent(completed, pending_for_token)
576:                 self._publish_completion(pending_for_token)
577:             if (
578:                 (expected_public_key is not None and completed["public_key"] != expected_public_key)
579:                 or (key_ref is not None and completed["key_ref"] != key_ref)
580:                 or (
581:                     token_record is not None
582:                     and (
583:                         token_record.get("provenance") != provenance
584:                         or token_record.get("key_ref") != completed["key_ref"]
585:                     )
586:                 )
587:             ):
588:                 raise BindingConflict("existing mapped binding has a different public key")
589:             return {
590:                 "completed": True,
591:                 "key_ref": completed["key_ref"],
592:                 "public_key": completed["public_key"],
593:                 "provenance": completed["provenance"],
594:                 "token": token,
595:             }
596:         pending_dir = self._subdir("pending-v1", create=False)
597:         pending = (
598:             None
599:             if pending_dir is None
600:             else self._read_json(
601:                 pending_dir / f"{self._slot(self.namespace, observer)}.json"
602:             )
603:         )
604:         if pending is not None:
605:             if pending_for_token is not None and pending_for_token != pending:
606:                 raise BindingConflict("mapped binding slot contradicts its token reservation")
607:             if pending.get("token") != token:
608:                 raise BindingRecoveryRequired(
609:                     "a different durable token owns this mapped binding slot"
610:                 )
611:             if (
612:                 pending.get("namespace") != self.namespace
613:                 or pending.get("observer") != observer
614:                 or pending.get("provenance") != provenance
615:                 or (key_ref is not None and pending.get("key_ref") != key_ref)
616:                 or (
617:                     expected_public_key is not None
618:                     and pending.get("expected_public_key") != expected_public_key
619:                 )
620:             ):
621:                 raise BindingConflict("mapped binding token was reused with different intent")
622:             self._ensure_token_index(pending)
623:             return pending
624:         intent = self._new_intent(
625:             observer,
626:             token=token,
627:             provenance=provenance,
628:             key_ref=key_ref,
629:             expected_public_key=expected_public_key,
630:         )
631:         pending_dir = self._subdir("pending-v1", create=True)
632:         assert pending_dir is not None
633:         # The slot marker contains the complete intent. If a process dies before
634:         # the token index is linked, callers which know observer/token can still
635:         # recover coherently; no directory scan or candidate deletion occurs.
636:         if not _publish_no_clobber(
637:             pending_dir / f"{self._slot(self.namespace, observer)}.json", _json_bytes(intent)
638:         ):
639:             return self._intent_for(
640:                 observer,
641:                 token=token,
642:                 provenance=provenance,
643:                 key_ref=key_ref,
644:                 expected_public_key=expected_public_key,
645:             )
646:         self._ensure_token_index(intent)
647:         return intent

649:     def _pending_for_recovery(self, observer: str, token: str) -> dict[str, object]:
650:         pending_dir = self._subdir("pending-v1", create=False)
651:         if pending_dir is None:
652:             raise BindingRecoveryRequired("no durable mapped binding intent exists")
653:         pending = self._read_json(
654:             pending_dir / f"{self._slot(self.namespace, observer)}.json"
655:         )
656:         if pending is None:
657:             raise BindingRecoveryRequired("no durable mapped binding intent exists")
658:         if (
659:             pending.get("schema") != _INTENT_SCHEMA
660:             or pending.get("namespace") != self.namespace
661:             or pending.get("observer") != observer
662:             or pending.get("token") != token
663:             or pending.get("stage") != "pending"
664:             or pending.get("provenance") not in {"created", "legacy-import", "existing-ref"}
665:             or not isinstance(pending.get("key_ref"), str)
666:         ):
667:             raise BindingRecoveryRequired("mapped binding recovery token does not match a pending intent")
668:         if self._pending_by_token(token) != pending:
669:             raise BindingConflict("mapped binding token owns a contradictory pending slot")
670:         self._key_dir(str(pending["key_ref"]))
671:         return pending

673:     def _hook(self, phase: str, intent: dict[str, object]) -> None:
674:         if self._creation_hook is not None:
675:             self._creation_hook(phase, dict(intent))

677:     def _ensure_created_key(self, intent: dict[str, object]):
678:         key_ref = str(intent["key_ref"])
679:         keys = self._subdir("keys-v1", create=True)
680:         assert keys is not None
681:         key_dir = keys / key_ref
682:         mode = _lstat_kind(key_dir)
683:         created = False
684:         if mode is None:
685:             _directory(key_dir, create=True)
686:             # A newly private directory is the only place convenience creation
687:             # is permitted. Existing managed objects are always loaded/checked.
688:             pair = ed25519.load_or_generate(key_dir)
689:             created = True
690:         else:
691:             if not S_ISDIR(mode):
692:                 raise ValueError("mapped custody candidate key path is not a directory")
693:             private = key_dir / "ed25519.key"
694:             public = key_dir / "ed25519.pub"
695:             if _lstat_kind(private) is None:
696:                 # This intent exclusively reserved the candidate ref. It is
697:                 # safe to resume a crash before the private object existed.
698:                 if _lstat_kind(public) is not None:
699:                     raise ValueError("incomplete candidate has public evidence without private key")
700:                 pair = ed25519.load_or_generate(key_dir)
701:                 created = True
702:             else:
703:                 pair = self._load_keypair(key_ref)
704:                 if _lstat_kind(public) is None:
705:                     # A crash between durable private and public publication is
706:                     # resumable only for this retained, provider-owned intent.
707:                     _publish_no_clobber(public, (pair.public_b64 + "\n").encode(), mode=0o644)
708:                     _fsync_directory(key_dir)
709:                 pair = self._load_keypair(key_ref, require_public_file=True)
710:         self._hook("key-published", intent)
711:         return pair, created

713:     def _copy_key(self, source: Path, intent: dict[str, object]):
714:         source_pair = self._load_external_key(source)
715:         keys = self._subdir("keys-v1", create=True)
716:         assert keys is not None
717:         key_dir = keys / str(intent["key_ref"])
718:         if _lstat_kind(key_dir) is None:
719:             _directory(key_dir, create=True)
720:             _publish_no_clobber(
721:                 key_dir / "ed25519.key", (source / "ed25519.key").read_bytes()
722:             )
723:             _publish_no_clobber(
724:                 key_dir / "ed25519.pub", (source_pair.public_b64 + "\n").encode(), mode=0o644
725:             )
726:             _fsync_directory(key_dir)
727:             created = True
728:         else:
729:             source_pair = self._load_keypair(
730:                 str(intent["key_ref"]), require_public_file=True
731:             )
732:             created = False
733:         candidate = self._load_keypair(str(intent["key_ref"]), require_public_file=True)
734:         if candidate.public_b64 != source_pair.public_b64:
735:             raise BindingConflict("copied candidate disagrees with the validated source key")
736:         self._hook("key-published", intent)
737:         return candidate, created

739:     def _load_external_key(self, key_dir: Path):
740:         mode = _lstat_kind(key_dir)
741:         if mode is None or not S_ISDIR(mode):
742:             raise FileNotFoundError(key_dir)
743:         private = key_dir / "ed25519.key"
744:         _regular(private)
745:         pair = ed25519.load(key_dir)
746:         public = key_dir / "ed25519.pub"
747:         if _regular(public, required=False) and public.read_text(encoding="utf-8") != pair.public_b64 + "\n":
748:             raise ValueError("source custody public evidence disagrees with private key")
749:         return pair

751:     def _publish_binding(self, intent: dict[str, object], pair) -> tuple[dict[str, object], bool]:
752:         expected = intent.get("expected_public_key")
753:         if expected is not None and pair.public_b64 != expected:
754:             raise BindingConflict("candidate key does not match expected public key")
755:         record: dict[str, object] = {
756:             "schema": _SCHEMA,
757:             "namespace": self.namespace,
758:             "observer": intent["observer"],
759:             "key_ref": intent["key_ref"],
760:             "algorithm": "ed25519",
761:             "public_key": pair.public_b64,
762:             "provenance": intent["provenance"],
763:         }
764:         bindings = self._subdir("bindings-v1", create=True)
765:         assert bindings is not None
766:         published = _publish_no_clobber(
767:             bindings / f"{self._slot(self.namespace, str(intent['observer']))}.json", _json_bytes(record)
768:         )
769:         if not published:
770:             existing = self._binding_record(self.namespace, str(intent["observer"]))
771:             assert existing is not None
772:             if existing["public_key"] != pair.public_b64:
773:                 raise BindingConflict("another binding won with a different public key")
774:             return existing, False
775:         self._hook("binding-published", intent)
776:         # Intent files are retained as durable recovery evidence. The marker is
777:         # deliberately not removed automatically, avoiding unsafe cleanup races.
778:         self._publish_completion(intent)
779:         return record, True

781:     def _mutate(
782:         self,
783:         observer: str,
784:         *,
785:         token: str,
786:         provenance: str,
787:         key_ref: str | None = None,
788:         expected_public_key: str | None = None,
789:         source_key_dir: Path | None = None,
790:         _locked: bool = False,
791:     ) -> BindingCreationResult:
792:         with (nullcontext() if _locked else self._mutation_lock()):
793:             intent = self._intent_for(
794:                 observer,
795:                 token=token,
796:                 provenance=provenance,
797:                 key_ref=key_ref,
798:                 expected_public_key=expected_public_key,
799:             )
800:             if intent.get("completed"):
801:                 return BindingCreationResult(
802:                     namespace=self.namespace,
803:                     observer=observer,
804:                     key_ref=str(intent["key_ref"]),
805:                     public_key=str(intent["public_key"]),
806:                     provenance=str(intent["provenance"]),
807:                     token=token,
808:                     binding_created=False,
809:                     key_created=False,
810:                 )
811:             phase = "intent-published"
812:             try:
813:                 self._hook(phase, intent)
814:                 if source_key_dir is None:
815:                     phase = "key-published"
816:                     pair, key_created = self._ensure_created_key(intent)
817:                 else:
818:                     phase = "key-published"
819:                     pair, key_created = self._copy_key(source_key_dir, intent)
820:                 phase = "binding-published"
821:                 record, binding_created = self._publish_binding(intent, pair)
822:             except BaseException as exc:
823:                 if not isinstance(exc, Exception):
824:                     raise
825:                 raise BindingMutationIncomplete(
826:                     namespace=self.namespace,
827:                     observer=observer,
828:                     token=token,
829:                     key_ref=str(intent["key_ref"]),
830:                     phase=phase,
831:                 ) from exc
832:             return BindingCreationResult(
833:                 namespace=self.namespace,
834:                 observer=observer,
835:                 key_ref=str(record["key_ref"]),
836:                 public_key=str(record["public_key"]),
837:                 provenance=str(record["provenance"]),
838:                 token=token,
839:                 binding_created=binding_created,
840:                 key_created=key_created,

847:     def recover_binding(self, observer: str, *, token: str) -> BindingCreationResult:
848:         """Resume the exact token's durable binding intent without a new key."""
849:         _exact(observer, "observer")
850:         _exact(token, "token")
851:         with self._mutation_lock():
852:             intent = self._pending_for_recovery(observer, token)
853:             phase = "intent-published"
854:             try:
855:                 # A published slot is authority over its candidate. Validate it
856:                 # against the retained pending marker so a missing/moved key can
857:                 # never be regenerated under an already-named reference.
858:                 published = self._binding_record(self.namespace, observer)
859:                 if published is not None:
860:                     self._validated_binding(self.namespace, observer)
861:                     self._assert_published_intent(
862:                         published, intent, require_token_index=False
863:                     )
864:                     # Restore the token index only after the material-valid
865:                     # slot proves it is the retained intent's exact binding.
866:                     self._ensure_token_index(intent)
867:                     self._assert_published_intent(published, intent)
868:                     phase = "binding-published"
869:                     self._publish_completion(intent)
870:                     return BindingCreationResult(
871:                         namespace=self.namespace,
872:                         observer=observer,
873:                         key_ref=str(published["key_ref"]),
874:                         public_key=str(published["public_key"]),
875:                         provenance=str(published["provenance"]),
876:                         token=str(intent["token"]),
877:                         binding_created=False,
878:                         key_created=False,
879:                     )
880:                 self._ensure_token_index(intent)
881:                 provenance = str(intent["provenance"])
882:                 if provenance == "created":
883:                     phase = "key-published"
884:                     pair, key_created = self._ensure_created_key(intent)
885:                 else:
886:                     # Import and existing-reference intents have already selected a
887:                     # durable candidate. Recovery never reopens the legacy source
888:                     # or creates a replacement key under the reserved reference.
889:                     key_ref = str(intent["key_ref"])
890:                     try:
891:                         pair = self._load_keypair(key_ref)
892:                     except FileNotFoundError as exc:
893:                         if provenance == "legacy-import":
894:                             raise BindingRecoveryRequired(
895:                                 "legacy import candidate is absent; repeat the explicit import with this token"
896:                             ) from exc
897:                         raise
898:                     expected = intent.get("expected_public_key")
899:                     if expected is not None and pair.public_b64 != expected:
900:                         raise BindingConflict("recovery candidate disagrees with its intent")
901:                     key_dir = self._key_dir(key_ref)
902:                     public = key_dir / "ed25519.pub"
903:                     if _lstat_kind(public) is None:
904:                         if provenance != "legacy-import":
905:                             raise BindingRecoveryRequired(
906:                                 "existing-reference candidate lacks durable public evidence"
907:                             )
908:                         _publish_no_clobber(
909:                             public, (pair.public_b64 + "\n").encode(), mode=0o644
910:                         )
911:                         _fsync_directory(key_dir)
912:                     pair = self._load_keypair(key_ref, require_public_file=True)
913:                     key_created = False
914:                 phase = "binding-published"
915:                 record, binding_created = self._publish_binding(intent, pair)
916:                 return BindingCreationResult(
917:                     namespace=self.namespace,
918:                     observer=observer,
919:                     key_ref=str(record["key_ref"]),
920:                     public_key=str(record["public_key"]),
921:                     provenance=str(record["provenance"]),
922:                     token=token,
923:                     binding_created=binding_created,
924:                     key_created=key_created,
925:                 )
926:             except BaseException as exc:
927:                 if not isinstance(exc, Exception):
928:                     raise
929:                 if isinstance(
930:                     exc, (BindingConflict, BindingRecoveryRequired, FileNotFoundError, ValueError)
931:                 ):
932:                     raise
933:                 raise BindingMutationIncomplete(
934:                     namespace=self.namespace,
935:                     observer=observer,
936:                     token=token,
937:                     key_ref=str(intent["key_ref"]),
938:                     phase=phase,

===== libs/custody/README.md =====
1: # custody
2: 
3: Loops signing composition — the store's at-rest signing format, below every writer.
4: 
5: Owns the `loops-tick-v1` / `loops-fact-v1` domain-separation constants, the
6: custody-co-located `keys/` layout (per-observer nesting + flat self-observer
7: back-compat), the signer/verifier builders, and `ensure_signing_key`. Observer
8: names remain exact identities: a filesystem spelling collision refuses rather
9: than borrowing another observer's key. A renamed vertex may retain a nested
10: self key; all signer domains and `ensure_signing_key` select that one key.
11: Identical flat and nested self copies are one self key; conflicting key material
12: refuses until repaired. A differently spelled nested observer remains a
13: distinct observer and is never used as the self fallback. Observer components
14: cannot name the reserved `ed25519.key` or `ed25519.pub` custody files when
15: mapped to directories. This is a local custody-layout restriction, not an
16: observer label restriction in the event protocol. A self observer with such a
17: stem uses only the flat layout.
18: 
19: Signer loading never creates keys. Concurrent explicit key creation publishes
20: one complete private key; all callers load that key and derive its public file.
21: 
22: Any client that emits signed facts or ticks must compose identically or
23: `sl store verify` reports breaks — that's why this is a lib, not CLI code.
24: Depends on `sign` (loops-agnostic Ed25519 primitives) and `engine`
25: (store-canonical declaration resolution). The engine itself never imports
26: this — it takes signer callables by injection.
27: 
28: 
29: ## Explicit mapped bindings
30: 
31: `MappedCredentialProvider(root, namespace=..., receipt_observer=...)` provides
32: an opt-in alternative to locator-based custody. Exact namespace/observer
33: strings select an opaque managed key reference recorded with its public key.
34: All three signing domains use that binding with separate domain prefixes.
35: Hashed filesystem slots are storage details; labels retain their exact case,
36: Unicode spelling, and non-path characters. A binding does not authorize an
37: Arrival claim by itself: the engine separately checks captured history.
38: 
39: `for_write` returns lazy configuration. `resolve` loads and validates the
40: binding, private-derived public material, and any existing public convenience
41: file without creating or repairing anything. Missing unbound identities can
42: remain unsigned; malformed mappings, conflicting public material, missing
43: referenced private keys, and filesystem aliases refuse.
44: 
45: Explicit mutations are `create_binding(observer, token=...)`,
46: `import_legacy(vertex_path, observer, token=...)`, and
47: `bind_existing_ref(observer, key_ref, expected_public_key, token=...)`.
48: Import copies private material into managed custody and preserves the original
49: files. Reusing a managed reference is deliberate and does not introduce its
50: public key into an Arrival lineage. Existing bindings are not silently
51: replaced; local rotation is a separate future operation.
52: 
53: Mutations use a provider-wide POSIX advisory lock, durable intents, and
54: complete-file publication with no-clobber links. The supported filesystem must
55: provide those lock, link, and fsync semantics. Concurrent creators converge
56: on one binding; resolution remains read-only. There is no automatic orphan
57: cleanup or key deletion.
58: 
59: Keep the operation token when a mutation is interrupted.
60: `BindingMutationIncomplete` retains namespace, observer, token, candidate key
61: reference, and phase; the binding may already have been published.
62: `recover_binding(observer, token=...)` reconciles the retained intent and
63: candidate. It never replaces a missing key referenced by a published binding.
64: Explicit recovery may restore a missing token index from its retained intent
65: before a later candidate-key refusal; it does not claim zero filesystem effects
66: or create a key/binding in that case.
67: An import interrupted before copying private material needs the original
68: source re-presented with the same import token and expected public material;
69: recovery cannot invent that key. A completed imported candidate can recover
70: without the old source directory. Retrying with a different token does not
71: bypass an unfinished operation. Creation results report the returned binding
72: and effects of that invocation; they contain no private key material.
