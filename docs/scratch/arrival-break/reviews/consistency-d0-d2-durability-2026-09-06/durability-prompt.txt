144: def _directory(path: Path, *, create: bool = False) -> None:
145:     """Validate a directory and durably publish newly needed ancestors.
146: 
147:     Mutation callers pass ``create=True``.  It fsyncs each parent even when a
148:     prior interrupted attempt already made the child, so retry never assumes a
149:     successful ``mkdir`` made its directory entry durable.
150:     """
151:     mode = _lstat_kind(path)
152:     if mode is None:
153:         if not create:
154:             raise FileNotFoundError(path)
155:         parent = path.parent
156:         # The configured provider root may sit below a trusted alias such as
157:         # macOS /tmp. Recurse only through missing ancestors; existing parent
158:         # aliases retain the longstanding caller-configured-root behavior.
159:         if parent != path and _lstat_kind(parent) is None:
160:             _directory(parent, create=True)
161:         try:
162:             path.mkdir(exist_ok=False)
163:         except FileExistsError:
164:             # Another process may have created it between lstat and mkdir.
165:             # Recheck its real type below; never treat a symlink as convergence.
166:             pass
167:         mode = _lstat_kind(path)
168:     if mode is None or not S_ISDIR(mode):
169:         raise ValueError(f"mapped custody path is not a directory: {path}")
170:     if create:
171:         _fsync_parent_chain(path)
172: 
173: 
174: def _fsync_parent_chain(path: Path) -> None:
175:     """Durably retain ``path``'s entry without reclassifying trusted parents.
176: 
177:     ``path`` itself is always lstat-validated by :func:`_directory`, so a
178:     provider-owned root or subdirectory can never converge through a symlink.
179:     Its already-existing ancestors, however, may include a caller-configured
180:     alias such as macOS ``/tmp``.  We deliberately open those ancestors only
181:     to fsync their directory entries; we do not apply the provider-owned
182:     no-symlink policy to them.
183: 
184:     Walking the whole lexical parent chain also makes a retry safe after an
185:     interruption just after ``mkdir A/B`` but before fsyncing ``A``'s parent:
186:     the retry may find both directories already present, yet still flushes the
187:     old ``A`` entry before it relies on ``B``.
188:     """
189:     parent = path.parent
190:     while parent != parent.parent:
191:         _fsync_directory(parent)
192:         parent = parent.parent
193:     _fsync_directory(parent)

196: def _fsync_directory(path: Path) -> None:
197:     fd = os.open(path, os.O_RDONLY)
198:     try:
199:         os.fsync(fd)
200:     finally:
201:         os.close(fd)

204: def _publish_no_clobber(path: Path, payload: bytes, *, mode: int = 0o600) -> bool:
205:     """Atomically publish bytes, returning false if another complete file won."""
206:     _directory(path.parent, create=True)
207:     fd, raw = tempfile.mkstemp(prefix=f".{path.name}.", dir=path.parent)

217:         except FileExistsError:
218:             return False

486:         key_ref = key_ref or uuid.uuid4().hex
487:         self._key_dir(key_ref)
488:         return {
489:             "schema": _INTENT_SCHEMA,
490:             "namespace": self.namespace,
491:             "observer": observer,
492:             "token": token,
493:             "key_ref": key_ref,
494:             "provenance": provenance,
495:             "expected_public_key": expected_public_key,
496:             "stage": "pending",
497:         }

499:     def _pending_by_token(self, token: str) -> dict[str, object] | None:
500:         """Find a durable token reservation while holding the provider lock."""
501:         pending_dir = self._subdir("pending-v1", create=False)
502:         if pending_dir is None:
503:             return None
504:         found: dict[str, object] | None = None
505:         for path in pending_dir.iterdir():
506:             if path.suffix != ".json":
507:                 continue
508:             record = self._read_json(path)
509:             if record is None or record.get("token") != token:
510:                 continue
511:             namespace = record.get("namespace")
512:             observer = record.get("observer")
513:             if not isinstance(namespace, str) or not isinstance(observer, str):
514:                 raise TypeError("mapped pending intent lacks exact slot identity")
515:             expected_name = f"{self._slot(namespace, observer)}.json"
516:             if path.name != expected_name:
517:                 raise ValueError("mapped pending intent is stored in the wrong slot")
518:             if found is not None:
519:                 raise BindingConflict("mapped binding token owns multiple pending slots")
520:             found = record
521:         return found

523:     def _ensure_token_index(self, intent: dict[str, object]) -> None:
524:         """Publish or validate the token index for an already durable intent."""
525:         token = str(intent["token"])
526:         intents = self._subdir("intents-v1", create=True)
527:         assert intents is not None
528:         path = intents / f"{_digest(token)}.json"
529:         if _publish_no_clobber(path, _json_bytes(intent)):
530:             return
531:         existing = self._read_json(path)
532:         if existing != intent:
533:             raise BindingConflict("mapped binding token already names another intent")
534:         _fsync_directory(intents)
535: 
536:     def _publish_completion(self, intent: dict[str, object]) -> None:
537:         """Record a validated completed intent without deleting recovery evidence."""
538:         complete = dict(intent)
539:         complete["stage"] = "complete"
540:         intents = self._subdir("intents-v1", create=True)
541:         assert intents is not None
542:         completion = intents / f"{_digest(str(intent['token']))}.complete.json"
543:         if not _publish_no_clobber(completion, _json_bytes(complete)):
544:             existing = self._read_json(completion)
545:             if existing != complete:
546:                 raise BindingConflict("mapped binding completion receipt conflicts with its intent")
547:             _fsync_directory(intents)

549:     def _assert_published_intent(
550:         self,
551:         published: dict[str, object],
552:         intent: dict[str, object],
553:         *,
554:         require_token_index: bool = True,
555:     ) -> None:
556:         """Ensure a material-valid slot still names this exact retained intent."""
557:         for name in ("namespace", "observer", "key_ref", "provenance"):
558:             if published.get(name) != intent.get(name):
559:                 raise BindingConflict("published mapped binding contradicts its retained intent")
560:         expected_public = intent.get("expected_public_key")
561:         if expected_public is not None and published.get("public_key") != expected_public:
562:             raise BindingConflict("published mapped binding contradicts its expected public key")
563:         intents = self._subdir("intents-v1", create=False)
564:         if intents is None:
565:             if require_token_index:
566:                 raise BindingConflict("published mapped binding has no token-index evidence")
567:             return
568:         indexed = self._read_json(intents / f"{_digest(str(intent['token']))}.json")
569:         if require_token_index and indexed != intent:
570:             raise BindingConflict("published mapped binding contradicts its token index")
571:         completion = self._read_json(
572:             intents / f"{_digest(str(intent['token']))}.complete.json"
573:         )
574:         if completion is not None:
575:             complete = dict(intent)
576:             complete["stage"] = "complete"
577:             if completion != complete:
578:                 raise BindingConflict("published mapped binding contradicts its completion receipt")

580:     def _intent_for(
581:         self, observer: str, *, token: str, provenance: str, key_ref: str | None = None,
582:         expected_public_key: str | None = None,
583:     ) -> dict[str, object]:
584:         intents = self._subdir("intents-v1", create=False)
585:         token_record = (
586:             None if intents is None else self._read_json(intents / f"{_digest(token)}.json")
587:         )
588:         if token_record is not None and (
589:             token_record.get("namespace") != self.namespace
590:             or token_record.get("observer") != observer
591:         ):
592:             raise BindingConflict("mapped binding token already names another intent")
593:         pending_for_token = self._pending_by_token(token)
594:         if pending_for_token is not None:
595:             if token_record is not None and token_record != pending_for_token:
596:                 raise BindingConflict("mapped binding pending slot contradicts its token index")
597:             if (
598:                 pending_for_token.get("namespace") != self.namespace
599:                 or pending_for_token.get("observer") != observer
600:             ):
601:                 raise BindingConflict("mapped binding token already owns another pending slot")
602:         pending_dir = self._subdir("pending-v1", create=False)
603:         slot_pending = (
604:             None
605:             if pending_dir is None
606:             else self._read_json(
607:                 pending_dir / f"{self._slot(self.namespace, observer)}.json"
608:             )
609:         )
610:         completed = self._validated_binding(self.namespace, observer)
611:         if completed is not None:
612:             indexed_slot: dict[str, object] | None = None
613:             if slot_pending is not None:
614:                 # A caller may legitimately retry an already-completed binding
615:                 # under a fresh operation token. The selected published material
616:                 # must still agree with this slot's original durable intent.
617:                 self._assert_published_intent(
618:                     completed, slot_pending, require_token_index=False
619:                 )
620:                 indexed_slot = (
621:                     None
622:                     if intents is None
623:                     else self._read_json(
624:                         intents / f"{_digest(str(slot_pending['token']))}.json"
625:                     )
626:                 )
627:                 if indexed_slot is not None:
628:                     self._assert_published_intent(completed, slot_pending)
629:             if pending_for_token is not None:
630:                 # The pending marker may be the only durable token evidence
631:                 # after interruption between its publication and token-index
632:                 # publication. Never restore that index until the published
633:                 # slot proves it still names this exact intent.
634:                 self._assert_published_intent(
635:                     completed, pending_for_token, require_token_index=False
636:                 )
637:             if (
638:                 (expected_public_key is not None and completed["public_key"] != expected_public_key)
639:                 or (key_ref is not None and completed["key_ref"] != key_ref)
640:                 or (
641:                     token_record is not None
642:                     and (
643:                         token_record.get("provenance") != provenance
644:                         or token_record.get("key_ref") != completed["key_ref"]
645:                     )
646:                 )
647:             ):
648:                 raise BindingConflict("existing mapped binding has a different public key")
649:             bindings = self._subdir("bindings-v1", create=False)
650:             assert bindings is not None
651:             # These writes all precede a completed result. Their contents have
652:             # been checked above, so a retry can now safely make any link whose
653:             # prior fsync was interrupted durable.
654:             _fsync_directory(bindings)
655:             _fsync_directory(self._key_dir(str(completed["key_ref"])))
656:             if slot_pending is not None:
657:                 assert pending_dir is not None
658:                 _fsync_directory(pending_dir)
659:                 if intents is not None and indexed_slot is not None:
660:                     _fsync_directory(intents)
661:             if pending_for_token is not None:
662:                 # The pending marker may be the only durable token evidence
663:                 # after interruption between its publication and token-index
664:                 # publication. Restore that index only after the published
665:                 # slot has passed the retained-intent check above.
666:                 self._ensure_token_index(pending_for_token)
667:                 self._assert_published_intent(completed, pending_for_token)
668:                 self._publish_completion(pending_for_token)
669:             return {
670:                 "completed": True,
671:                 "key_ref": completed["key_ref"],
672:                 "public_key": completed["public_key"],
673:                 "provenance": completed["provenance"],
674:                 "token": token,
675:             }
676:         pending = slot_pending
677:         if pending is not None:
678:             if pending_for_token is not None and pending_for_token != pending:
679:                 raise BindingConflict("mapped binding slot contradicts its token reservation")
680:             if pending.get("token") != token:
681:                 raise BindingRecoveryRequired(
682:                     "a different durable token owns this mapped binding slot"
683:                 )
684:             if (
685:                 pending.get("namespace") != self.namespace
686:                 or pending.get("observer") != observer
687:                 or pending.get("provenance") != provenance
688:                 or (key_ref is not None and pending.get("key_ref") != key_ref)
689:                 or (
690:                     expected_public_key is not None
691:                     and pending.get("expected_public_key") != expected_public_key
692:                 )
693:             ):
694:                 raise BindingConflict("mapped binding token was reused with different intent")
695:             assert pending_dir is not None
696:             _fsync_directory(pending_dir)
697:             self._ensure_token_index(pending)
698:             return pending
699:         intent = self._new_intent(
700:             observer,
701:             token=token,
702:             provenance=provenance,
703:             key_ref=key_ref,
704:             expected_public_key=expected_public_key,
705:         )
706:         pending_dir = self._subdir("pending-v1", create=True)
707:         assert pending_dir is not None
708:         # The slot marker contains the complete intent. If a process dies before
709:         # the token index is linked, callers which know observer/token can still
710:         # recover coherently; no directory scan or candidate deletion occurs.
711:         if not _publish_no_clobber(
712:             pending_dir / f"{self._slot(self.namespace, observer)}.json", _json_bytes(intent)
713:         ):
714:             return self._intent_for(
715:                 observer,
716:                 token=token,
717:                 provenance=provenance,
718:                 key_ref=key_ref,
719:                 expected_public_key=expected_public_key,
720:             )
721:         self._ensure_token_index(intent)
722:         return intent

724:     def _pending_for_recovery(self, observer: str, token: str) -> dict[str, object]:
725:         pending_dir = self._subdir("pending-v1", create=False)
726:         if pending_dir is None:
727:             raise BindingRecoveryRequired("no durable mapped binding intent exists")
728:         pending = self._read_json(
729:             pending_dir / f"{self._slot(self.namespace, observer)}.json"
730:         )
731:         if pending is None:
732:             raise BindingRecoveryRequired("no durable mapped binding intent exists")
733:         if (
734:             pending.get("schema") != _INTENT_SCHEMA
735:             or pending.get("namespace") != self.namespace
736:             or pending.get("observer") != observer
737:             or pending.get("token") != token
738:             or pending.get("stage") != "pending"
739:             or pending.get("provenance") not in {"created", "legacy-import", "existing-ref"}
740:             or not isinstance(pending.get("key_ref"), str)
741:         ):
742:             raise BindingRecoveryRequired("mapped binding recovery token does not match a pending intent")
743:         if self._pending_by_token(token) != pending:
744:             raise BindingConflict("mapped binding token owns a contradictory pending slot")
745:         self._key_dir(str(pending["key_ref"]))
746:         return pending

748:     def _hook(self, phase: str, intent: dict[str, object]) -> None:
749:         if self._creation_hook is not None:
750:             self._creation_hook(phase, dict(intent))

752:     def _ensure_created_key(self, intent: dict[str, object]):
753:         key_ref = str(intent["key_ref"])
754:         keys = self._subdir("keys-v1", create=True)
755:         assert keys is not None
756:         key_dir = keys / key_ref
757:         _directory(key_dir, create=True)
758:         mode = _lstat_kind(key_dir)
759:         created = False
760:         if mode is None:  # defensive: _directory above must establish it
761:             raise FileNotFoundError(key_dir)
762:         if not S_ISDIR(mode):
763:             raise ValueError("mapped custody candidate key path is not a directory")
764:         private = key_dir / "ed25519.key"
765:         public = key_dir / "ed25519.pub"
766:         if _lstat_kind(private) is None:
767:             # This intent exclusively reserved the candidate ref. It is safe
768:             # to resume a crash before private-key publication completed.
769:             if _lstat_kind(public) is not None:
770:                 raise ValueError("incomplete candidate has public evidence without private key")
771:             pair = ed25519.load_or_generate(key_dir)
772:             _fsync_directory(key_dir)
773:             created = True
774:         else:
775:             pair = self._load_keypair(key_ref)
776:             if _lstat_kind(public) is None:
777:                 # A crash between durable private and public publication is
778:                 # resumable only for this retained, provider-owned intent.
779:                 _publish_no_clobber(public, (pair.public_b64 + "\n").encode(), mode=0o644)
780:                 _fsync_directory(key_dir)
781:             pair = self._load_keypair(key_ref, require_public_file=True)
782:         # A prior attempt may have published both key files and then failed to
783:         # fsync this directory. Validate the pair first, then make the retry
784:         # retain those existing entries before returning success.
785:         _fsync_directory(key_dir)
786:         self._hook("key-published", intent)
787:         return pair, created

789:     def _copy_key(self, source: Path, intent: dict[str, object]):
790:         source_pair = self._load_external_key(source)
791:         keys = self._subdir("keys-v1", create=True)
792:         assert keys is not None
793:         key_dir = keys / str(intent["key_ref"])
794:         _directory(key_dir, create=True)
795:         if _lstat_kind(key_dir / "ed25519.key") is None:
796:             _publish_no_clobber(
797:                 key_dir / "ed25519.key", (source / "ed25519.key").read_bytes()
798:             )
799:             _publish_no_clobber(
800:                 key_dir / "ed25519.pub", (source_pair.public_b64 + "\n").encode(), mode=0o644
801:             )
802:             _fsync_directory(key_dir)
803:             created = True
804:         else:
805:             source_pair = self._load_keypair(
806:                 str(intent["key_ref"]), require_public_file=True
807:             )
808:             created = False
809:         candidate = self._load_keypair(str(intent["key_ref"]), require_public_file=True)
810:         if candidate.public_b64 != source_pair.public_b64:
811:             raise BindingConflict("copied candidate disagrees with the validated source key")
812:         _fsync_directory(key_dir)
813:         self._hook("key-published", intent)
814:         return candidate, created

816:     def _load_external_key(self, key_dir: Path):
817:         mode = _lstat_kind(key_dir)
818:         if mode is None or not S_ISDIR(mode):
819:             raise FileNotFoundError(key_dir)
820:         private = key_dir / "ed25519.key"
821:         _regular(private)
822:         pair = ed25519.load(key_dir)
823:         public = key_dir / "ed25519.pub"
824:         if _regular(public, required=False) and public.read_text(encoding="utf-8") != pair.public_b64 + "\n":
825:             raise ValueError("source custody public evidence disagrees with private key")
826:         return pair

828:     def _publish_binding(self, intent: dict[str, object], pair) -> tuple[dict[str, object], bool]:
829:         expected = intent.get("expected_public_key")
830:         if expected is not None and pair.public_b64 != expected:
831:             raise BindingConflict("candidate key does not match expected public key")
832:         record: dict[str, object] = {
833:             "schema": _SCHEMA,
834:             "namespace": self.namespace,
835:             "observer": intent["observer"],
836:             "key_ref": intent["key_ref"],
837:             "algorithm": "ed25519",
838:             "public_key": pair.public_b64,
839:             "provenance": intent["provenance"],
840:         }
841:         bindings = self._subdir("bindings-v1", create=True)
842:         assert bindings is not None
843:         published = _publish_no_clobber(
844:             bindings / f"{self._slot(self.namespace, str(intent['observer']))}.json", _json_bytes(record)
845:         )
846:         if not published:
847:             existing = self._binding_record(self.namespace, str(intent["observer"]))
848:             assert existing is not None
849:             if existing["public_key"] != pair.public_b64:
850:                 raise BindingConflict("another binding won with a different public key")
851:             _fsync_directory(bindings)
852:             return existing, False
853:         self._hook("binding-published", intent)
854:         # Intent files are retained as durable recovery evidence. The marker is
855:         # deliberately not removed automatically, avoiding unsafe cleanup races.
856:         self._publish_completion(intent)
857:         return record, True

859:     def _mutate(
860:         self,
861:         observer: str,
862:         *,
863:         token: str,
864:         provenance: str,
865:         key_ref: str | None = None,
866:         expected_public_key: str | None = None,
867:         source_key_dir: Path | None = None,
868:         _locked: bool = False,
869:     ) -> BindingCreationResult:
870:         with (nullcontext() if _locked else self._mutation_lock()):
871:             intent = self._intent_for(
872:                 observer,
873:                 token=token,
874:                 provenance=provenance,
875:                 key_ref=key_ref,
876:                 expected_public_key=expected_public_key,
877:             )
878:             if intent.get("completed"):
879:                 return BindingCreationResult(
880:                     namespace=self.namespace,
881:                     observer=observer,
882:                     key_ref=str(intent["key_ref"]),
883:                     public_key=str(intent["public_key"]),
884:                     provenance=str(intent["provenance"]),
885:                     token=token,
886:                     binding_created=False,
887:                     key_created=False,
888:                 )
889:             phase = "intent-published"
890:             try:
891:                 self._hook(phase, intent)
892:                 if source_key_dir is None:
893:                     phase = "key-published"
894:                     pair, key_created = self._ensure_created_key(intent)
895:                 else:
896:                     phase = "key-published"
897:                     pair, key_created = self._copy_key(source_key_dir, intent)
898:                 phase = "binding-published"
899:                 record, binding_created = self._publish_binding(intent, pair)
900:             except BaseException as exc:
901:                 if not isinstance(exc, Exception):
902:                     raise
903:                 raise BindingMutationIncomplete(
904:                     namespace=self.namespace,
905:                     observer=observer,
906:                     token=token,
907:                     key_ref=str(intent["key_ref"]),
908:                     phase=phase,
909:                 ) from exc
910:             return BindingCreationResult(
911:                 namespace=self.namespace,
912:                 observer=observer,
913:                 key_ref=str(record["key_ref"]),
914:                 public_key=str(record["public_key"]),
915:                 provenance=str(record["provenance"]),
916:                 token=token,
917:                 binding_created=binding_created,
918:                 key_created=key_created,
919:             )

925:     def recover_binding(self, observer: str, *, token: str) -> BindingCreationResult:
926:         """Resume the exact token's durable binding intent without a new key."""
927:         _exact(observer, "observer")
928:         _exact(token, "token")
929:         with self._mutation_lock():
930:             intent = self._pending_for_recovery(observer, token)
931:             phase = "intent-published"
932:             try:
933:                 # A published slot is authority over its candidate. Validate it
934:                 # against the retained pending marker so a missing/moved key can
935:                 # never be regenerated under an already-named reference.
936:                 published = self._binding_record(self.namespace, observer)
937:                 if published is not None:
938:                     self._validated_binding(self.namespace, observer)
939:                     self._assert_published_intent(
940:                         published, intent, require_token_index=False
941:                     )
942:                     pending_dir = self._subdir("pending-v1", create=False)
943:                     assert pending_dir is not None
944:                     bindings = self._subdir("bindings-v1", create=False)
945:                     assert bindings is not None
946:                     _fsync_directory(pending_dir)
947:                     _fsync_directory(bindings)
948:                     _fsync_directory(self._key_dir(str(published["key_ref"])))
949:                     # Restore the token index only after the material-valid
950:                     # slot proves it is the retained intent's exact binding.
951:                     self._ensure_token_index(intent)
952:                     self._assert_published_intent(published, intent)
953:                     phase = "binding-published"
954:                     self._publish_completion(intent)
955:                     return BindingCreationResult(
956:                         namespace=self.namespace,
957:                         observer=observer,
958:                         key_ref=str(published["key_ref"]),
959:                         public_key=str(published["public_key"]),
960:                         provenance=str(published["provenance"]),
961:                         token=str(intent["token"]),
962:                         binding_created=False,
963:                         key_created=False,
964:                     )
965:                 pending_dir = self._subdir("pending-v1", create=False)
966:                 assert pending_dir is not None
967:                 _fsync_directory(pending_dir)
968:                 self._ensure_token_index(intent)
969:                 provenance = str(intent["provenance"])
970:                 if provenance == "created":
971:                     phase = "key-published"
972:                     pair, key_created = self._ensure_created_key(intent)
973:                 else:
974:                     # Import and existing-reference intents have already selected a
975:                     # durable candidate. Recovery never reopens the legacy source
976:                     # or creates a replacement key under the reserved reference.
977:                     key_ref = str(intent["key_ref"])
978:                     try:
979:                         pair = self._load_keypair(key_ref)
980:                     except FileNotFoundError as exc:
981:                         if provenance == "legacy-import":
982:                             raise BindingRecoveryRequired(
983:                                 "legacy import candidate is absent; repeat the explicit import with this token"
984:                             ) from exc
985:                         raise
986:                     expected = intent.get("expected_public_key")
987:                     if expected is not None and pair.public_b64 != expected:
988:                         raise BindingConflict("recovery candidate disagrees with its intent")
989:                     key_dir = self._key_dir(key_ref)
990:                     public = key_dir / "ed25519.pub"
991:                     if _lstat_kind(public) is None:
992:                         if provenance != "legacy-import":
993:                             raise BindingRecoveryRequired(
994:                                 "existing-reference candidate lacks durable public evidence"
995:                             )
996:                         _publish_no_clobber(
997:                             public, (pair.public_b64 + "\n").encode(), mode=0o644
998:                         )
999:                         _fsync_directory(key_dir)
1000:                     pair = self._load_keypair(key_ref, require_public_file=True)
1001:                     key_created = False
1002:                 phase = "binding-published"
1003:                 record, binding_created = self._publish_binding(intent, pair)
1004:                 return BindingCreationResult(
1005:                     namespace=self.namespace,
1006:                     observer=observer,
1007:                     key_ref=str(record["key_ref"]),
1008:                     public_key=str(record["public_key"]),
1009:                     provenance=str(record["provenance"]),
1010:                     token=token,
1011:                     binding_created=binding_created,
1012:                     key_created=key_created,
1013:                 )
1014:             except BaseException as exc:
1015:                 if not isinstance(exc, Exception):
1016:                     raise
1017:                 if isinstance(
1018:                     exc, (BindingConflict, BindingRecoveryRequired, FileNotFoundError, ValueError)
1019:                 ):
1020:                     raise
1021:                 raise BindingMutationIncomplete(
1022:                     namespace=self.namespace,
1023:                     observer=observer,
1024:                     token=token,
1025:                     key_ref=str(intent["key_ref"]),
1026:                     phase=phase,

===== libs/custody/README.md =====
1: # custody
2: 
3: Loops signing composition — the store's at-rest signing format, below every writer.
4: 
5: Owns the `loops-tick-v1` / `loops-fact-v1` domain-separation constants, the
6: custody-co-located `keys/` layout (per-observer nesting + flat self-observer
7: back-compat), the signer/verifier builders, and `ensure_signing_key`. Observer
8: names remain exact identities: a filesystem spelling collision refuses rather
9: than borrowing another observer's key. A renamed vertex may retain a nested
10: self key; all signer domains and `ensure_signing_key` select that one key.
11: Identical flat and nested self copies are one self key; conflicting key material
12: refuses until repaired. A differently spelled nested observer remains a
13: distinct observer and is never used as the self fallback. Observer components
14: cannot name the reserved `ed25519.key` or `ed25519.pub` custody files when
15: mapped to directories. This is a local custody-layout restriction, not an
16: observer label restriction in the event protocol. A self observer with such a
17: stem uses only the flat layout.
18: 
19: Signer loading never creates keys. Concurrent explicit key creation publishes
20: one complete private key; all callers load that key and derive its public file.
21: 
22: Any client that emits signed facts or ticks must compose identically or
23: `sl store verify` reports breaks — that's why this is a lib, not CLI code.
24: Depends on `sign` (loops-agnostic Ed25519 primitives) and `engine`
25: (store-canonical declaration resolution). The engine itself never imports
26: this — it takes signer callables by injection.
27: 
28: 
29: ## Explicit mapped bindings
30: 
31: `MappedCredentialProvider(root, namespace=..., receipt_observer=...)` provides
32: an opt-in alternative to locator-based custody. Exact namespace/observer
33: strings select an opaque managed key reference recorded with its public key.
34: All three signing domains use that binding with separate domain prefixes.
35: Hashed filesystem slots are storage details; labels retain their exact case,
36: Unicode spelling, and non-path characters. A binding does not authorize an
37: Arrival claim by itself: the engine separately checks captured history.
38: 
39: `for_write` returns lazy configuration. `resolve` loads and validates the
40: binding, private-derived public material, and any existing public convenience
41: file without creating or repairing anything. Missing unbound identities can
42: remain unsigned; malformed mappings, conflicting public material, missing
43: referenced private keys, and filesystem aliases refuse.
44: 
45: Explicit mutations are `create_binding(observer, token=...)`,
46: `import_legacy(vertex_path, observer, token=...)`, and
47: `bind_existing_ref(observer, key_ref, expected_public_key, token=...)`.
48: Import copies private material into managed custody and preserves the original
49: files. Reusing a managed reference is deliberate and does not introduce its
50: public key into an Arrival lineage. Existing bindings are not silently
51: replaced; local rotation is a separate future operation.
52: 
53: Mutations use a provider-wide POSIX advisory lock, durable intents, and
54: complete-file publication with no-clobber links. The supported filesystem must
55: provide those lock, link, and fsync semantics. Concurrent creators converge
56: on one binding; resolution remains read-only. There is no automatic orphan
57: cleanup or key deletion.
58: 
59: Keep the operation token when a mutation is interrupted.
60: `BindingMutationIncomplete` retains namespace, observer, token, candidate key
61: reference, and phase; the binding may already have been published.
62: `recover_binding(observer, token=...)` reconciles the retained intent and
63: candidate. It never replaces a missing key referenced by a published binding.
64: Explicit recovery may restore a missing token index from its retained intent
65: before a later candidate-key refusal; it does not claim zero filesystem effects
66: or create a key/binding in that case.
67: An import interrupted before copying private material needs the original
68: source re-presented with the same import token and expected public material;
69: recovery cannot invent that key. A completed imported candidate can recover
70: without the old source directory. Retrying with a different token does not
71: bypass an unfinished operation. Creation results report the returned binding
72: and effects of that invocation; they contain no private key material.
