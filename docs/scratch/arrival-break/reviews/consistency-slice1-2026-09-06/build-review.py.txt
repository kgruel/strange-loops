def excerpt(file, symbols=None, ranges=None):
    p = root / file
    data = p.read_text()
    hashes[file] = hashlib.sha256(p.read_bytes()).hexdigest()
    lines = data.splitlines()
    if symbols is not None:
        found = {}
        def walk(nodes, prefix=''):
            for n in nodes:
                if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                    found[prefix+n.name] = (min([n.lineno]+[d.lineno for d in n.decorator_list]), n.end_lineno)
                    if isinstance(n, ast.ClassDef):
                        walk(n.body, prefix+n.name+'.')
        walk(ast.parse(data).body)
        ranges = [found[n] for n in symbols]
    ranges = ranges or [(1, len(lines))]
    selected = sorted({i for lo,hi in ranges for i in range(lo,min(hi,len(lines))+1)})
    return '\n### '+file+' (original line numbers)\n```\n'+'\n'.join(f'{i}: {lines[i-1]}' for i in selected)+'\n```\n'
