from pathlib import Path
import ast
import datetime
import difflib
import hashlib
import json
import subprocess

root = Path('/Users/kaygee/Code/loops-wt/arrival-finish')
base = Path('/tmp/loops-arrival-c1-2026-09-06')
out = root / 'docs/scratch/arrival-break/reviews/consistency-c1-2026-09-06'
out.mkdir(parents=True, exist_ok=True)
baseline = json.loads((base / 'baseline.json').read_text())
hashes = {}

def excerpt(file, symbols=None):
    p = root / file
    data = p.read_text()
    hashes[file] = hashlib.sha256(p.read_bytes()).hexdigest()
    lines = data.splitlines()
    ranges = [(1, len(lines))]
    if symbols:
        found = {}
        def walk(nodes, prefix=''):
            for n in nodes:
                if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                    found[prefix+n.name] = (min([n.lineno]+[d.lineno for d in n.decorator_list]), n.end_lineno)
                    if isinstance(n, ast.ClassDef):
                        walk(n.body, prefix+n.name+'.')
        walk(ast.parse(data).body)
        ranges = [found[s] for s in symbols]
    ids = sorted({i for a,b in ranges for i in range(a,b+1)})
    return '\n### '+file+'\n```\n'+'\n'.join(f'{i}: {lines[i-1]}' for i in ids)+'\n```\n'

Do not claim test execution or unseen source access. Review C1 only: optional registry injection for SDK resolve_entity/read_timeline/inspect_declaration, across all currently supported routes. Check production diff and meaningful tests for real registered adapter dispatch (not only mocked engine entrypoints), exact opaque locations, unknown backend refusal without built-in/legacy fallback, and single root capture. for name in ['consistency-c1-2026-09-06.md', 'consistency-c1-validation-2026-09-06.md']:
    parts.append(excerpt('docs/scratch/arrival-break/'+name))
paths = subprocess.check_output(['git','ls-files','--cached','--others','--exclude-standard','-z'], cwd=root).decode().split('\0')
changed = []
patches = []
for file in sorted(set(filter(None, paths))):
    if not file.startswith(('libs/sdk/', 'libs/engine/')):
        continue
    p = root/file
    if not p.is_file() or hashlib.sha256(p.read_bytes()).hexdigest() == baseline.get(file):
        continue
    changed.append(file)
    hashes[file] = hashlib.sha256(p.read_bytes()).hexdigest()
    old = base/'before'/file
    patches.append(''.join(difflib.unified_diff(old.read_text().splitlines(True) if old.exists() else [], p.read_text().splitlines(True), fromfile='before-c1/'+file, tofile='after-c1/'+file)))
parts.append('\n## C1-only diff\n```diff\n'+''.join(patches)+'\n```\n')
parts.append(excerpt('libs/sdk/src/sdk/read.py', ['_open_arrival_read', '_arrival_declaration', '_aggregate_aware_arrival_descriptor', '_aggregate_timeline', '_aggregate_timeline_from', 'resolve_entity', 'read_timeline']))

parts.append(excerpt('libs/sdk/src/sdk/aggregate.py'))
parts.append(excerpt('libs/sdk/src/sdk/target.py', ['_arrival_descriptor', '_refuse_arrival_aggregate_members']))
parts.append(excerpt('libs/engine/src/engine/arrival_aggregate.py', ['capture_aggregate']))
parts.append(excerpt('libs/engine/src/engine/arrival_registry.py', ['descriptor_for', 'BackendRegistry']))
parts.append(excerpt('libs/engine/src/engine/arrival_consumer.py', ['open_read']))

    if '/tests/' in file:
        parts.append(excerpt(file))
