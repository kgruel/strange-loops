```python
1: """Ed25519 detached signatures over digests, with domain separation.
2: 
3: The provenance-arc signing primitive (loops delta 2: tick signing). A distinct
4: surface from sign.keys/sign.jwt — those mint RSA JWTs over claims documents;
5: this signs already-computed digests with deterministic Ed25519 (RFC 8032).
6: Determinism is the load-bearing property for tamper-evidence: the same
7: commitment always yields the same signature, so re-signing never produces
8: spurious diffs and signature equality is meaningful.
9: 
10: Domain separation is mandatory: every signature binds a caller-supplied domain
11: prefix (e.g. ``myapp-tick-v1``) so a signature minted for one surface can
12: never be replayed as a commitment on another. The prefix constant belongs to
13: the COMPOSING layer (apps/loops), not here — this library stays loops-agnostic.

125: def _publish_public(key_dir: Path, keypair: Keypair) -> None:
126:     """Atomically make the convenience public-key file agree with the private key."""
127:     public_path = key_dir / _PUB_FILE
128:     expected = public_key_b64(keypair.public) + "\n"
129:     try:
130:         if public_path.read_text() == expected:
131:             return
132:     except FileNotFoundError:
133:         pass
134:     fd, raw = tempfile.mkstemp(prefix=f".{_PUB_FILE}.", dir=key_dir)
135:     temporary = Path(raw)
136:     try:
137:         with os.fdopen(fd, "w") as stream:
138:             stream.write(expected)
139:             stream.flush()
140:             os.fsync(stream.fileno())
141:         temporary.replace(public_path)
142:     finally:
143:         temporary.unlink(missing_ok=True)
144: 
145: 
146: def load(key_dir: str | Path) -> Keypair:
147:     """Load an existing Ed25519 keypair without creating any filesystem state.
148: 
149:     ``FileNotFoundError`` is deliberate: read-side callers can distinguish the
150:     pre-signature era from a request to mint a new key.
151:     """
152:     key_path = Path(key_dir) / _KEY_FILE
153:     private = serialization.load_pem_private_key(key_path.read_bytes(), password=None)
154:     if not isinstance(private, Ed25519PrivateKey):
155:         kind = type(private).__name__
156:         raise ValueError(f"expected Ed25519 private key in {key_path}, got {kind}")
157:     return Keypair(private=private, public=private.public_key())
158: 
159: 
160: def load_or_generate(key_dir: str | Path) -> Keypair:
161:     """Ensure an Ed25519 keypair exists in ``key_dir``, then load and return it."""
162:     key_dir = Path(key_dir)
163:     try:
164:         keypair = load(key_dir)
165:     except FileNotFoundError:
166:         _generate(key_dir)
167:         keypair = load(key_dir)
168:     _publish_public(key_dir, keypair)
169:     return keypair
170: 
171: 
172: def sign(keypair: Keypair, digest: bytes, *, domain: str) -> str:
173:     """Sign a digest under a domain prefix; returns the signature as base64.
174: 
175:     Deterministic (RFC 8032): the same (key, domain, digest) always yields
176:     the same signature.
177:     """
178:     return base64.b64encode(keypair.private.sign(_message(domain, digest))).decode()
179: 
180: 
181: def verify(
182:     public: Ed25519PublicKey, signature_b64: str, digest: bytes, *, domain: str
183: ) -> bool:
184:     """Check a base64 signature over a digest under a domain prefix.
185: 
186:     Returns False for any failure mode short of caller error — bad base64,
187:     wrong key, tampered digest, wrong domain. An empty domain still raises
188:     ValueError (that is a composition bug, not a verification outcome).
189:     """
190:     message = _message(domain, digest)
191:     try:
192:         raw = base64.b64decode(signature_b64, validate=True)
193:         public.verify(raw, message)
194:     except (InvalidSignature, binascii.Error, ValueError):
195:         return False
196:     return True
```

```python
774:         the same name into the same physical ledger. Originless evidence
775:         (``None`` or ``""``) is never local proof, including for an unnamed
776:         runtime candidate.
777:         """
778:         return bool(self._name) and getattr(tick, "origin", None) == self._name
784:     def hydrate_snapshot(self, facts: Sequence[Any], ticks: Sequence[Any]) -> int:
785:         """Rebuild this fresh, storeless vertex from already-captured facts.
786: 
787:         The caller supplies one receipt-ordered snapshot.  Hydration is replay:
788:         it never calls ordinary ingress, persists nothing, and never evaluates
789:         or fires a boundary.  Runtime writing uses it to make a candidate from
790:         its CURRENT projection rather than trusting a live vertex's mutable
791:         fold state.
792:         """
793:         if self._store is not None:
794:             raise RuntimeError("hydration requires a storeless vertex")
795:         if self._has_children:
796:             raise RuntimeError("stage 3A hydration does not support child vertices")
797:         count = 0
798:         ticked_loops: set[str] = set()
799:         last_vertex_tick: datetime | None = None
800:         entries = [
801:             (fact.arrival_ordinal, fact.arrival_seq, "fact", fact) for fact in facts
802:         ] + [
803:             (tick.arrival_ordinal, tick.arrival_seq, "tick", tick) for tick in ticks
804:         ]
805:         with self._replay_guard():
806:             for _, _, entry_kind, entry in sorted(entries, key=lambda row: row[:2]):
807:                 if entry_kind == "tick":
808:                     tick = entry
809:                     if not self._owns_tick(tick):
810:                         continue
811:                     loop = self._loops.get(tick.name)
812:                     if loop is not None:
813:                         loop.replay_boundary()
814:                         ticked_loops.add(tick.name)
815:                     if tick.name == self._name:
816:                         # Match ``replay()``: a vertex tick snapshots all
817:                         # loops but does not reset them; its timestamp is the
818:                         # period context restored after the bounded replay.
819:                         last_vertex_tick = datetime.fromtimestamp(tick.ts, tz=timezone.utc)
820:                     continue
821:                 fact = entry
822:                 routed_kind = self._routed_kind(fact.kind) if self._has_routes else fact.kind
823:                 payload = fact.payload
824:                 if self._parse_pipelines:
825:                     payload = self._apply_parse_pipeline(routed_kind, payload)
826:                     if payload is None:
827:                         continue
828:                 loop = self._loops.get(routed_kind)
829:                 if loop is not None:
830:                     loop.receive(
831:                         {**payload, "_ts": fact.ts},
832:                         ts=datetime.fromtimestamp(fact.ts, tz=timezone.utc),
833:                     )
834:                 count += 1
835:         # A historical loop tick is its own exact count/reset checkpoint. A
836:         # loop with no persisted tick retains legacy replay's modulo recovery.
837:         for name, loop in self._loops.items():
838:             if loop.boundary_count is not None and name not in ticked_loops:
839:                 replayed = loop._projection.events_folded
840:                 if loop.boundary_mode == "after":
841:                     loop._boundary_exhausted = replayed >= loop.boundary_count
842:                     loop._count_since_boundary = 0 if loop._boundary_exhausted else replayed
843:                 elif loop.boundary_mode == "every":
844:                     loop._count_since_boundary = replayed % loop.boundary_count
845:                 else:
846:                     loop._count_since_boundary = replayed
847:         if last_vertex_tick is not None:
848:             self._vertex_period_start = last_vertex_tick
849:         return count
1260:     def plan_pending_boundaries(
1261:         self,
1262:         facts: tuple[Any, ...],
1263:         ticks: tuple[Any, ...],
1264:         *,
1265:         evaluated_at: float,
1266:     ) -> tuple[Tick, ...]:
1267:         """Evaluate externally pending boundaries from captured evidence only.
1268: 
1269:         This is the storeless counterpart of :meth:`evaluate_boundaries` for
1270:         Arrival source planning. Each boundary's recorded ticks establish its
1271:         event-time period edge. At equal event time, facts received before the
1272:         closing tick belong to that closed period; later arrivals are eligible
1273:         for a new period. A tick's fact cursor is its evidence horizon, not a
1274:         trigger cursor: future-dated facts can already occur before that cursor.
1275:         The method mutates only this detached candidate and never persists.
1276:         Count boundaries remain ingress-driven and are not invented from replay.
1277:         """
1278:         if self._store is not None:
1279:             raise RuntimeError("pending-boundary planning requires a detached vertex")
1280: 
1281:         def tick_time(ts: float) -> float:
1282:             # Runtime ticks are datetimes (microsecond precision). Compare on
1283:             # that same axis or rounding down repeats a consumed fact, while
1284:             # rounding up hides a later receipt carrying the exact same ts.
1285:             # The original fact timestamps and receipt coordinates stay intact.
1286:             return datetime.fromtimestamp(ts, tz=UTC).timestamp()
1287: 
1288:         edges: dict[str, tuple[float, int, int]] = {}
1289:         for tick in ticks:
1290:             if not self._owns_tick(tick):
1291:                 continue
1292:             edge = (tick_time(tick.ts), tick.arrival_ordinal, tick.arrival_seq)
1293:             if tick.name not in edges or edge > edges[tick.name]:
1294:                 edges[tick.name] = edge
1295: 
1296:         def unconsumed(fact: Any, name: str) -> bool:
1297:             edge = edges.get(name)
1298:             return edge is None or (
1299:                 tick_time(fact.ts), fact.arrival_ordinal, fact.arrival_seq
1300:             ) > edge
1301: 
1302:         period_facts = sorted(
1303:             (fact for fact in facts if 0.0 <= fact.ts <= evaluated_at),
1304:             key=lambda fact: (
1305:                 fact.ts,
1306:                 getattr(fact, "arrival_ordinal", 0),
1307:                 getattr(fact, "arrival_seq", 0),
1308:             ),
1309:         )
1310:         if not period_facts:
1311:             return ()
1312: 
1313:         planned: list[Tick] = []
1314:         fired_boundaries: set[str] = set()
1315:         for fact in period_facts:
1316:             kind = fact.kind
1317:             payload = fact.payload
1318:             fact_ts = datetime.fromtimestamp(fact.ts, tz=timezone.utc)
1319:             if (
1320:                 self._has_vertex_boundary
1321:                 and self._vertex_period_start is None
1322:                 and unconsumed(fact, self._name)
1323:             ):
1324:                 self._vertex_period_start = fact_ts
1325: 
1326:             if kind in self._vertex_boundary_kinds:
1327:                 if not unconsumed(fact, self._name):
1328:                     continue
1329:                 spec = self._match_vertex_boundary(kind, payload)
1330:                 if spec is not None:
1331:                     planned.append(self._fire_vertex_boundary(fact_ts, payload, run=spec.run))
1332:                     break
1333:                 continue
1334: 
1335:             fold_kind = self._boundary_map.get(kind)
1336:             if fold_kind is None or fold_kind in fired_boundaries:
1337:                 continue
1338:             if not unconsumed(fact, fold_kind):
1339:                 continue
1340:             match = self._boundary_match.get(kind, ())
1341:             if match and not all(payload.get(key) == value for key, value in match):
1342:                 continue
1343:             conditions = self._boundary_conditions.get(kind, ())
1344:             target_loop = self._loops[fold_kind]
1345:             if conditions and not _eval_conditions(target_loop.state, conditions):
1346:                 continue
1347:             planned.append(
1348:                 target_loop.fire(
1349:                     fact_ts,
1350:                     origin=self._name,
1351:                     boundary_payload=payload,
1352:                 )
1353:             )
1354:             fired_boundaries.add(fold_kind)
1355:         return tuple(planned)
```
