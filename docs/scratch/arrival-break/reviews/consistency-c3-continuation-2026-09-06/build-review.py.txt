hashes = {}

def git(*args):
    return subprocess.check_output(['git', *args], cwd=root, text=True)

def span(node):
    return min([node.lineno] + [d.lineno for d in getattr(node, 'decorator_list', [])]), node.end_lineno

def add(file, symbols=None):
    p = root / file
    data = p.read_text()
    hashes[file] = hashlib.sha256(p.read_bytes()).hexdigest()
    lines = data.splitlines()
    ranges = [(1, len(lines))]
    if symbols:
        tree = ast.parse(data)
        ranges = []
        for symbol in symbols:
            level = tree.body
            for name in symbol.split('.'):
                node = next(n for n in level if getattr(n, 'name', None) == name)
                level = node.body
            ranges.append(span(node))
    indices = sorted({i for a, b in ranges for i in range(a, b + 1)})
    parts.append('\n## ' + file + '\n```\n' + '\n'.join(
        f'{i}: {lines[i - 1]}' if lines[i - 1] else f'{i}:' for i in indices
    ) + '\n```\n')
