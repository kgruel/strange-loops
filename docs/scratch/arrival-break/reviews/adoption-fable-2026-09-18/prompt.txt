1: # Adopt a migrated Arrival declaration
2: 
3: `sdk.adopt_arrival` appends the declaration anchor that makes a migrated
4: Arrival history usable by ordinary SDK reads and writes. It preserves the
5: migration prefix and uses the existing physical lineage. Fresh stores use
6: `init_vertex`; stores with an adopted declaration use `edit_declaration`.
7: 
8: The caller supplies two distinct snapshots:
9: 
10: - `reviewed_text` and its `reviewed_sha256`: exact declaration bytes selected
11:   for adoption, such as the copied legacy vertex used for migration.
12: - `declaration_text`: the current published vertex, naming an explicit backend,
13:   physical lineage, and authority role.
14: 
15: Their declaration documents must agree apart from residence. This first
16: operation cannot introduce or replace observer keys, or adopt a changed
17: declaration as an incidental part of migration. It checks the selected head,
18: registry-forming signatures, declaration identities and captured observer keys
19: before signing the new anchor.
20: 
21: ```python
22: from sdk import adopt_arrival
23: 
24: # Keep these snapshots and the report verification receipt from the offline
25: # migration workflow. Verify its report at selected_head BEFORE adoption.
26: result = adopt_arrival(
27:     published_vertex,
28:     selected_head=migration.head,
29:     reviewed_text=reviewed_text,
30:     reviewed_sha256=reviewed_sha256,
31:     declaration_text=published_vertex.read_text(encoding="utf-8"),
32:     observer="alice",
33:     credentials=mapped_provider,
34: )
35: receipt = result.as_dict()
36: ```
37: 
38: The mapped binding must already exist and match the observer's captured key
39: history. Adoption resolves distinct FACT and ARRIVAL authorship requests;
40: neither lookup creates keys nor falls back to legacy credentials. The result
41: retains the adoption commit, projection outcome and public binding evidence.
42: Migration report verification remains the offline caller's responsibility;
43: the report's live-head comparison no longer applies after adoption advances
44: the history.
45: 
46: Before appending, the operation durably reserves the exact signed draft in
47: `<vertex>.arrival-adopt.intent`. If interrupted, retain the intent and the
48: typed error's commit/head evidence, then reconcile explicitly:
49: 
50: ```python
51: from sdk import recover_arrival_adoption
52: 
53: recovered = recover_arrival_adoption(intent_path)
54: ```
55: 
56: Recovery requires no signing credentials. It either appends the reserved draft
57: at its original predecessor or recognizes that exact draft at its expected
58: ordinal, synchronizes the projection and checks the published cache. It does
59: not re-sign or append a duplicate. If later records exist, `head` still names
60: the adoption head; `observed_head` records the later tip. A reconstructed commit
61: labels its durability evidence as recovery evidence rather than claiming the
62: original append receipt was retained.
63: 
64: The operation reconciles the already-published cache without rewriting it.
65: A changed cache or a post-append projection failure leaves a recovery boundary;
66: a durable append is not rolled back. Process-exit tests cover reservation and
67: append boundaries, not arbitrary power loss. Live writer coordination and a
68: representative user-store rehearsal remain separate from this SDK operation.

664: def recover_arrival_adoption(

158:     path.parent.mkdir(parents=True, exist_ok=True)
159:     fd, raw = tempfile.mkstemp(prefix=f".{path.name}.", dir=path.parent)
160:     tmp = Path(raw)
161:     try:
162:         with os.fdopen(fd, "w", encoding="utf-8") as stream:
163:             stream.write(text)

337: def _declared_key_drafts(
338:     documents: Sequence[Mapping[str, Any]],
339:     *,
340:     observer: str,
341:     public_key: str,
342:     at: float,
343:     signer: Callable[[str, str], str | None],
344:     verify: Callable[[str, str, str], bool] | None,
345: ) -> tuple[RecordDraft, ...]:
346:     """Introduce declared observer keys using the valid founding key."""
347:     keys: dict[str, str] = {}
348:     for document in documents:
349:         if document["kind"] != "_decl.observer-defined":
350:             continue
351:         key = document["payload"].get("key")
352:         if key is None:
353:             continue
354:         subject = document["subject"]
355:         try:
356:             if not isinstance(key, str) or len(base64.b64decode(key, validate=True)) != 32:
357:                 raise ValueError
358:         except (ValueError, TypeError):
359:             raise InitializationConflict(
360:                 f"observer {subject!r} has a malformed public key"
361:             ) from None
362:         if subject == observer:
363:             if key != public_key:
364:                 raise InitializationConflict(
365:                     "declared founding key differs from genesis public key"
366:                 )
367:             continue
368:         if subject in keys and keys[subject] != key:
369:             raise InitializationConflict(f"observer {subject!r} has conflicting declared keys")
370:         keys[subject] = key
371:     drafts: list[RecordDraft] = []
372:     for subject, key in keys.items():
373:         body = {"observer": subject, "key": key}
374:         digest = content_commitment("key", at, observer, "", body)
375:         signature = signer(observer, digest)
376:         if not signature or (
377:             verify is not None and not verify(public_key, signature, digest)
378:         ):
379:             raise InitializationConflict("founding signer cannot authenticate key introduction")
380:         drafts.append(RecordDraft("key", at, observer, "", body, signature))
381:     return tuple(drafts)

406: def _check_preconditions(
407:     target: Path,
408:     descriptor: StoreDescriptor,
409:     documents: Sequence[Mapping[str, Any]],
410:     declaration_text: str,
411:     observer: str,
412:     public_key: str,
413:     signer: Callable[[str, str], str | None],
414:     fact_signer: Callable[[str, str], str | None],
415:     arrival_signer: Callable[[str, str], str | None],
416: ) -> None:
417:     if descriptor.role is not Profile.AUTHORITY:
418:         raise InitializationConflict("fresh initialization requires an Authority descriptor")
419:     if not observer or not isinstance(observer, str):
420:         raise InitializationConflict("fresh initialization requires a non-empty observer")
421:     if not isinstance(public_key, str) or not public_key:
422:         raise InitializationConflict("fresh initialization requires a founding public key")
423:     try:
424:         if len(base64.b64decode(public_key, validate=True)) != 32:
425:             raise ValueError
426:     except (ValueError, TypeError):
427:         raise InitializationConflict(
428:             "founding public key must be raw-32-byte base64"
429:         ) from None
430:     if not isinstance(declaration_text, str) or not declaration_text.strip():
431:         raise InitializationConflict("declaration text must be non-empty UTF-8 text")
432:     if not callable(signer) or not callable(fact_signer) or not callable(arrival_signer):
433:         raise InitializationConflict("fresh initialization requires signer callables")
434:     if not isinstance(documents, Sequence) or isinstance(documents, (str, bytes)):
435:         raise InitializationConflict("declaration documents must be a sequence")
436:     try:
437:         json.dumps([dict(document) for document in documents], ensure_ascii=False)
438:     except (TypeError, ValueError) as exc:
439:         raise InitializationConflict(f"declaration documents are not JSON-safe: {exc}") from exc
440:     if target.exists():
441:         raise InitializationConflict(f"vertex file already exists: {target}")

444: def _validate_declaration(
445:     target: Path,
446:     descriptor: StoreDescriptor,
447:     documents: Sequence[Mapping[str, Any]],
448:     declaration_text: str,
449:     lineage: str,
450: ) -> None:
451:     """Check the signed declaration inputs before reserving or minting.
452: 
453:     The declaration text is the human ingress form and the document sequence
454:     is what gets committed into ``_decl.genesis``.  They must describe the
455:     same residence and identity, otherwise recovery could faithfully preserve
456:     a plan that never represented the requested target.
457:     """
458:     try:
459:         from lang import parse_vertex, validate_vertex, vertex_to_documents
460: 
461:         ast = parse_vertex(declaration_text, path=target)
462:         validate_vertex(ast)
463:     except Exception as exc:
464:         raise InitializationConflict(
465:             f"declaration text is not valid Vertex grammar: {exc}"
466:         ) from exc
467:     from .declaration import validate_arrival_runtime_identity
468: 
469:     validate_arrival_runtime_identity(
470:         ast.name,
471:         ast.loops,
472:         refusal=InitializationConflict,
473:     )
474:     parsed_documents = [document.as_json() for document in vertex_to_documents(ast)]
475:     if [dict(document) for document in documents] != parsed_documents:
476:         raise InitializationConflict(
477:             "declaration text and declaration documents disagree"
478:         )
479:     backend = ast.store_backend
480:     if (
481:         backend is None
482:         or backend.name != descriptor.backend
483:         or backend.lineage != lineage
484:         or backend.role != Profile.AUTHORITY.value
485:     ):
486:         raise InitializationConflict(
487:             "declaration residence must match the requested backend, lineage and authority role"
488:         )
489:     location = ast.store_location
490:     if descriptor.backend == "file" and ast.store is not None:
491:         location = str(ast.store)
492:     if location is None:
493:         raise InitializationConflict("declaration residence must include a store location")
494:     if descriptor.backend == "file":
495:         parsed_location = (target.parent / location).resolve()
496:         requested_location = Path(descriptor.location).resolve()
497:         locations_match = parsed_location == requested_location
498:     else:
499:         # DSNs are adapter-owned and must survive the engine untouched.
500:         locations_match = location == descriptor.location
501:     if not locations_match:
502:         raise InitializationConflict(
503:             "declaration residence location does not match the requested descriptor"
504:         )

507: def initialize_arrival(
508:     registry: BackendRegistry,
509:     descriptor: StoreDescriptor,
510:     *,
511:     target: Path | str,
512:     documents: Sequence[Mapping[str, Any]],
513:     declaration_text: str,
514:     observer: str,
515:     public_key: str,
516:     signer: Callable[[str, str], str | None],
517:     fact_signer: Callable[[str, str], str | None],
518:     arrival_signer: Callable[[str, str], str | None],
519:     fact_verify: Callable[[str, str, str], bool] | None = None,
520:     arrival_verify: Callable[[str, str, str], bool] | None = None,
521:     lineage: str | None = None,
522:     authored_at: float | None = None,
523:     failure_hook: Callable[[str], None] | None = None,
524: ) -> ArrivalInitializationResult:
525:     """Mint and publish a fresh Arrival declaration with recoverable intent."""
526:     target_path = Path(target).resolve()
527:     intent = arrival_intent_path(target_path)
528:     _check_preconditions(
529:         target_path, descriptor, documents, declaration_text, observer, public_key,
530:         signer, fact_signer, arrival_signer,
531:     )
532:     if intent.exists():
533:         raise InitializationRecoveryRequired(
534:             f"unfinished initialization intent exists: {intent}; recover explicitly"
535:         )
536:     chosen_lineage = lineage or descriptor.lineage or new_lineage()
537:     if descriptor.lineage is not None and descriptor.lineage != chosen_lineage:
538:         raise InitializationConflict("descriptor lineage does not match reserved lineage")
539:     _validate_declaration(
540:         target_path, descriptor, documents, declaration_text, chosen_lineage
541:     )
542:     at = time.time() if authored_at is None else authored_at
543:     genesis_body = {
544:         "protocol": 1,
545:         "lineage": chosen_lineage,
546:         "key": public_key,
547:     }
548:     genesis_digest = content_commitment(
549:         "genesis", at, observer, "", genesis_body
550:     )
551:     genesis_signature = signer(observer, genesis_digest)
552:     if not genesis_signature:
553:         raise InitializationConflict("founding signer produced no genesis signature")
554:     if arrival_verify is not None and not arrival_verify(
555:         public_key, genesis_signature, genesis_digest
556:     ):
557:         raise InitializationConflict("founding genesis signer does not match the public key")
558:     draft = build_declaration_anchor_draft(
559:         lineage=chosen_lineage,
560:         authored_at=at,
561:         observer=observer,
562:         documents=documents,
563:         fact_signer=fact_signer,
564:         arrival_signer=arrival_signer,
565:         public_key=public_key,
566:         fact_verify=fact_verify,
567:         arrival_verify=arrival_verify,
568:     )
569:     key_drafts = _declared_key_drafts(
570:         documents, observer=observer, public_key=public_key, at=at,
571:         signer=arrival_signer, verify=arrival_verify,
572:     )
573:     if key_drafts:
574:         # The minimum backend contract supports one record. Additional keys
575:         # and declaration genesis form one larger atomic unit; refuse its
576:         # unsupported size before reserving an intent or minting a lineage.
577:         ledger, query = registry.open(descriptor)
578:         try:
579:             limit = ledger.capabilities().max_atomic_records
580:             if limit is not None and len(key_drafts) + 1 > limit:
581:                 raise AtomicLimitExceeded(
582:                     f"initialization needs {len(key_drafts) + 1} records; atomic limit is {limit}"
583:                 )
584:         finally:
585:             for resource in (query, ledger):
586:                 close = getattr(resource, "close", None)
587:                 if callable(close):
588:                     close()
589:     planned = {
590:         "schema": _INTENT_SCHEMA,
591:         "phase": "reserved",
592:         "target": str(target_path),
593:         "descriptor": _descriptor_dict(
594:             StoreDescriptor(
595:                 descriptor.backend, descriptor.location, chosen_lineage,
596:                 descriptor.role, descriptor.query, descriptor.witness,
597:             )
598:         ),
599:         "lineage": chosen_lineage,
600:         "observer": observer,
601:         "public_key": public_key,
602:         "at": at,
603:         "genesis": {
604:             "kind": "genesis",
605:             "at": at,
606:             "observer": observer,
607:             "origin": "",
608:             "body": genesis_body,
609:             "signature": genesis_signature,
610:         },
611:         "documents": [dict(document) for document in documents],
612:         "declaration_text": declaration_text,
613:         "draft": _draft_dict(draft),
614:         "key_drafts": [_draft_dict(key_draft) for key_draft in key_drafts],
615:     }
616:     _reserve(intent, planned)
617:     try:
618:         return _continue(
619:             registry, intent, planned, failure_hook=failure_hook,
620:         )
621:     except BaseException:
622:         # The sidecar is the recovery record.  It is removed only once the
623:         # declaration bytes have been atomically published.
624:         raise

627: def _continue(
628:     registry: BackendRegistry,
629:     intent: Path,
630:     plan: dict[str, Any],
631:     *,
632:     failure_hook: Callable[[str], None] | None,
633: ) -> ArrivalInitializationResult:
634:     descriptor = _descriptor_from(plan["descriptor"])
635:     target = Path(plan["target"])
636:     draft_data = plan["draft"]
637:     draft = RecordDraft(
638:         kind=draft_data["kind"], authored_at=draft_data["authored_at"],
639:         observer=draft_data["observer"], origin=draft_data["origin"],
640:         body=draft_data["body"], signature=draft_data["signature"],
641:     )
642:     drafts = tuple(RecordDraft(**data) for data in plan.get("key_drafts", ())) + (draft,)
643:     ledger = query = None
644:     head: Head | None = None
645:     commit = None
646:     try:
647:         if plan["phase"] == "reserved":
648:             ledger, query = registry.open(descriptor)
649:             # PreGenesis carries no absence claim.  The adapter's exclusive
650:             # mint is the authority for an absent, existing, or corrupt
651:             # location and refuses safely in the latter two cases.
652:             try:
653:                 head = ledger.mint({
654:                     "observer": plan["observer"],
655:                     "signer": _reserved_genesis_signer(plan),
656:                     "key": plan["public_key"], "lineage": plan["lineage"],
657:                     "origin": "", "at": plan["at"],
658:                 })
659:             except Exception as mint_error:
660:                 from .arrival import GenesisRefused
661:                 from .arrival_contract import ContractRefusal
662:                 from .arrival_head_seam import NotWitnessed
663: 
664:                 if isinstance(mint_error, NotWitnessed):
665:                     raise InitializationUnwitnessed(
666:                         "genesis committed but its witness is unknown; recover "
667:                         "the initialization intent before retrying",
668:                         phase="reserved", lineage=plan["lineage"],
669:                         fact_id=plan["lineage"], head=mint_error.head,
670:                         commit=mint_error.commit, intent_path=intent,
671:                         cause=mint_error,
672:                     ) from mint_error
673:                 # A prior mint may have committed while its witness was
674:                 # unknown.  The failed exclusive mint is the proof that this
675:                 # is not a fresh location; only an exact custody comparison
676:                 # can turn that case into recoverable progress.
677:                 try:
678:                     head = _head_and_check_genesis(registry, descriptor, plan)
679:                 except Exception:
680:                     if isinstance(mint_error, (ContractRefusal, GenesisRefused)):
681:                         raise mint_error from None
682:                     raise InitializationOutcomeUnknown(
683:                         "genesis mint outcome is unknown; recover the durable "
684:                         "initialization intent before retrying",
685:                         phase="reserved", lineage=plan["lineage"],
686:                         fact_id=plan["lineage"], captured_head=None,
687:                         intent_path=intent, cause=mint_error,
688:                     ) from mint_error
689:                 if head.ordinal != 0:
690:                     raise mint_error from None
691:             _phase(plan, "minted")
692:             _save_intent(intent, plan)
693:             if failure_hook:
694:                 failure_hook("after-mint")

182: def sync_projection(
183:     registry: BackendRegistry,
184:     descriptor: StoreDescriptor,
185:     *,
186:     through: Head | None = None,
187:     rebuild: bool = False,
188: ) -> ProjectionSyncResult:
189:     """Advance one projection after normal registry attestation.
190: 
191:     Catch-up is the only supported operation.  Existing inconsistent state is
192:     refused; this coordinator has no path that discards rows or identity
193:     markers.  A projection already at or beyond the target is validated and
194:     reported at its actual prefix without moving backward.
195:     """
196:     if descriptor.role not in (Profile.AUTHORITY, Profile.REPLICA):
197:         raise NotAuthority(
198:             "projection maintenance requires an explicit Authority or Replica role"
199:         )
200: 
201:     ledger, query = registry.open(descriptor)
202:     maintenance: ProjectionMaintenance | None = None
203:     try:
204:         if not isinstance(ledger, AttestedLedger):
205:             raise TypeError("BackendRegistry.open must return an AttestedLedger")
206:         captured = _captured_head(ledger)
207:         target = _verify_target(
208:             ledger, captured, captured if through is None else through
209:         )
210: 
211:         maintenance = registry._maintenance_for(descriptor)
212:         capabilities = maintenance.capabilities()
213:         if rebuild or not capabilities.catch_up:
214:             raise NotSupported(
215:                 "projection rebuild is not supported"
216:                 if rebuild
217:                 else f"backend {descriptor.backend!r} does not support projection catch-up"
218:             )
219:         if capabilities.rebuild:
220:             raise TypeError(
221:                 "first-stage maintenance providers must advertise rebuild=False"
222:             )
223: 
224:         projected_before, before_generation = _snapshot_state(
225:             query, ledger, bound=captured
226:         )
227:         if (
228:             projected_before is not None
229:             and projected_before.lineage == target.lineage
230:             and projected_before.ordinal >= target.ordinal
231:         ):
232:             generation = (
233:                 None
234:                 if projected_before.ordinal > captured.ordinal
235:                 else before_generation
236:             )
237:             return ProjectionSyncResult(
238:                 captured_head=captured,
239:                 target=target,
240:                 projected_before=projected_before,
241:                 projected_after=projected_before,
242:                 view_generation=generation,
243:                 changed=False,
244:             )
245: 
246:         try:
247:             advance = maintenance.catch_up(target)
248:             projected_after = _resolve_head(ledger, advance.after)
249:             if projected_after is None:  # the adapter contract requires one
250:                 raise TypeError("maintenance returned no resulting watermark")
251:             if projected_after.lineage != target.lineage:
252:                 raise NotAuthority("maintenance returned a foreign projection lineage")
253:             if projected_after.ordinal < target.ordinal:
254:                 raise HeadMismatch("maintenance returned a prefix behind its target")
255:             if projected_after.ordinal == target.ordinal and projected_after != target:
256:                 raise HeadMismatch(
257:                     "maintenance returned a same-height head different from its target"
258:                 )
259:             projected_before = _resolve_head(ledger, advance.before)
260:             observed_after, generation = _snapshot_state(
261:                 query, ledger, bound=projected_after
262:             )
263:             if (
264:                 observed_after is None
265:                 or observed_after.lineage != projected_after.lineage
266:                 or observed_after.ordinal < projected_after.ordinal
267:             ):
268:                 raise HeadMismatch(
269:                     "query projection does not reach the prefix maintenance returned"
270:                 )
271:             if observed_after != projected_after:
272:                 # Another serialized maintainer advanced after this handle
273:                 # committed. The first snapshot's generation is bounded at
274:                 # our returned prefix, so it cannot identify the newer view.
275:                 # Report the attested actual prefix and leave generation
276:                 # unknown rather than mislabel it or report success as loss.
277:                 projected_after = observed_after
278:                 generation = None
279:             return ProjectionSyncResult(
280:                 captured_head=captured,
281:                 target=target,
282:                 projected_before=projected_before,
283:                 projected_after=projected_after,
284:                 view_generation=generation,
285:                 changed=advance.changed,
286:             )
287:         except Exception as cause:
288:             observed_after = None
289:             with suppress(Exception):
290:                 observed_after, _ = _snapshot_state(query, ledger, bound=captured)
291:             raise ProjectionSyncError(
292:                 f"projection maintenance through ordinal {target.ordinal} failed; "
293:                 "inspect observed_after before retrying",
294:                 target=target,
295:                 projected_before=projected_before,
296:                 observed_after=observed_after,
297:                 cause=cause,
298:                 coordinator_phase="derived-sync",
299:                 effects={
300:                     "derived": {"attempt": "entered", "state": "unknown"}
301:                 },
302:             ) from cause
303:     finally:
304:         if maintenance is not None:
305:             _close_quietly(maintenance)
306:         _close_quietly(query)
307:         _close_quietly(ledger)

225:     """Two explicit causal nodes, never implicit context or record bodies."""

438:     exc: BaseException, *, context: Mapping[str, Any] | None = None
439: ) -> BaseException:
440:     """Return a stable SDK exception for engine boundary failures.
441: 
442:     Existing SDK errors are returned unchanged.  Classification is by the
443:     engine's public exception types, never by message text.
444:     """
445:     if not isinstance(exc, Exception) or isinstance(exc, SdkError):
446:         return exc
447:     details = _identity_details(exc)
448:     context_fields = {
449:         "captured_head",
450:         "fact_id",
451:         "tick_id",
452:         "fact_ids",
453:         "tick_ids",
454:         "items",
455:         "item_index",
456:     }
457:     for name, value in (context or {}).items():
458:         if name in details or name not in context_fields:
459:             continue
460:         coordinate = _context_detail(name, value)
461:         if coordinate is not None or value is None:
462:             details[name] = coordinate

464:     credential = _credential_binding_details(exc)
465:     if credential is not None:
466:         details["credential_binding"] = credential
467:     evidence = _evidence_details(exc, details)
468:     if evidence is not None:
469:         details["evidence"] = evidence

471:         return CommittedOutcomeUnknown(str(exc), source_type=type(exc).__name__, details=details)
472:     if isinstance(exc, InitializationOutcomeUnknown):

474:     if isinstance(exc, (AdoptionOutcomeUnknown, DeclarationOutcomeUnknown, RestoreForwardUnknown)):
475:         return CommittedOutcomeUnknown(str(exc), source_type=type(exc).__name__, details=details)
