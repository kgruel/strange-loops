537: async def _close_owned_streams(
538:     stream: object | None,
539:     iterator: object | None,
540:     *,
541:     primary: BaseException | None,
542: ) -> Exception | None:
543:     """Attempt async close on the iterator and owner before propagating interruption.
544: 
545:     This releases Python iterator resources only. It does not claim that an
546:     underlying collector reversed or terminated any external effect.
547:     """
548:     failures: list[Exception] = []
549:     handles: list[object] = []
550:     deferred_interrupt: BaseException | None = None
551:     primary_is_interrupt = primary is not None and not isinstance(primary, Exception)
552:     for handle in (iterator, stream):
553:         if handle is None or any(handle is existing for existing in handles):
554:             continue
555:         handles.append(handle)
556:         try:
557:             close = getattr(handle, "aclose", None)
558:             if not callable(close):
559:                 continue
560:             await close()
561:         except Exception as exc:
562:             failures.append(exc)
563:         except BaseException as exc:
564:             if not primary_is_interrupt and deferred_interrupt is None:
565:                 deferred_interrupt = exc
566:     if deferred_interrupt is not None:
567:         raise deferred_interrupt
568:     return failures[0] if failures else None

571: async def _collect_one(
572:     invocation: SourceInvocationPlan,
573:     state: _CollectionState,
574:     source: Any,
575:     collector_factory: CollectorFactory | None,
576:     *,
577:     clock: Callable[[], float],
578:     id_factory: Callable[[], str],
579: ) -> CollectedSource:
580:     status: Literal["ok", "error", "invalid"] = "ok"
581:     stderr = ""
582:     returncode: int | None = None
583:     error_type: str | None = None
584:     error_message: str | None = None
585:     invalid: InvalidSourceOutput | None = None
586:     stream: object | None = None
587:     iterator: object | None = None
588:     primary: BaseException | None = None
589:     try:
590:         stream = source.collect() if collector_factory is None else collector_factory(source)
591:         iterator = aiter(stream)
592:         async for value in iterator:
593:             if not isinstance(value, Fact):
594:                 invalid = InvalidSourceOutput(state.source_index, value)
595:                 primary = invalid
596:                 status = "invalid"
597:                 break
598:             try:
599:                 fact_id = id_factory()

601:             except Exception as exc:
602:                 failure = _CollectionStepFailed(state, "bookkeeping", exc)
603:                 primary = failure
604:                 raise failure from exc
605:     except _CollectionStepFailed:
606:         raise
607:     except asyncio.CancelledError as exc:
608:         primary = exc
609:         raise
610:     except SourceError as exc:
611:         primary = exc
612:         status = "error"
613:         stderr = exc.stderr
614:         returncode = exc.returncode
615:         error_type = type(exc).__name__
616:         error_message = str(exc)
617:     except Exception as exc:
618:         primary = exc
619:         status = "error"
620:         error_type = type(exc).__name__
621:         error_message = str(exc)
622:     except BaseException as exc:
623:         primary = exc
624:         raise
625:     finally:
626:         cleanup_error = await _close_owned_streams(
627:             stream, iterator, primary=primary
628:         )
629:         if cleanup_error is not None and primary is None:
630:             raise _CollectionStepFailed(state, "cleanup", cleanup_error) from cleanup_error
631: 
632:     try:
633:         payload: dict[str, Any] = {
634:             "command": state.command,
635:             "status": "error" if status != "ok" else "ok",
636:             "count": len(state.pairs),
637:             "invocation_id": invocation.invocation_id,
638:             "source_index": state.source_index,
639:         }
640:         if stderr:
641:             payload["stderr"] = stderr
642:         if returncode is not None:
643:             payload["returncode"] = returncode
644:         if error_type is not None:
645:             payload["error_type"] = error_type
646:         if error_message is not None:
647:             payload["error"] = error_message
648:         if invalid is not None:
649:             payload["error_type"] = type(invalid).__name__
650:             payload["error"] = str(invalid)
651:         lifecycle = Fact(
652:             f"_sync.{state.kind}",
653:             clock(),
654:             payload,
655:             observer=invocation.observer,
656:         )
657:         return CollectedSource(
658:             source_index=state.source_index,
659:             kind=state.kind,
660:             command=state.command,
661:             source_observer=state.source_observer,
662:             facts=tuple(fact for fact, _fact_id in state.pairs),
663:             fact_ids=tuple(fact_id for _fact, fact_id in state.pairs),
664:             lifecycle_fact=lifecycle,
665:             lifecycle_id=invocation.lifecycle_ids[state.source_index],
666:             status=status,
667:             stderr=stderr,
668:             returncode=returncode,
669:             error_type=error_type,
670:             error_message=error_message,
671:             invalid_output=invalid,
672:         )
673:     except Exception as exc:
674:         raise _CollectionStepFailed(state, "bookkeeping", exc) from exc

677: async def collect_source_tier(
678:     invocation: SourceInvocationPlan,
679:     capture: RuntimeCapture,
680:     tier_index: int,
681:     collector_factory: CollectorFactory | None = None,
682:     *,
683:     clock: Callable[[], float] = time.time,
684:     id_factory: Callable[[], str] = _new_id,
685: ) -> CollectedTier:
686:     """Collect one fixed tier concurrently while retaining source-list order."""
687:     indices = invocation.tiers[tier_index]
688:     sources = invocation.initial_capture.sources
689:     states = [
690:         _CollectionState(
691:             source_index=index,
692:             kind=sources[index][0].kind,
693:             command=sources[index][0].command,
694:             source_observer=sources[index][0].observer,
695:             pairs=[],

706:                 clock=clock,
707:                 id_factory=id_factory,
708:             )

710:         for state in states

713:         collected = tuple(await asyncio.gather(*tasks))
714:     except _CollectionStepFailed as primary:
715:         for task in tasks:
716:             if not task.done():
717:                 task.cancel()

719:         completed = tuple(
720:             result for result in settled if isinstance(result, CollectedSource)
721:         )
722:         failed = tuple(
723:             result.state.partial(result.phase, result.cause)
724:             for result in settled
725:             if isinstance(result, _CollectionStepFailed)
726:         )
727:         cancelled = tuple(
728:             state.partial("cancelled")
729:             for state, result in zip(states, settled, strict=True)

731:         )

733:             tier_index=tier_index,
734:             basis=capture.basis,
735:             completed_sources=completed,
736:             failed_sources=failed,
737:             cancelled_sources=cancelled,
738:             cause=primary.cause,
739:         ) from primary.cause
740:     except BaseException:
741:         for task in tasks:
742:             if not task.done():
743:                 task.cancel()
744:         await asyncio.gather(*tasks, return_exceptions=True)
745:         raise
746:     return CollectedTier(
747:         tier_index=tier_index,
748:         basis=capture.basis,
749:         sources=collected,
750:         include_pending_boundaries=tier_index == 0,

429: async def run_sources(
430:     target: Path | str,
431:     *,
432:     observer: str,
433:     force: bool = False,

441:     target_path = Path(target).resolve()
442:     arrival = _arrival_descriptor(target_path)
443:     if arrival is None:
444:         _refuse_arrival_aggregate_members(target_path)
445:         raise TargetUnsupported(

447:         )
448:     _path, locator, descriptor = arrival

456:     provider = credentials or CustodyCredentialProvider()

469:         raise
470:     except SourcePreparationRefused as exc:
