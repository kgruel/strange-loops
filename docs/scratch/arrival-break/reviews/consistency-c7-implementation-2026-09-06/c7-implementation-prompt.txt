41: def resolve_target(target: Path | str) -> TargetInfo:
42:     """Resolve and probe a target path without modifying it or opening SQLite.
43:
44:     Returns `TargetInfo` describing target type, canonical path, and mode.
45:     Raises `TargetNotFound` if the path doesn't exist, or `TargetUnsupported`
46:     if the target cannot be classified as a loops artifact.
47:     """
48:     path = Path(target).resolve()
49:     if not path.exists():
50:         raise TargetNotFound(f"target path does not exist: {path}")
51:
52:     info = probe_target(path)
53:     if info.target_type == "unknown":
54:         raise TargetUnsupported(
55:             f"target {path} is not a recognized loops artifact "
56:             "(accepted: .vertex, .jsonl, .db, .sqlite)"
57:         )
58:
59:     return info

62: def _arrival_definition(
63:     target: Path | str, *, allow_aggregate: bool = False
64: ) -> tuple[Path, Any, StoreDescriptor | None] | None:
65:     """Parse one Arrival-capable root without probing or opening its residence.
66:
67:     ``TargetInfo`` is a legacy filesystem classification and necessarily turns
68:     a store spelling into a ``Path``. Descriptor resolution instead parses the
69:     vertex locator directly, so another backend's DSN or service URL remains
70:     byte-for-byte opaque until that named adapter receives it. Aggregate shape
71:     remains refused by default; declaration inspection may opt in because it
72:     reports bounded root structure without claiming one basis for the aggregate.
73:
74:     A valid storeless vertex retains its parsed AST with ``descriptor=None`` so
75:     a structural caller can make its descriptor/storeless decision from these
76:     same bytes. Malformed and non-vertex targets retain the legacy ``None``
77:     classification.
78:     """
79:     path = Path(target).resolve()
80:     if path.suffix.lower() != ".vertex":
81:         return None
82:     if not path.exists():
83:         raise TargetNotFound(f"target path does not exist: {path}")
84:     try:
85:         ast = parse_vertex_file(path)
86:     except Exception:
87:         return None  # the legacy resolver owns malformed-target classification
88:     descriptor = descriptor_for(ast, path)
89:     if descriptor is None:
90:         return path, ast, None
91:     if descriptor.role is None:
92:         raise SdkValueError(
93:             f"Arrival target {path} must declare store role explicitly"
94:         )
95:     if not allow_aggregate and (ast.combine is not None or ast.discover is not None):
96:         raise TargetUnsupported(
97:             f"Arrival target {path} is an aggregate; member-basis reads are not "
98:             "implemented in this single-store stage"
99:         )
100:     return path, ast, descriptor

105: ) -> tuple[Path, Any, StoreDescriptor] | None:

107:     resolved = _arrival_definition(target, allow_aggregate=allow_aggregate)
108:     if resolved is None:
109:         return None
110:     path, ast, descriptor = resolved
111:     if descriptor is None:
112:         return None
113:     return path, ast, descriptor

120:         return

+def _opaque_root(
+    tmp_path: Path, *, effective_aggregate: bool = False
+) -> tuple[Path, ArrivalLog, str]:
+    log = ArrivalLog.mint(
+        tmp_path / "root.arrival",
+        observer="custodian",
+        signer=_sign,
+        key=base64.b64encode(b"k" * 32).decode(),
+        at=1.0,
+    )
+    # Keep the binding identity unique across parametrized tests: the
+    # attestation journal is intentionally persistent for an opaque locator.
+    locator = f"opaque://tenant//root-{tmp_path.name}?projection=contract"
+    vertex = tmp_path / "root.vertex"
+    shape = 'discover "missing/*.vertex"\n' if effective_aggregate else ""
+    vertex.write_text(
+        f'name "root"\nstore "{locator}" backend="opaque" '
+        f'lineage="{log.lineage()}" role="authority"\n'
+        + shape
+        + 'loops { note { fold { items "collect" 5 } } }\n',
+        encoding="utf-8",
+    )
+    _append_genesis(log, vertex)
+    return vertex, log, locator

+class _TrackingLedger(FileLedger):
+    def __init__(self, log: ArrivalLog, events: list[str]) -> None:
+        super().__init__(log)
+        self._events = events
+
+    def close(self) -> None:
+        self._events.append("ledger")
+        super().close()

+class _TrackingQuery(FileQuery):
+    def __init__(self, path: Path, events: list[str]) -> None:
+        super().__init__(path)
+        self._events = events
+
+    def close(self) -> None:
+        self._events.append("query")
+        super().close()

+def _registry(
+    locator: str, log: ArrivalLog, events: list[str], calls: list[str]
+) -> BackendRegistry:
+    registry = BackendRegistry()
+
+    def opener(descriptor):
+        calls.append(descriptor.location)
+        if descriptor.location != locator:
+            raise AssertionError(f"member open attempted: {descriptor.location!r}")
+        return _TrackingLedger(log, events), _TrackingQuery(
+            index_path_for(log.path), events
+        )
+
+    registry.register("opaque", opener)
+    return registry
