522:         expected_public = intent.get("expected_public_key")
523:         if expected_public is not None and published.get("public_key") != expected_public:
524:             raise BindingConflict("published mapped binding contradicts its expected public key")
525:         intents = self._subdir("intents-v1", create=False)
526:         if intents is None:
527:             if require_token_index:
528:                 raise BindingConflict("published mapped binding has no token-index evidence")
529:             return
530:         indexed = self._read_json(intents / f"{_digest(str(intent['token']))}.json")
531:         if require_token_index and indexed != intent:
532:             raise BindingConflict("published mapped binding contradicts its token index")
533:         completion = self._read_json(
534:             intents / f"{_digest(str(intent['token']))}.complete.json"
535:         )

539:             if completion != complete:
540:                 raise BindingConflict("published mapped binding contradicts its completion receipt")

542:     def _intent_for(
543:         self, observer: str, *, token: str, provenance: str, key_ref: str | None = None,
544:         expected_public_key: str | None = None,
545:     ) -> dict[str, object]:
546:         intents = self._subdir("intents-v1", create=False)
547:         token_record = (
548:             None if intents is None else self._read_json(intents / f"{_digest(token)}.json")
549:         )
550:         if token_record is not None and (
551:             token_record.get("namespace") != self.namespace
552:             or token_record.get("observer") != observer
553:         ):
554:             raise BindingConflict("mapped binding token already names another intent")
555:         pending_for_token = self._pending_by_token(token)
556:         if pending_for_token is not None:
557:             if token_record is not None and token_record != pending_for_token:
558:                 raise BindingConflict("mapped binding pending slot contradicts its token index")
559:             if (
560:                 pending_for_token.get("namespace") != self.namespace
561:                 or pending_for_token.get("observer") != observer
562:             ):
563:                 raise BindingConflict("mapped binding token already owns another pending slot")
564:         pending_dir = self._subdir("pending-v1", create=False)
565:         slot_pending = (
566:             None
567:             if pending_dir is None
568:             else self._read_json(
569:                 pending_dir / f"{self._slot(self.namespace, observer)}.json"
570:             )
571:         )
572:         completed = self._validated_binding(self.namespace, observer)
573:         if completed is not None:
574:             if slot_pending is not None:
575:                 # A caller may legitimately retry an already-completed binding
576:                 # under a fresh operation token. The selected published material
577:                 # must still agree with this slot's original durable intent.
578:                 self._assert_published_intent(
579:                     completed, slot_pending, require_token_index=False
580:                 )
581:                 indexed_slot = (
582:                     None
583:                     if intents is None
584:                     else self._read_json(
585:                         intents / f"{_digest(str(slot_pending['token']))}.json"
586:                     )
587:                 )
588:                 if indexed_slot is not None:
589:                     self._assert_published_intent(completed, slot_pending)
590:             if pending_for_token is not None:
591:                 # The pending marker may be the only durable token evidence
592:                 # after interruption between its publication and token-index
593:                 # publication. Never restore that index until the published
594:                 # slot proves it still names this exact intent.
595:                 self._assert_published_intent(
596:                     completed, pending_for_token, require_token_index=False
597:                 )
598:                 self._ensure_token_index(pending_for_token)
599:                 self._assert_published_intent(completed, pending_for_token)
600:                 self._publish_completion(pending_for_token)
601:             if (
602:                 (expected_public_key is not None and completed["public_key"] != expected_public_key)
603:                 or (key_ref is not None and completed["key_ref"] != key_ref)
604:                 or (
605:                     token_record is not None
606:                     and (
607:                         token_record.get("provenance") != provenance
608:                         or token_record.get("key_ref") != completed["key_ref"]
609:                     )
610:                 )
611:             ):
612:                 raise BindingConflict("existing mapped binding has a different public key")
613:             return {
614:                 "completed": True,
615:                 "key_ref": completed["key_ref"],
616:                 "public_key": completed["public_key"],
617:                 "provenance": completed["provenance"],
618:                 "token": token,
619:             }
620:         pending = slot_pending
621:         if pending is not None:
622:             if pending_for_token is not None and pending_for_token != pending:
623:                 raise BindingConflict("mapped binding slot contradicts its token reservation")
624:             if pending.get("token") != token:
625:                 raise BindingRecoveryRequired(
626:                     "a different durable token owns this mapped binding slot"
627:                 )
628:             if (
629:                 pending.get("namespace") != self.namespace
630:                 or pending.get("observer") != observer
631:                 or pending.get("provenance") != provenance
632:                 or (key_ref is not None and pending.get("key_ref") != key_ref)
633:                 or (
634:                     expected_public_key is not None
635:                     and pending.get("expected_public_key") != expected_public_key
636:                 )
637:             ):
638:                 raise BindingConflict("mapped binding token was reused with different intent")
639:             self._ensure_token_index(pending)
640:             return pending
641:         intent = self._new_intent(
642:             observer,
643:             token=token,
644:             provenance=provenance,
645:             key_ref=key_ref,
646:             expected_public_key=expected_public_key,
647:         )
648:         pending_dir = self._subdir("pending-v1", create=True)
649:         assert pending_dir is not None
650:         # The slot marker contains the complete intent. If a process dies before
651:         # the token index is linked, callers which know observer/token can still
652:         # recover coherently; no directory scan or candidate deletion occurs.
653:         if not _publish_no_clobber(
654:             pending_dir / f"{self._slot(self.namespace, observer)}.json", _json_bytes(intent)
655:         ):
656:             return self._intent_for(
657:                 observer,
658:                 token=token,
659:                 provenance=provenance,
660:                 key_ref=key_ref,
661:                 expected_public_key=expected_public_key,
662:             )
663:         self._ensure_token_index(intent)
664:         return intent

666:     def _pending_for_recovery(self, observer: str, token: str) -> dict[str, object]:
667:         pending_dir = self._subdir("pending-v1", create=False)
668:         if pending_dir is None:
669:             raise BindingRecoveryRequired("no durable mapped binding intent exists")
670:         pending = self._read_json(
671:             pending_dir / f"{self._slot(self.namespace, observer)}.json"
672:         )
673:         if pending is None:
674:             raise BindingRecoveryRequired("no durable mapped binding intent exists")
675:         if (
676:             pending.get("schema") != _INTENT_SCHEMA
677:             or pending.get("namespace") != self.namespace
678:             or pending.get("observer") != observer
679:             or pending.get("token") != token
680:             or pending.get("stage") != "pending"
681:             or pending.get("provenance") not in {"created", "legacy-import", "existing-ref"}
682:             or not isinstance(pending.get("key_ref"), str)
683:         ):
684:             raise BindingRecoveryRequired("mapped binding recovery token does not match a pending intent")

686:             raise BindingConflict("mapped binding token owns a contradictory pending slot")

688:         return pending

690:     def _hook(self, phase: str, intent: dict[str, object]) -> None:
691:         if self._creation_hook is not None:

694:     def _ensure_created_key(self, intent: dict[str, object]):
695:         key_ref = str(intent["key_ref"])
696:         keys = self._subdir("keys-v1", create=True)
697:         assert keys is not None
698:         key_dir = keys / key_ref
699:         mode = _lstat_kind(key_dir)
700:         created = False
701:         if mode is None:
702:             _directory(key_dir, create=True)
703:             # A newly private directory is the only place convenience creation
704:             # is permitted. Existing managed objects are always loaded/checked.
705:             pair = ed25519.load_or_generate(key_dir)
706:             created = True
707:         else:
708:             if not S_ISDIR(mode):
709:                 raise ValueError("mapped custody candidate key path is not a directory")
710:             private = key_dir / "ed25519.key"
711:             public = key_dir / "ed25519.pub"
712:             if _lstat_kind(private) is None:
713:                 # This intent exclusively reserved the candidate ref. It is
714:                 # safe to resume a crash before the private object existed.
715:                 if _lstat_kind(public) is not None:
716:                     raise ValueError("incomplete candidate has public evidence without private key")
717:                 pair = ed25519.load_or_generate(key_dir)
718:                 created = True
719:             else:

721:                 if _lstat_kind(public) is None:
722:                     # A crash between durable private and public publication is
723:                     # resumable only for this retained, provider-owned intent.
724:                     _publish_no_clobber(public, (pair.public_b64 + "\n").encode(), mode=0o644)
725:                     _fsync_directory(key_dir)
726:                 pair = self._load_keypair(key_ref, require_public_file=True)
727:         self._hook("key-published", intent)
728:         return pair, created

730:     def _copy_key(self, source: Path, intent: dict[str, object]):
731:         source_pair = self._load_external_key(source)
732:         keys = self._subdir("keys-v1", create=True)
733:         assert keys is not None
734:         key_dir = keys / str(intent["key_ref"])
735:         if _lstat_kind(key_dir) is None:
736:             _directory(key_dir, create=True)
737:             _publish_no_clobber(
738:                 key_dir / "ed25519.key", (source / "ed25519.key").read_bytes()
739:             )
740:             _publish_no_clobber(
741:                 key_dir / "ed25519.pub", (source_pair.public_b64 + "\n").encode(), mode=0o644
742:             )
743:             _fsync_directory(key_dir)
744:             created = True
745:         else:
746:             source_pair = self._load_keypair(
747:                 str(intent["key_ref"]), require_public_file=True
748:             )
749:             created = False

751:         if candidate.public_b64 != source_pair.public_b64:
752:             raise BindingConflict("copied candidate disagrees with the validated source key")

754:         return candidate, created

756:     def _load_external_key(self, key_dir: Path):
757:         mode = _lstat_kind(key_dir)
758:         if mode is None or not S_ISDIR(mode):
759:             raise FileNotFoundError(key_dir)
760:         private = key_dir / "ed25519.key"
761:         _regular(private)
762:         pair = ed25519.load(key_dir)
763:         public = key_dir / "ed25519.pub"
764:         if _regular(public, required=False) and public.read_text(encoding="utf-8") != pair.public_b64 + "\n":
765:             raise ValueError("source custody public evidence disagrees with private key")
766:         return pair

768:     def _publish_binding(self, intent: dict[str, object], pair) -> tuple[dict[str, object], bool]:
769:         expected = intent.get("expected_public_key")
770:         if expected is not None and pair.public_b64 != expected:
771:             raise BindingConflict("candidate key does not match expected public key")
772:         record: dict[str, object] = {
773:             "schema": _SCHEMA,
774:             "namespace": self.namespace,
775:             "observer": intent["observer"],
776:             "key_ref": intent["key_ref"],
777:             "algorithm": "ed25519",
778:             "public_key": pair.public_b64,
779:             "provenance": intent["provenance"],
780:         }
781:         bindings = self._subdir("bindings-v1", create=True)
782:         assert bindings is not None
783:         published = _publish_no_clobber(
784:             bindings / f"{self._slot(self.namespace, str(intent['observer']))}.json", _json_bytes(record)
785:         )
786:         if not published:
787:             existing = self._binding_record(self.namespace, str(intent["observer"]))
788:             assert existing is not None
789:             if existing["public_key"] != pair.public_b64:
790:                 raise BindingConflict("another binding won with a different public key")
791:             return existing, False
792:         self._hook("binding-published", intent)
793:         # Intent files are retained as durable recovery evidence. The marker is
794:         # deliberately not removed automatically, avoiding unsafe cleanup races.
795:         self._publish_completion(intent)

===== libs/custody/README.md =====
1: # custody
2: 
3: Loops signing composition — the store's at-rest signing format, below every writer.
4: 
5: Owns the `loops-tick-v1` / `loops-fact-v1` domain-separation constants, the
6: custody-co-located `keys/` layout (per-observer nesting + flat self-observer
7: back-compat), the signer/verifier builders, and `ensure_signing_key`. Observer
8: names remain exact identities: a filesystem spelling collision refuses rather
9: than borrowing another observer's key. A renamed vertex may retain a nested
10: self key; all signer domains and `ensure_signing_key` select that one key.
11: Identical flat and nested self copies are one self key; conflicting key material
12: refuses until repaired. A differently spelled nested observer remains a
13: distinct observer and is never used as the self fallback. Observer components
14: cannot name the reserved `ed25519.key` or `ed25519.pub` custody files when
15: mapped to directories. This is a local custody-layout restriction, not an
16: observer label restriction in the event protocol. A self observer with such a
17: stem uses only the flat layout.
18: 
19: Signer loading never creates keys. Concurrent explicit key creation publishes
20: one complete private key; all callers load that key and derive its public file.
21: 
22: Any client that emits signed facts or ticks must compose identically or
23: `sl store verify` reports breaks — that's why this is a lib, not CLI code.
24: Depends on `sign` (loops-agnostic Ed25519 primitives) and `engine`
25: (store-canonical declaration resolution). The engine itself never imports
26: this — it takes signer callables by injection.
27: 
28: 
29: ## Explicit mapped bindings
30: 
31: `MappedCredentialProvider(root, namespace=..., receipt_observer=...)` provides
32: an opt-in alternative to locator-based custody. Exact namespace/observer
33: strings select an opaque managed key reference recorded with its public key.
34: All three signing domains use that binding with separate domain prefixes.
35: Hashed filesystem slots are storage details; labels retain their exact case,
36: Unicode spelling, and non-path characters. A binding does not authorize an
37: Arrival claim by itself: the engine separately checks captured history.
38: 
39: `for_write` returns lazy configuration. `resolve` loads and validates the
40: binding, private-derived public material, and any existing public convenience
41: file without creating or repairing anything. Missing unbound identities can
42: remain unsigned; malformed mappings, conflicting public material, missing
43: referenced private keys, and filesystem aliases refuse.
44: 
45: Explicit mutations are `create_binding(observer, token=...)`,
46: `import_legacy(vertex_path, observer, token=...)`, and
47: `bind_existing_ref(observer, key_ref, expected_public_key, token=...)`.
48: Import copies private material into managed custody and preserves the original
49: files. Reusing a managed reference is deliberate and does not introduce its
50: public key into an Arrival lineage. Existing bindings are not silently
51: replaced; local rotation is a separate future operation.
52: 
53: Mutations use a provider-wide POSIX advisory lock, durable intents, and
54: complete-file publication with no-clobber links. The supported filesystem must
55: provide those lock, link, and fsync semantics. Concurrent creators converge
56: on one binding; resolution remains read-only. There is no automatic orphan
57: cleanup or key deletion.
58: 
59: Keep the operation token when a mutation is interrupted.
60: `BindingMutationIncomplete` retains namespace, observer, token, candidate key
61: reference, and phase; the binding may already have been published.
62: `recover_binding(observer, token=...)` reconciles the retained intent and
63: candidate. It never replaces a missing key referenced by a published binding.
64: Explicit recovery may restore a missing token index from its retained intent
65: before a later candidate-key refusal; it does not claim zero filesystem effects
66: or create a key/binding in that case.
67: An import interrupted before copying private material needs the original
68: source re-presented with the same import token and expected public material;
69: recovery cannot invent that key. A completed imported candidate can recover
70: without the old source directory. Retrying with a different token does not
71: bypass an unfinished operation. Creation results report the returned binding
72: and effects of that invocation; they contain no private key material.
