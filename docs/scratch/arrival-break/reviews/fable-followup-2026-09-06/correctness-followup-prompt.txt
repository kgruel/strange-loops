```python
277:     active: list[Path] = []
278: 
279:     def visit(node: AggregatePlanNode) -> CapturedDefinition:
280:         locator = node.locator.resolve()
281:         if locator in active:
282:             chain = " -> ".join(str(path) for path in (*active, locator))
283:             raise AggregateCycle(f"composition cycle: {chain}")
284:         active.append(locator)
285:         definition_index = len(definitions)
286:         definitions.append(None)
287:         try:
288:             if node.descriptor is None:
289:                 if getattr(node.local_ast, "store", None) is not None:
290:                     raise AggregateCaptureRefused(
291:                         "aggregate occurrence "
292:                         f"{node.locator} has storage without an explicit Arrival descriptor"
293:                     )
294:                 effective = _detached(node.local_ast)
295:                 basis = None
296:             else:
297:                 opened = opened_root if node is root and opened_root is not None else open_read(
298:                     registry, node.descriptor
299:                 )
300:                 try:
301:                     effective, declaration_identity = _effective_declaration(node, opened)
302:                 except BaseException:
303:                     with suppress(BaseException):
304:                         opened.close()
305:                     raise
306:                 captured = _CapturedNode(
307:                     node, opened, _detached(effective), declaration_identity
308:                 )
309:                 captured_nodes.append(captured)
310:                 effective = captured.effective_declaration
311:                 basis = captured.opened.basis
312:                 # Discover's own descriptor is a retained overlay role on this
313:                 # exact resource, never a recursive self-child or second open.
314:                 if effective.discover is not None:
315:                     members.append(_member(captured, role="own-overlay"))
316:                 elif node.role != "root" or effective.combine is None:
317:                     # A descriptor root may resolve to a plain declaration
318:                     # even when its local ingress still describes composition.
319:                     # It remains a leaf occurrence of its own captured store.
320:                     members.append(_member(captured, role=node.role))
321:             child_definitions = tuple(
322:                 visit(child) for child in resolver.children(node, effective)
323:             )
324:             definition = CapturedDefinition(
325:                 identity=_occurrence(node),
326:                 local_definition=node.definition,
327:                 effective_declaration=effective,
328:                 basis=basis,
329:                 children=tuple(child.identity for child in child_definitions),
330:             )
331:             definitions[definition_index] = definition
332:             return definition
333:         finally:
334:             active.pop()
335: 
336:     try:
337:         visit(root)
338:         return AggregateCapture(
339:             tuple(members),
340:             tuple(definition for definition in definitions if definition is not None),
341:             tuple(captured_nodes),
342:         )
343:     except BaseException:
344:         for node in reversed(captured_nodes):
345:             with suppress(BaseException):
346:                 node.close()
347:         raise
```
