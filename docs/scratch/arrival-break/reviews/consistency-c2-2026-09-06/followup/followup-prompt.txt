178: def _cause_details(exc: BaseException) -> dict[str, Any] | None:
179:     """Two explicit causal nodes, never implicit context or record bodies."""
180:     seen = {id(exc)}
181:     result: dict[str, Any] | None = None
182:     parent: dict[str, Any] | None = None
183:     for _ in range(2):

185:         if cause is None or id(cause) in seen:
186:             break

208: def _evidence_details(exc: BaseException, details: Mapping[str, Any]) -> dict[str, Any] | None:

218:             effect = effects.get(resource)

220:                 continue

394:     """Return a stable SDK exception for engine boundary failures.

463: def _publish_cache(
464:     plan: DeclarationEditPlan,
465:     *,
466:     intent: Path,
467:     intent_data: dict[str, Any],
468:     failure_hook: Callable[[str], None] | None,
469: ) -> None:
470:     """Detach the verified old cache, then exclusively link a full new file."""
471:     target = plan.target_path
472:     temp = (
473:         Path(intent_data["cache_temp"])
474:         if intent_data.get("cache_temp")
475:         else _unique_sibling(target, "cache")
476:     )
477:     backup = (
478:         Path(intent_data["cache_backup"])
479:         if intent_data.get("cache_backup")
480:         else _unique_sibling(target, "backup")
481:     )
482:     intent_data["cache_temp"] = str(temp)
483:     intent_data["cache_backup"] = str(backup)
484:     _replace_json(intent, intent_data)
485:     try:
486:         proposed = plan.proposed_text.encode("utf-8")
487:         if temp.exists():
488:             _verify_cache_temp(temp, proposed)
489:         else:
490:             _create_cache_temp(temp, proposed)
491:             if failure_hook:
492:                 failure_hook("after-cache-temp")
493:         if target.exists():
494:             current = target.read_bytes()
495:             if current != plan.old_cache_bytes:
496:                 raise DeclarationStale(
497:                     "declaration cache changed after preparation; refusing publication"
498:                 )
499:             if backup.exists():
500:                 raise DeclarationApplyError(
501:                     "a prior cache publication left a backup beside the live declaration"
502:                 )
503:             target.rename(backup)
504:             _fsync_parent(target)
505:         elif not backup.exists():
506:             raise DeclarationApplyError(
507:                 "declaration cache is absent without the verified detached backup"
508:             )
509:         if failure_hook:
510:             failure_hook("after-cache-detach")
511:         if backup.read_bytes() != plan.old_cache_bytes:
512:             raise DeclarationApplyError(
513:                 "detached declaration cache bytes differ from prepared evidence"
514:             )
515:         os.link(temp, target)
516:         temp.unlink()
517:         _fsync_parent(target)
518:         if failure_hook:
519:             failure_hook("after-cache-link")
520:         with suppress(FileNotFoundError):
521:             backup.unlink()
522:         _fsync_parent(backup)
523:     except BaseException:
524:         raise

798: def prepare_declaration_edit(
799:     registry: BackendRegistry,
800:     descriptor: StoreDescriptor,
801:     *,
802:     target: Path | str,
803:     proposed_text: str,
804:     observer: str,
805:     credentials: WriteCredentials | None,
806:     fact_verify: Callable[[str, str, str], bool],
807:     arrival_verify: Callable[[str, str, str], bool],
808:     source_cache_bytes: bytes | None = None,
809:     require_cache_basis: bool = False,
810: ) -> DeclarationEditPlan:
811:     """Prepare one exact declaration transition against an attested CURRENT head.
812: 
813:     This function only reads and signs in memory. It creates no intent, does
814:     not append, and does not touch the declaration cache. The registry opener
815:     is expected to return an attested custody handle; its ``scan`` is the
816:     structurally verified genesis-through-head sequence required by the
817:     iterable authorship seam.
818:     """
819:     target_path = Path(target).resolve()
820:     if not isinstance(proposed_text, str) or not proposed_text:
821:         raise _preparation_refused("proposed declaration text is empty")
822:     try:
823:         old_cache_bytes = target_path.read_bytes()
824:     except OSError as exc:
825:         raise _preparation_refused(
826:             f"cannot read declaration cache {target_path}: {exc}"
827:         ) from exc
828:     if source_cache_bytes is not None and old_cache_bytes != source_cache_bytes:
829:         raise _preparation_refused(
830:             "declaration cache changed while preparing the requested splice"
831:         )
832:     try:
833:         ast = parse_vertex(proposed_text, path=target_path)
834:         validate_vertex(ast)
835:     except Exception as exc:
836:         raise _preparation_refused(
837:             f"proposed declaration is not valid Vertex grammar/semantics: {exc}"
838:         ) from exc
839:     _validate_residence(target_path, ast, descriptor)
840:     proposed_documents = tuple(
841:         document.as_json() for document in vertex_to_documents(ast)
842:     )
843: 
844:     ledger: ArrivalLedger | None = None
845:     query: ArrivalQuery | None = None
846:     snapshot: Any | None = None
847:     atomic_limit: int | None = None
848:     captured_head: Head | None = None
849:     try:
850:         ledger, query = registry.open(descriptor)
851:         report = getattr(ledger, "opened", None)
852:         comparison = getattr(report, "comparison", None)
853:         if isinstance(comparison, PreGenesis):
854:             if comparison.ledger_refusal is not None:
855:                 raise _preparation_refused(
856:                     f"Arrival ledger is not opened: {comparison.ledger_refusal}"
857:                 )
858:             raise _preparation_refused(
859:                 "Arrival ledger has no captured head for declaration editing"
860:             )
861:         if isinstance(comparison, Indeterminate):
862:             raise _preparation_refused(
863:                 f"Arrival head attestation is indeterminate: {comparison.refusal}"
864:             )
865:         if not isinstance(comparison, Compared):
866:             raise _preparation_refused(
867:                 "registry opener did not provide a compared Arrival head"
868:             )
869:         captured_head = comparison.presented
870:         snapshot = query.open_snapshot(
871:             captured_head=captured_head,
