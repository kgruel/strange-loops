222:                     lineage=captured.lineage,
223:                     captured_head=captured,
224:                     projected_through=captured,
225:                     view_generation=snapshot.view_generation,
226:                 )

554:                 "}",
555:                 "",
556:             ]
557:         )
558:         declaration_text = "\n".join(lines)
559:         try:
560:             ast = parse_vertex(declaration_text, path=vertex_path)
561:         except Exception as exc:
562:             raise SdkValueError(f"invalid Arrival declaration text: {exc}") from exc
563:         parsed_documents = [doc.as_json() for doc in vertex_to_documents(ast)]
564:         if documents is None:
565:             documents = parsed_documents
566:         elif documents != parsed_documents:
567:             raise SdkValueError(
568:                 "Arrival declaration text and declaration documents disagree"
569:             )
570:         if (
571:             ast.store_backend is None
572:             or ast.store_backend.name != chosen_backend
573:             or ast.store_backend.role != "authority"
574:             or ast.store_backend.lineage != lineage
575:         ):
576:             raise SdkValueError(
577:                 "Arrival declaration must carry the requested backend, lineage and authority role"
578:             )
579:     else:
580:         # Recovery reads the stable observer and descriptor from the intent;
581:         # the target is not re-rendered and no new lineage is selected.
582:         try:
583:             intent_data = intent_data or json.loads(arrival_intent_path(vertex_path).read_text())
584:             lineage = str(intent_data["lineage"])
585:             declaration_text = str(intent_data["declaration_text"])
586:             if documents is None:
587:                 documents = list(intent_data["documents"])
588:             reserved_descriptor = intent_data["descriptor"]
589:             chosen_backend = str(reserved_descriptor["backend"])
590:             descriptor_location = str(reserved_descriptor["location"])
591:             relative_location = descriptor_location
592:             parsed = parse_vertex(declaration_text, path=vertex_path)
593:             v_name = parsed.name
594:         except (OSError, ValueError, KeyError) as exc:
595:             raise TargetError(f"cannot load Arrival initialization intent: {exc}") from exc
596: 
597:     descriptor = StoreDescriptor(
598:         backend=chosen_backend,
599:         location=descriptor_location,
600:         lineage=lineage,
601:         role=Profile.AUTHORITY,
602:     )
603:     active_registry = (
604:         registry if registry is not None else BackendRegistry.with_builtin_backends()
605:     )
606:     try:
607:         if recover:
608:             result = recover_arrival_initialization(
609:                 active_registry,
610:                 arrival_intent_path(vertex_path),

616:                 target=vertex_path,
617:                 documents=documents or (),

619:                 observer=observer,
620:                 public_key=public_key,
621:                 signer=signer,
622:                 fact_signer=fact_signer,
623:                 arrival_signer=signer,
624:                 fact_verify=fact_verify,
625:                 arrival_verify=arrival_verify,
626:                 lineage=lineage,
627:             )
628:     except BaseException as exc:
629:         details = {"intent_path": str(arrival_intent_path(vertex_path))}
630:         context = {"fact_id": lineage} if lineage is not None else None
631:         normalized = normalize_exception(exc, context=context)
632:         if hasattr(normalized, "details"):
633:             normalized.details.update(details)
634:         if normalized is exc:
635:             # Keep SDK exceptions and process-control exceptions exactly as
636:             # raised.  ``raise exc from exc`` would install a self-cause and
637:             # obscure both cancellation and an already-established cause.
638:             raise

292: def normalize_exception(
293:     exc: BaseException, *, context: Mapping[str, Any] | None = None
294: ) -> BaseException:
295:     """Return a stable SDK exception for engine boundary failures.
296: 
297:     Existing SDK errors are returned unchanged.  Classification is by the
298:     engine's public exception types, never by message text.
299:     """
300:     if isinstance(exc, SdkError):
301:         return exc
302:     details = _identity_details(exc)
303:     context_fields = {
304:         "captured_head",
305:         "fact_id",
306:         "tick_id",
307:         "fact_ids",
308:         "tick_ids",
309:         "items",
310:         "item_index",
311:     }
312:     for name, value in (context or {}).items():
313:         if name in details or name not in context_fields:
314:             continue
315:         details[name] = _context_detail(name, value)
316:     details["source_type"] = type(exc).__name__
317:     if isinstance(exc, (BatchWriteCommitUnknown, WriteCommitUnknown)):
318:         return CommittedOutcomeUnknown(str(exc), source_type=type(exc).__name__, details=details)
319:     if isinstance(exc, InitializationOutcomeUnknown):
320:         return CommittedOutcomeUnknown(str(exc), source_type=type(exc).__name__, details=details)
321:     if isinstance(exc, (DeclarationOutcomeUnknown, RestoreForwardUnknown)):
322:         return CommittedOutcomeUnknown(str(exc), source_type=type(exc).__name__, details=details)
323:     if isinstance(exc, InitializationUnwitnessed):
324:         return CommittedUnwitnessed(str(exc), source_type=type(exc).__name__, details=details)
325:     if isinstance(exc, DeclarationUnwitnessed):
326:         return CommittedUnwitnessed(str(exc), source_type=type(exc).__name__, details=details)
327:     if isinstance(exc, InitializationCommittedIncomplete):
328:         return CommittedIncomplete(str(exc), source_type=type(exc).__name__, details=details)
329:     if isinstance(exc, (DeclarationCommittedIncomplete, RestoreForwardIncomplete)):
330:         return CommittedIncomplete(str(exc), source_type=type(exc).__name__, details=details)
331:     if isinstance(exc, NotWitnessed):
332:         return CommittedUnwitnessed(str(exc), source_type=type(exc).__name__, details=details)
333:     if isinstance(exc, DeclarationStale):
334:         return ArrivalRefusal(str(exc), source_type=type(exc).__name__, details=details)
335:     if isinstance(exc, BatchPostCommitProjectionFailed):
336:         return CommittedProjectionFailed(str(exc), source_type=type(exc).__name__, details=details)
337:     if isinstance(exc, PostCommitProjectionFailed):
338:         return CommittedProjectionFailed(str(exc), source_type=type(exc).__name__, details=details)
339:     if isinstance(exc, ProjectionSyncError):
340:         return ProjectionOutcomeUnknown(str(exc), source_type=type(exc).__name__, details=details)
341:     if isinstance(exc, SearchIndexSyncError):
342:         return ProjectionOutcomeUnknown(str(exc), source_type=type(exc).__name__, details=details)
343:     if isinstance(exc, BatchWritePreparationRefused):
344:         fact = exc.item.fact
345:         if exc.item.fact_id is not None:
346:             details.setdefault("fact_id", exc.item.fact_id)
347:         details.setdefault("observer", fact.observer)
348:         details.setdefault("kind", fact.kind)
349:         details.setdefault("vertex", exc.effective_declaration.name)
350:         if exc.admission_refused:
351:             return AdmissionRefusal(
352:                 str(exc),
353:                 source_type=type(exc).__name__,
354:                 details=details,
355:                 observer=fact.observer,
356:                 kind=fact.kind,
357:                 vertex=exc.effective_declaration.name,
358:             )
359:         return ArrivalRefusal(str(exc), source_type=type(exc).__name__, details=details)
360:     if isinstance(exc, AdmissionError):
361:         return AdmissionRefusal(
362:             str(exc),
363:             source_type=type(exc).__name__,
364:             details=details,
365:             observer=getattr(exc, "observer", None),
366:             kind=getattr(exc, "kind", None),
367:             vertex=getattr(exc, "vertex", None),
368:         )
369:     if isinstance(
370:         exc,
371:         (
372:             AttestationRefusal,
373:             ContractRefusal,
374:             ArrivalError,
375:             ProjectionSyncError,
376:             RuntimeWriteRefused,
377:         ),
378:     ):
379:         return ArrivalRefusal(str(exc), source_type=type(exc).__name__, details=details)
380:     return exc
