464: def _verify_arrival(
465:     arrival: Path,
466:     prepared: Path,
467:     source: _SourceEvidence,
468:     *,
469:     lineage: str,
470:     migration_ordinal: int,
471:     migration_hash: str,
472: ) -> dict[str, Any]:
473:     before_hash = _sha256(arrival)
474:     if not _ends_with_newline(arrival):
475:         raise ProvenanceRefused("Arrival target has an incomplete trailing record")
476:     metadata_count = migration_ordinal + 1 - source.rows
477:     if metadata_count < 1:
478:         raise ProvenanceRefused("pinned migration head cannot contain the prepared source")
479:     expected = iter(_prepared_rows(prepared))
480:     chain = _TickChain("Arrival target")
481:     registry: dict[str, list[str]] = {}
482:     facts = ticks = post_ticks = historical_explained = historical_unchanged = 0
483:     record_count = 0
484:     final_head: dict[str, Any] | None = None
485:     historical_tick_index = 0
486:     records = None
487:     try:
488:         records = ArrivalLog(arrival).walk()
489:         for record in records:
490:             ordinal = record["ord"]
491:             record_count += 1
492:             final_head = {
493:                 "lineage": record["lin"],
494:                 "ordinal": ordinal,
495:                 "record_hash": record["rh"],
496:             }
497:             if record["lin"] != lineage:
498:                 raise ProvenanceRefused("Arrival target lineage differs from pinned S")
499:             if ordinal < metadata_count:
500:                 if ordinal == 0:
501:                     if record["k"] != GENESIS_KIND:
502:                         raise ProvenanceRefused("migration prefix does not begin with genesis")
503:                     key = record["body"]["key"]
504:                     _verify_registry_signature(record, {record["observer"]: [key]}, ordinal)
505:                     registry[record["observer"]] = [key]
506:                 else:
507:                     if record["k"] != KEY_INTRODUCTION_KIND:
508:                         raise ProvenanceRefused(
509:                             f"non-protocol record appears before source at ordinal {ordinal}"
510:                         )
511:             if ordinal > 0 and record["k"] == KEY_INTRODUCTION_KIND:
512:                 _verify_registry_signature(record, registry, ordinal)
513:                 registry.setdefault(record["body"]["observer"], []).append(
514:                     record["body"]["key"]
515:                 )
516:             if metadata_count <= ordinal <= migration_ordinal:
517:                 try:
518:                     wanted = next(expected)
519:                 except StopIteration as exc:
520:                     raise ProvenanceRefused(
521:                         "migration prefix contains extra source rows"
522:                     ) from exc
523:                 actual = rows_of_record(record)
524:                 if len(actual) != 1 or actual[0] != wanted:
525:                     raise ProvenanceRefused(
526:                         f"migration prefix row mismatch at ordinal {ordinal}"
527:                     )
528:             if ordinal == migration_ordinal and record["rh"] != migration_hash:
529:                 raise ProvenanceRefused("Arrival record at S does not match the pinned hash")
530: 
531:             for kind, row in rows_of_record(record):
532:                 if kind == "fact":
533:                     facts += 1
534:                     fact_index = len(chain.fact_ids)
535:                     chain.fact(
536:                         row,
537:                         changed=(
538:                             ordinal <= migration_ordinal
539:                             and fact_index < len(source.changed_facts)
540:                             and source.changed_facts[fact_index]
541:                         ),
542:                     )
543:                 else:
544:                     ticks += 1
545:                     require = ordinal > migration_ordinal
546:                     calculated, transformed = chain.tick(
547:                         row, ordinal, require_window=require
548:                     )
549:                     if ordinal <= migration_ordinal:
550:                         historical_tick_index += 1
551:                         if calculated is not None and calculated != row[9]:
552:                             if not transformed:
553:                                 raise ProvenanceRefused(
554:                                     f"unexplained historical Arrival window mismatch at ordinal {ordinal}"
555:                                 )
556:                             historical_explained += 1
557:                         elif calculated is not None:
558:                             historical_unchanged += 1
559:                     else:
560:                         post_ticks += 1
561:                         if calculated != row[9]:
562:                             raise ProvenanceRefused(
563:                                 f"post-migration tick window mismatch at ordinal {ordinal}"
564:                             )
565:         try:
566:             next(expected)
567:         except StopIteration:
568:             pass
569:         else:
570:             raise ProvenanceRefused("migration prefix is missing prepared source rows")
571:     except ProvenanceRefused:
572:         raise
573:     except Exception as exc:
574:         raise ProvenanceRefused("Arrival target failed complete structural verification") from exc
575:     finally:
576:         for iterator in (expected, records):
577:             closer = getattr(iterator, "close", None)
578:             if callable(closer):
579:                 closer()
580:     if record_count <= migration_ordinal:
581:         raise ProvenanceRefused("Arrival target does not reach pinned migration head S")
582:     assert final_head is not None
583:     if historical_tick_index != source.ticks:
584:         raise ProvenanceRefused("migration prefix tick count differs from prepared source")
585:     if historical_explained != source.explained_windows:
586:         raise ProvenanceRefused("historical transformed-window evidence differs after migration")
587:     if historical_unchanged != source.unchanged_windows:
588:         raise ProvenanceRefused("historical unchanged-window evidence differs after migration")
589:     if _sha256(arrival) != before_hash:
590:         raise ProvenanceRefused("Arrival target changed during verification")
591:     return {
592:         "sha256": before_hash,
593:         "records": record_count,
594:         "captured_head": final_head,
595:         "facts": facts,
596:         "ticks": ticks,
597:         "protocol_metadata_records": metadata_count,
598:         "source_first_ordinal": metadata_count,
599:         "source_last_ordinal": migration_ordinal,
600:         "post_migration_ticks_verified": post_ticks,
601:     }

604: def verify_legacy_provenance(
605:     original: Path | str,
606:     prepared: Path | str,
607:     manifest: Path | str,
608:     arrival: Path | str,
609:     *,
610:     reviewed_source_sha256: str,
611:     reviewed_manifest_sha256: str,
612:     migration_lineage: str,
613:     migration_ordinal: int,
614:     migration_record_hash: str,
615: ) -> dict[str, Any]:
616:     """Return metadata-only evidence or refuse without changing an artifact."""
617:     source_hash = _hex(reviewed_source_sha256, "reviewed source hash")
618:     manifest_hash = _hex(reviewed_manifest_sha256, "reviewed manifest hash")
619:     migration_hash = _hex(migration_record_hash, "migration record hash")
620:     if not migration_lineage or type(migration_ordinal) is not int or migration_ordinal < 0:
621:         raise ProvenanceRefused("migration head S is malformed")
622:     original_path = _regular(original, "original source")
623:     prepared_path = _regular(prepared, "prepared source")
624:     manifest_path = _regular(manifest, "preparation manifest")
625:     arrival_path = _regular(arrival, "Arrival target")
626:     if len({original_path, prepared_path, manifest_path, arrival_path}) != 4:
627:         raise ProvenanceRefused("all four input artifacts must be distinct")
628: 
629:     manifest_value, actual_manifest_hash = _manifest(manifest_path, manifest_hash)
630:     source = _verify_sources(
631:         original_path, prepared_path, manifest_value, source_hash
632:     )
633:     arrival_evidence = _verify_arrival(
634:         arrival_path,
635:         prepared_path,
636:         source,
637:         lineage=migration_lineage,
638:         migration_ordinal=migration_ordinal,
639:         migration_hash=migration_hash,
640:     )
641:     if _sha256(manifest_path) != actual_manifest_hash:
642:         raise ProvenanceRefused("preparation manifest changed during verification")
643:     if _sha256(original_path) != source_hash:
644:         raise ProvenanceRefused("original source changed during verification")
645:     if _sha256(prepared_path) != manifest_value["output_sha256"]:
646:         raise ProvenanceRefused("prepared source changed during verification")
647:     return {
648:         "schema": SCHEMA,
649:         "status": "preserved-boundaries-with-audited-transformation",
650:         "trust": {
651:             "reviewed_source_sha256": source_hash,
652:             "reviewed_manifest_sha256": manifest_hash,
653:             "migration_head": {
654:                 "lineage": migration_lineage,
655:                 "ordinal": migration_ordinal,
656:                 "record_hash": migration_hash,
657:             },
658:             "pins": "caller-supplied",
659:             "pin_independence_verified": False,
660:             "migration_report_signature_verified": False,
661:         },
662:         "preparation": {
663:             "policy": POLICY,
664:             "source_sha256": source_hash,
665:             "prepared_sha256": manifest_value["output_sha256"],
666:             "rows": source.rows,
667:             "facts": source.facts,
668:             "ticks": source.ticks,
669:             "transformed_rows": source.changed,
670:             "byte_identical_rows": source.unchanged,
671:         },
672:         "commitments": {
673:             "original_tick_chain": "structurally-verified",
674:             "unchained_historical_ticks": source.unchained_ticks,
675:             "unchanged_historical_windows": source.unchanged_windows,
676:             "explained_transformation_windows": source.explained_windows,
677:             "historical_authorship_verified": False,
678:             "ordinary_envelope_signatures_verified": False,
679:             "inner_fact_tick_signatures_verified": False,
680:             "meaning": "old commitments bind the pinned original rows; mapped rows carry no historical authorship claim",
681:         },
682:         "arrival": {
683:             **arrival_evidence,
684:             "record_hash_chain": "verified",
685:             "registry_forming_signatures": "verified",
686:             "prepared_rows_at_pinned_prefix": "exact-in-order",
687:         },
688:     }

708: def _publish_output(path: Path, value: dict[str, Any], inputs: set[Path]) -> None:
709:     resolved = path.expanduser().resolve(strict=False)
710:     if resolved in inputs:
711:         raise ProvenanceRefused("output must not alias an input artifact")
712:     if resolved.exists() or resolved.is_symlink():
713:         raise ProvenanceRefused("output already exists and will not be overwritten")
714:     if not resolved.parent.is_dir():
715:         raise ProvenanceRefused("output parent directory does not exist")
716:     fd, raw = tempfile.mkstemp(prefix=f".{resolved.name}.", suffix=".stage", dir=resolved.parent)
717:     staged = Path(raw)
718:     try:
719:         with os.fdopen(fd, "wb") as stream:
720:             stream.write(json.dumps(value, indent=2, sort_keys=True).encode())
721:             stream.write(b"\n")
722:             stream.flush()
723:             os.fsync(stream.fileno())
724:         try:
725:             os.link(staged, resolved)
726:         except FileExistsError as exc:
727:             raise ProvenanceRefused("output appeared during verification") from exc
728:     finally:
729:         staged.unlink(missing_ok=True)

739:             args.arrival,
740:             reviewed_source_sha256=args.reviewed_source_sha256,
741:             reviewed_manifest_sha256=args.reviewed_manifest_sha256,
742:             migration_lineage=args.migration_lineage,
743:             migration_ordinal=args.migration_ordinal,
744:             migration_record_hash=args.migration_record_hash,
745:         )
746:         if args.output is None:
747:             print(json.dumps(result, sort_keys=True))
748:         else:
749:             inputs = {
750:                 Path(item).expanduser().resolve(strict=False)
751:                 for item in (args.original, args.prepared, args.manifest, args.arrival)
752:             }
753:             _publish_output(args.output, result, inputs)
754:     except ProvenanceError as exc:
755:         print(f"provenance verification refused: {exc}", file=sys.stderr)
756:         return 2
757:     # CLI errors must never echo malformed row content through a dependency's
758:     # exception. Library callers still receive the typed refusals above.

761:             "provenance verification refused: unexpected input or storage "
762:             f"failure ({type(exc).__name__})",
763:             file=sys.stderr,
764:         )
765:         return 2
766:     return 0

261: def attest_tick_commit(
262:     vertex: Path | str,
263:     *,
264:     first_ordinal: int,
265:     last_ordinal: int,
266:     public_key: str,
267:     expected_tick_id: str | None = None,
268:     committed_records: tuple[Mapping[str, Any], ...] | None = None,
269: ) -> dict[str, Any]:
270:     """Re-attest one physical tick in a bounded ordinary commit range.
271: 
272:     This helper reads only the supplied ordinal range and its immediate prior
273:     tick chain. It does not require a historical deep audit, so an operator
274:     can use it to supplement an already completed rehearsal artifact.
275:     """
276:     from custody.signing import TICK_DOMAIN
277:     from engine.arrival_registry import BackendRegistry, descriptor_for
278:     from engine.row_commitment import tick_commitment_hash, tick_row_hash
279:     from lang import parse_vertex_file
280: 
281:     if first_ordinal < 0 or last_ordinal < first_ordinal:
282:         raise RehearsalRefused("tick attestation range is invalid")
283:     target = Path(vertex).resolve()
284:     descriptor = descriptor_for(parse_vertex_file(target), target)
285:     if descriptor is None:
286:         raise RehearsalRefused("tick attestation target has no Arrival descriptor")
287:     ledger, query = BackendRegistry.with_builtin_backends().open(descriptor)
288:     try:
289:         # Arrival ``read`` verifies from the beginning on every call.  Scan
290:         # once through this bounded prefix instead, retaining only the one
291:         # predecessor needed for the tick-chain assertion.
292:         prior: dict[str, Any] | None = None
293:         ticks: list[dict[str, Any]] = []
294:         reached_last = False
295:         for record in ledger.scan():
296:             ordinal = record.get("ord")
297:             if not isinstance(ordinal, int):
298:                 raise RehearsalRefused("physical tick range has a malformed ordinal")
299:             if ordinal > last_ordinal:
300:                 break
301:             if record.get("k") == "tick":
302:                 if ordinal < first_ordinal:
303:                     prior = dict(record)
304:                 else:
305:                     ticks.append(dict(record))
306:             if ordinal == last_ordinal:
307:                 reached_last = True
308:                 break
309:     finally:
310:         _close(query)
311:         _close(ledger)
312:     if not reached_last:
313:         raise RehearsalRefused("tick attestation range is not a physical prefix")
314:     if len(ticks) != 1:
315:         raise RehearsalRefused("expected exactly one tick in the ordinary commit range")
316:     tick = ticks[0]
317:     if committed_records is not None:
318:         committed_ticks = tuple(
319:             record for record in committed_records if record.get("k") == "tick"
320:         )
321:         if len(committed_ticks) != 1 or committed_ticks[0] != tick:
322:             raise RehearsalRefused(
323:                 "physical tick differs from the committed ordinary records"
324:             )
325:     body = tick.get("body")
326:     if not isinstance(body, dict):
327:         raise RehearsalRefused("physical tick body is malformed")
328:     tick_id = body.get("id")
329:     if not isinstance(tick_id, str) or (expected_tick_id is not None and tick_id != expected_tick_id):
330:         raise RehearsalRefused("physical tick id differs from the committed receipt")
331:     fields = (
332:         "id", "name", "ts", "since", "origin", "payload", "prev_hash",
333:         "window_start", "fact_cursor", "window_hash",
334:     )
335:     if any(field not in body for field in fields):
336:         raise RehearsalRefused("physical tick lacks signed fields")
337:     signature = body.get("signature")
338:     if not isinstance(signature, str):
339:         raise RehearsalRefused("physical tick lacks its TICK signature")
340:     tick_row = tuple(body[field] for field in fields)
341:     if not _verify_signature(
342:         public_key, signature, tick_commitment_hash(tick_row), TICK_DOMAIN
343:     ):
344:         raise RehearsalRefused("physical tick TICK signature did not verify")
345:     expected_prev_hash = None
346:     prior_id = None
347:     if prior is not None:
348:         prior_body = prior.get("body")
349:         if not isinstance(prior_body, dict) or any(field not in prior_body for field in fields):
350:             raise RehearsalRefused("prior physical tick body is malformed")
351:         prior_signature = prior_body.get("signature")
352:         prior_row = tuple(prior_body[field] for field in fields) + (prior_signature,)
353:         expected_prev_hash = tick_row_hash(prior_row)
354:         prior_id = prior_body.get("id")
355:     if body["prev_hash"] != expected_prev_hash:
356:         raise RehearsalRefused("physical tick prev_hash does not name the latest prior tick")
357:     if tick.get("sig") is not None:
358:         raise RehearsalRefused("tick outer Arrival signature is not part of the current wire")
359:     return {
360:         "id": tick_id,
361:         "physical_matches_commit": None if committed_records is None else True,
362:         "tick_signature_verified": True,
363:         "outer_arrival_absent": True,
364:         "prev_hash_matches_prior_tick": True,
365:         "prior_tick_id": prior_id,
366:     }

1: """Pure commitments for persisted fact and tick rows.
2: 
3: These functions have no store or connection dependency. The SQLite writer,
4: Arrival runtime planner, and readers that verify chain evidence use the same
5: row bytes so a new write path cannot quietly fork an existing chain.
6: """
7: 
8: from __future__ import annotations
9: 
10: import hashlib
11: from collections.abc import Callable
12: 
13: from .admission import _canonical_bytes
14: 
15: __all__ = ["fact_row_hash", "tick_commitment_hash", "tick_envelope", "tick_row_hash"]
16: 
17: 
18: def tick_envelope(row: tuple) -> dict:
19:     """The ten tick fields that its inner signature commits to."""
20:     return {
21:         "id": row[0], "name": row[1], "ts": row[2], "since": row[3],
22:         "origin": row[4], "payload": row[5], "prev_hash": row[6],
23:         "window_start": row[7], "fact_cursor": row[8], "window_hash": row[9],
24:     }
25: 
26: 
27: def tick_commitment_hash(
28:     row: tuple, *, canonical_bytes: Callable[[object], bytes] = _canonical_bytes
29: ) -> str:
30:     """Hash one tick's signed content, excluding its signature field."""
31:     return hashlib.sha256(canonical_bytes(tick_envelope(row))).hexdigest()
32: 
33: 
34: def tick_row_hash(
35:     row: tuple, *, canonical_bytes: Callable[[object], bytes] = _canonical_bytes
36: ) -> str:
37:     """Hash one tick row as its successor's ``prev_hash`` value."""
38:     envelope = tick_envelope(row)
39:     if len(row) > 10 and row[10] is not None:
40:         envelope["signature"] = row[10]
41:     return hashlib.sha256(canonical_bytes(envelope)).hexdigest()
42: 
43: 
44: def fact_row_hash(
45:     row: tuple, *, canonical_bytes: Callable[[object], bytes] = _canonical_bytes
46: ) -> str:
47:     """Hash one fact row for a tick window, preserving exact payload text."""
48:     envelope = {
49:         "id": row[0], "kind": row[1], "ts": row[2],
50:         "observer": row[3], "origin": row[4], "payload": row[5],
51:     }
52:     if len(row) > 6 and row[6] is not None:
53:         envelope["signature"] = row[6]
54:     return hashlib.sha256(canonical_bytes(envelope)).hexdigest()

=== libs/engine/src/engine/arrival_projection.py ===
1: """arrival_projection — the two projections of an arrival log, re-derived.
2: 
3: Cut B of the arrival substrate (decision:design/arrival-sliceB-projections):
4: an arrival-canonical store has exactly two projections — the sqlite index at
5: the sibling ``.db`` and the derived ``.jsonl`` log beside it. Both are
6: functions of the arrival log, both are rebuildable at will, and neither is
7: ever written by anything that is not deriving it.
8: 
9: Why a module and not a store method
10: -----------------------------------
11: The states re-derivation resolves are exactly the states
12: :class:`~engine.arrival_store.ArrivalStore` refuses to open on: catch-up runs
13: inside the constructor, so a method on the class would be unreachable on the
14: very stores that need it. Making the constructor tolerate the refusal would
15: mean a store that opens in a broken state, which is what the consume-or-refuse
16: posture forbids. The precedent is :mod:`engine.canonical_audit`'s "pure
17: reader, by contract" — the module that must not repair never constructs the
18: store that does. Here it is the inverse and the same rule: the module that
19: repairs never constructs the store that refuses.
20: 
21: The second consequence is the reason it is worth a module: **no write path
22: can call re-derivation as a side effect.** It is not on the object the write
23: path holds.
24: 
25: What survives a re-derivation
26: -----------------------------
27: ``facts`` and ``ticks`` are cleared and rebuilt from ordinal 0 forward —
28: never partially, because reproducing the original rowids is what keeps
29: outstanding witness positions and seals valid, and that only holds for a
30: replay from empty. The FTS tables are DROPPED for the reason
31: :meth:`engine.jsonl_store.JsonlStore._rebuild` drops them: ``DELETE FROM
32: facts`` resets sqlite's rowid counter, so a surviving FTS index would resolve
33: stale text to new facts. Search then reports ``missing``, the honest "run
34: reindex" state. Every ``store_meta`` row survives except the two this module
35: owns as projections — the resume mark and ``own_lineage``, both restamped
36: from the log.
37: 
38: Locking
39: -------
40: Re-derivation takes **the sqlite write lock only** (``BEGIN IMMEDIATE`` for
41: the whole clear-and-replay), and never the arrival append lock. Taking the
42: sqlite lock serializes it against every ``ArrivalStore._write``, so no
43: appender can interleave an index write into a half-replayed index. NOT
44: taking the append lock means an appender may land a record mid-walk; that is
45: harmless, because the walk stops at the last complete record and the stamped
46: mark names what was actually consumed. The result is the ordinary "index
47: behind the log" state, which the next catch-up tails forward. Holding the
48: append lock for an O(n) replay would block every writer in the system to buy
49: a property the mark already provides.
50: """

112:     """Does this index carry any fact or tick row?

240: def derived_lines(log: ArrivalLog) -> list[str]:
241:     """Every derived-log line for an arrival log, byte-lexicographically sorted.
242: 
243:     **NON-NEGOTIABLE: line order carries no meaning, and no reader may
244:     attribute one to it.** Byte sort is the only order that is a pure
245:     function of the SET: it needs no field semantics, is reproducible by
246:     ``sort(1)`` in any language, and is what lets the git merge driver
247:     produce the same bytes a fresh derivation produces without knowing any
248:     ordinal.
249: 
250:     Two consequences, stated here rather than left to be found:
251: 
252:     1. **The derived log is not a chain-verification surface.** Tick chain
253:        order is not file order — a chain walk over a byte-sorted projection
254:        would report breaks on a healthy store. Chain verification reads the
255:        arrival log.
256:     2. **A last-0.x reader rebuilding from this file folds in byte order.**
257:        That is a real interim hazard for the migration sidecar, and it is
258:        named here for that design; the sidecar's own answer is to consume
259:        the ``.arrival``, which carries the ordinal.
260: 
261:     **NON-NEGOTIABLE: derived from the arrival log alone, never from the
262:     index.** Deriving it from sqlite would make it a projection of a
263:     projection, and a poisoned index would launder itself into a second
264:     artifact.
265:     """
266:     lines = set()
267:     for record in log.walk():
268:         line = line_of_record(record)
269:         if line is not None:
270:             lines.add(line)
271:     return sorted(lines, key=sort_key)

274: def write_derived_log(canonical: Path) -> int:
275:     """Materialize the derived ``.jsonl`` projection. Returns the line count.
276: 
277:     **NON-NEGOTIABLE: on demand only, never on the append path.** Appending
278:     to a byte-sorted file is not an append, and regenerating it per emit is
279:     O(n) per write — the O(n^2) ingest shape this arc already paid for once.
280:     A lagging derived log is NOT an error: nothing in ``libs/`` reads it,
281:     because reads execute through the index in every mode (arrival law 1).

283:     Written through a temp file and renamed, so a crash mid-derivation
284:     leaves the previous projection rather than a truncated one.
285:     """

292:         fh.write(payload)

300: class DerivedLogAgreement:
301:     """Whether the derived log and the arrival log hold the same set.
302: 
303:     There is **no staleness marker**. The offset/count triple does not get a
304:     third sibling: the derived log's currency is a SET question, and the
305:     honest answer to a set question is a set comparison, not a stamp. A
306:     marker would also reintroduce offset custody inside the mutable artifact
307:     being judged.

311:     missing: int
312:     """Row-class records the arrival log carries and the file does not.
313: 
314:     Structural records are outside the projection, so a healthy store is

322: def _derived_digests(canonical: Path) -> Counter[bytes]:
323:     """Digest every line the arrival log projects, retaining none of them.
324: 
325:     Deliberately NOT ``{_digest(line) for line in derived_lines(...)}``:
326:     :func:`derived_lines` must materialize every line because it SORTS them,
327:     and an audit that borrowed it inherited a cost it does not need. Here
328:     each line is produced, hashed, and counted in a multiset (:class:`Counter`),
329:     so what survives the walk is 32 bytes per unique digest.
330: 
331:     The multiset structure makes the audit order-insensitive while preserving
332:     multiplicity: duplicated lines in either artifact are caught as extra
333:     lines, rather than collapsing into an innocence-granting set union.
334: 
335:     The bound, stated exactly rather than loosely: peak is **flat in record
336:     count** and **linear in the size of the LARGEST single record**. What
337:     survives the walk is a 32-byte digest count per record; what is in flight is
338:     one record, which costs a small multiple of its own size because it is
339:     decoded, re-encoded and hashed. Measured: 5, 10, 20 and 40 records of
340:     1 MB each all peak at 7.02 MB, while one 4 MB record peaks at 28 MB.
341: 
342:     So "bounded by record count, never payload size" holds in the sense
343:     that matters — an audit over a big store does not grow with the store —
344:     but a single enormous record is still transiently expensive, and saying
345:     otherwise would be the kind of unbounded promise this module refuses to
346:     make elsewhere.
347:     """
348:     digests: Counter[bytes] = Counter()
349:     for record in ArrivalLog(canonical).walk():
350:         line = line_of_record(record)
351:         if line is not None:
352:             digests[_digest(line)] += 1
353:     return digests

356: def audit_derived_log(canonical: Path) -> DerivedLogAgreement:
357:     """Re-derive and diff. An operation, not a promise.
358: 
359:     The repo's established contract for derived artifacts (``verify_rebirth``
360:     is the precedent). Each side's lines are hashed to 32 bytes as they are
361:     produced and two multiset differences taken, so memory is bounded by RECORD
362:     COUNT and never by total payload size — see :func:`_derived_digests` for
363:     the exact bound.
364: 
365:     The reported counts survive digest-only multisets, because a multiset difference
366:     over digests has the same cardinality as one over the lines they stand
367:     for. Nothing in :class:`DerivedLogAgreement` echoes line CONTENT — the
368:     detail names paths and counts — so the result is bounded too.
369:     """
370:     canonical = Path(canonical)
371:     target = derived_log_path_for(canonical)
372:     derived = _derived_digests(canonical)
373: 
374:     if not target.exists():
375:         missing_count = derived.total()
376:         return DerivedLogAgreement(
377:             ok=not derived,
378:             missing=missing_count,
379:             extra=0,
380:             detail=(
381:                 f"no derived log at {target}"
382:                 + ("" if not derived else f"; the arrival log projects {missing_count} line(s)")
383:             ),
384:         )
385: 
386:     present: Counter[bytes] = Counter()
387:     with target.open("rb") as fh:
388:         for raw in fh:
389:             if raw.endswith(b"\n"):
390:                 present[hashlib.sha256(raw[:-1]).digest()] += 1
391:             elif raw:
392:                 # A torn tail is not a line: it was never terminated, so it
393:                 # never claimed to be a record. Counting it as `extra` would
394:                 # report a crashed derivation as a disagreement about
395:                 # content; the honest reading is that the file is short.
396:                 pass
397: 
398:     missing_counter = derived - present
399:     extra_counter = present - derived
400:     missing = missing_counter.total()
401:     extra = extra_counter.total()
402:     ok = missing == 0 and extra == 0
403:     return DerivedLogAgreement(
404:         ok=ok,
405:         missing=missing,
406:         extra=extra,
407:         detail=(
408:             f"{target} agrees with {canonical}"
409:             if ok
410:             else (
411:                 f"{target} disagrees with {canonical}: {missing} record(s) the "
412:                 f"arrival log carries are absent, {extra} line(s) it never "
413:                 "carried are present"
414:             )
415:         ),
416:     )

465: def rederive_projections(

467: ) -> Rederivation:
468:     """Discard an arrival store's projections and rebuild them from the log.
469: 
470:     The explicit operator verb. Building an ABSENT projection stays
471:     automatic (an absent projection destroys nothing when built); an
472:     EXISTING projection carrying rows is evidence, and discarding evidence
473:     is an operator's decision — so it is spelled here, off the object the
474:     write path holds.

477:     replay from empty reproduce the original assignment exactly, which is
478:     what keeps outstanding witness positions and seals valid across a
479:     re-derivation. A partial rebuild would renumber.
480: 
481:     Refuses (:class:`~engine.arrival_store.ArrivalCanonicalUnsupported`)
482:     when ``own_lineage`` is present and names a lineage other than the
483:     log's — that index is some other log's projection, and silently
484:     repointing it would be the strongest lie available in this design. The
485:     check runs BEFORE anything is cleared.
486:     """
487:     canonical = Path(canonical)
488:     log = ArrivalLog(canonical)

490:     index = index_path_for(canonical)
491: 
492:     conn = sqlite3.connect(str(index))
493:     conn.isolation_level = None  # explicit transaction control
494:     try:
495:         _ensure_index_schema(conn, log)
496:         conn.execute("BEGIN IMMEDIATE")
497:         try:
498:             marker = _meta_get(conn, _OWN_LINEAGE_KEY)
499:             if marker is not None and marker != lineage:
500:                 _unsupported(

502:                     f"but the log at {canonical} opens lineage {lineage!r} — "
503:                     "this index is some other log's projection; re-deriving "
504:                     "it would repoint an identity claim rather than rebuild "
505:                     "one"
506:                 )

508:             conn.execute("DELETE FROM facts")
509:             conn.execute("DELETE FROM ticks")
510:             conn.execute("DROP TABLE IF EXISTS facts_fts")
511:             conn.execute("DROP TABLE IF EXISTS fts_state")
512:             # Arrival exact-prefix FTS coverage is derived from these rows too.
513:             conn.execute("DROP TABLE IF EXISTS arrival_facts_fts")
514:             conn.execute("DROP TABLE IF EXISTS arrival_fts_state")
515: 
516:             records = facts = ticks = 0
517:             last: ResumeMark | None = None
518:             _, walked = log.walk_marked(None)

527:                         facts += 1

1: """arrival — the append-only log that holds canonical custody for a store.
2: 
3: Slice 0 of the arrival substrate (design fact
4: ``decision:design/arrival-slice0-record-grammar``; the wire format is written
5: out in ``docs/dev/arrival-slice0-record-grammar-2026-08-17.md``). One file per
6: store, ``<name>.arrival``, one JSON object per line. A record's position in
7: that file — its **ordinal** — is its identity coordinate, paired with the
8: **lineage** the genesis record opens. Everything derived from the log is a
9: projection and is never authoritative.
10: 
11: What slice 0 delivered: the record grammar, genesis minting, append under an
12: interprocess lock, read by ordinal, the walk with density and chain
13: verification, torn-tail truncation, corrupt-interior refusal, and the resume
14: mark. Cut A (decision:design/arrival-sliceA-authority) added the authority
15: half: the genesis carries the founding public key, self-signed at ordinal 0,
16: and :func:`verify_authorship` resolves every verifying key from a
17: ``(lineage, ordinal)`` coordinate in the log — never from any artifact
18: outside it. Projection re-derivation, declared projection orders, and audit
19: re-basing are later slices and are not here.
20: 
21: Two postures carry over from the existing store and are load-bearing:
22: 
23: * **Truncate only what was never complete, and only at the tail.** A final
24:   line with no ``\\n`` is an interrupted append; anything else that fails to
25:   decode, fails its ``rh``, breaks the ordinal succession, or breaks the
26:   ``prev`` link is corruption, and the answer is a human look rather than a
27:   silent delete — the posture ``ceremony.IntentCorrupt`` already takes. On a
28:   file that is the only copy, silent recovery is data loss with extra steps.

30:   reader cannot distinguish "torn" from "an append is in flight right now";
31:   it sees a complete newline-terminated record or it sees nothing yet. Only
32:   the appender, holding the lock and knowing no append is in progress, may

129: 
130: # Reverse scan granularity for finding a line boundary. Chunked so the scan
131: # never loads the whole log and never gives up as a function of line length.
132: _CHUNK = 64 * 1024
133: 
134: _HEX = frozenset("0123456789abcdef")
135: 
136: # A signer is injected, never imported: engine does not depend on ``sign``.
137: # Shape matches the store's ``fact_signer`` — (observer, commitment digest
138: # hex) -> signature, or None when that observer has no key.
139: Signer = Callable[[str, str], "str | None"]
140: 
141: # Verification is injected under the same posture. Shape: (public key in the
142: # ratified wire format, signature, commitment digest hex) -> bool. The
143: # composing layer supplies the algorithm and the domain-separation prefix;
144: # this module supplies WHICH key may speak at WHICH position, and nothing
145: # about how a signature is checked.
146: Verify = Callable[[str, str, str], bool]
147: 
148: # The ratified key wire format: an Ed25519 public key as raw-32-byte base64,
149: # copy-identical to the composing layer's on-disk public-key file. The
150: # grammar pins the shape (so a malformed key is refused at the append site);
151: # the algorithm stays the injected verifier's business.
152: _KEY_RAW_LEN = 32

258: class ForkedHeight(ArrivalError):

273: def _refuse_stale_head(actual: dict, expected: Head | None) -> None:
274:     """Refuse unless ``actual`` is the head the caller pinned.
275: 
276:     ALL THREE coordinate fields, which is what makes this a compare-and-swap
277:     over the log's identity rather than over its length. The ordinal alone
278:     cannot tell "nothing arrived" from "the record at this ordinal was
279:     replaced": a log truncated back and rewritten to the same height passes an
280:     ordinal compare and fails this one, and that case is precisely the
281:     rollback the backend contract asks a store to detect (§11).
282: 
283:     Called from inside the fence at both append sites, never from outside it —
284:     a compare that ran before the lock would be answering about a head some
285:     other writer is free to move before the write lands. The two call sites
286:     stay separate calls on purpose: each is independently the thing that makes
287:     its own method a compare-and-swap, and each has its own test.
288:     """
289:     if expected is None:
290:         return
291:     if (
292:         actual["lin"] == expected.lineage
293:         and actual["ord"] == expected.ordinal
294:         and actual[_RH] == expected.record_hash
295:     ):
296:         return
297:     raise StaleHead(
298:         f"the log's head is ({actual['lin']}, {actual['ord']}, {actual[_RH]}), "
299:         f"not the expected ({expected.lineage}, {expected.ordinal}, "
300:         f"{expected.record_hash}) — records arrived since the caller "
301:         "reconciled, or this coordinate now holds a different record"
302:     )

305: def _checked_pin(following: object) -> Head | None:
306:     """``following`` as a head pin, refusing anything weaker.
307: 
308:     An ordinal is REFUSED rather than widened into a head, because widening
309:     it here would mean this function inventing the two fields the caller did
310:     not supply — and a pin the callee completed is not a pin. Construction
311:     over detection (``decision:practice/construction-vs-detection-ratchets``):
312:     there is no spelling of a weaker compare-and-swap for a caller to reach
313:     for, so none can be reached for by accident.
314: 
315:     :class:`ArrivalError` and deliberately not :class:`AppendRejected`: this
316:     is a caller bug, and ``store.merge_store`` retries on
317:     :class:`AppendRejected` — a type error caught by that loop would spin
318:     instead of surfacing.
319:     """
320:     if following is None or isinstance(following, Head):
321:         return following
322:     if _is_int(following):
323:         raise ArrivalError(
324:             f"following must be a Head, not the bare ordinal {following!r} — "
325:             "an ordinal-only pin cannot tell an untouched log from one "
326:             "truncated and rewritten to the same height. Pass the full head "
327:             "(engine.arrival_contract.Head), or None for an unpinned append"
328:         )
329:     raise ArrivalError(
330:         f"following must be a Head or None, got {type(following).__name__}"
331:     )

381: def _canonical_bytes(obj: dict) -> bytes:
382:     """Canonical encoding for hashing: JCS, RFC 8785.
383: 
384:     The same canonicalization every commitment in this codebase hashes
385:     through. It is a total function on the JSON value, so any conforming
386:     implementation in any language derives the same bytes — which is what a
387:     cross-language conformance oracle needs and why JCS was adopted.
388:     """
389:     return rfc8785.dumps(obj)

392: def _bad(message: str) -> NoReturn:
393:     """Refuse a record. ``NoReturn`` is load-bearing, not decoration: the
394:     validators below call this instead of raising inline, and without it a
395:     reader — human or type checker — cannot see that the checks are total,
396:     so every subsequent use of the value looks conditional."""
397:     raise ArrivalGrammarError(message)

400: def _is_int(value: object) -> TypeGuard[int]:
401:     """A JSON integer. ``bool`` is excluded: ``True`` is an ``int`` in Python
402:     and would silently pass an ordinal check while encoding as ``true``."""
403:     return isinstance(value, int) and not isinstance(value, bool)

406: def _check_hex(value: object, field: str) -> None:
407:     if not isinstance(value, str) or len(value) != 64 or not set(value) <= _HEX:
408:         _bad(f"{field} must be a 64-char lowercase sha256 hex digest, got {value!r}")

410: 
411: def _check_jcs_number(value: object, field: str) -> None:
412:     """Hold a number to what JCS can canonicalize.
413: 
414:     Integers outside the 2^53 domain and non-finite floats have no canonical
415:     form, so a record carrying one could never be hashed.
416:     """
417:     if _is_int(value):
418:         if not _JCS_INT_MIN <= value <= _JCS_INT_MAX:
419:             _bad(f"{field} {value} is outside the JCS integer domain")
420:         return
421:     if isinstance(value, float):
422:         if not math.isfinite(value):
423:             _bad(f"{field} must be a finite number, got {value!r}")
424:         return
425:     _bad(f"{field} must be a number, got {type(value).__name__}")
426:

428: def _check_body(body: object) -> None:
429:     """``body`` is a JSON object, and one JCS can canonicalize.
430: 
431:     A named departure from the line codec, which carries payload as verbatim
432:     TEXT so that two custody holders cannot disagree byte-wise. In the
433:     arrival model there is no second copy — the log IS the store — so there
434:     is no other byte-form to disagree with, and JCS defines the canonical
435:     bytes. Keeping the body an opaque string would double every escape,
436:     make the file unreadable by eye, and leave the canonical bytes defined
437:     by whichever encoder happened to produce the string.
438:     """
439:     if not isinstance(body, dict):
440:         _bad(f"body must be a JSON object, got {type(body).__name__}")
441:     try:
442:         _canonical_bytes(body)
443:     except Exception as exc:  # rfc8785 raises its own family
444:         _bad(f"body is not canonicalizable under JCS: {exc}")

447: def _validate(record: object, *, require_rh: bool) -> None:
448:     """The one grammar validator, run in both directions.
449: 
450:     ``require_rh`` is False while building a record (``rh`` is computed from
451:     the validated object) and True for anything read back off disk.
452: 
453:     Takes ``object``, not ``dict``, and the annotation is doing real work:
454:     :func:`decode_record` hands this whatever ``json.loads`` produced, which
455:     may be a list or a string, and the first check below is what refuses
456:     those. Narrowing the signature to ``dict`` would make that check look
457:     dead to a reader and to a type checker while it stayed live at runtime.
458:     """
459:     if not isinstance(record, dict):
460:         _bad(f"a record must be a JSON object, got {type(record).__name__}")
461: 
462:     unknown = sorted(set(record) - _ALLOWED)
463:     if unknown:
464:         _bad(f"unknown record field(s): {', '.join(unknown)}")
465:     missing = [f for f in RECORD_FIELDS if f not in record]
466:     if missing:
467:         _bad(f"missing record field(s): {', '.join(missing)}")
468:     if require_rh and _RH not in record:
469:         _bad("missing record field: rh")
470: 
471:     if not _is_int(record["v"]) or record["v"] != GRAMMAR_VERSION:
472:         _bad(f"unsupported grammar version {record['v']!r} (this build writes {GRAMMAR_VERSION})")
473: 
474:     if not isinstance(record["lin"], str) or not record["lin"]:
475:         _bad("lin must be a non-empty string")
476: 
477:     if not _is_int(record["ord"]) or record["ord"] < 0:
478:         _bad(f"ord must be an integer >= 0, got {record['ord']!r}")
479:     _check_jcs_number(record["ord"], "ord")
480: 
481:     # prev is null exactly at ordinal 0 and a digest everywhere else. Both
482:     # directions are checked: a genesis carrying a prev is as wrong as a
483:     # successor carrying none.
484:     if record["ord"] == 0:
485:         if record["prev"] is not None:
486:             _bad("prev must be null at ordinal 0")
487:     else:
488:         _check_hex(record["prev"], "prev")
489: 
490:     _check_jcs_number(record["at"], "at")
491: 
492:     if not isinstance(record["k"], str) or not record["k"]:
493:         _bad("k must be a non-empty string")
494:     if not isinstance(record["observer"], str) or not record["observer"]:
495:         _bad("observer must be a non-empty string")
496:     if not isinstance(record["origin"], str):
497:         _bad(f"origin must be a string (may be empty), got {type(record['origin']).__name__}")
498: 
499:     _check_body(record["body"])
500: 
501:     # Absent, never null: a record with no signature and a record with
502:     # ``"sig":null`` must not both be spellable, or they hash differently
503:     # while meaning the same thing.
504:     if _SIG in record and (not isinstance(record[_SIG], str) or not record[_SIG]):
505:         _bad("sig, when present, must be a non-empty string")
506: 
507:     if _RH in record:
508:         _check_hex(record[_RH], "rh")
509:

511: def _key_shape_fault(value: object) -> str | None:
512:     """Why ``value`` is not a well-formed public key, or None when it is.
513: 
514:     Shape only — raw-32-byte base64, the ratified wire format. Whether a
515:     signature actually verifies under it is the injected verifier's answer
516:     (:func:`verify_authorship`); the grammar refuses what could never be a
517:     key so a malformed one fails at the append site rather than bricking
518:     verification later.
519:     """
520:     if not isinstance(value, str) or not value:
521:         return f"key must be a non-empty string, got {type(value).__name__}"
522:     try:
523:         raw = base64.b64decode(value, validate=True)
524:     except (binascii.Error, ValueError) as exc:
525:         return f"key is not valid base64: {exc}"
526:     if len(raw) != _KEY_RAW_LEN:
527:         return f"key must decode to {_KEY_RAW_LEN} bytes, got {len(raw)}"
528:     return None

531: def _placement_fault(record: dict, ordinal: int) -> str | None:
532:     """Why ``record`` may not sit at ``ordinal``, or None when it may.
533: 
534:     One rule stated once, in both directions: **the genesis kind belongs at
535:     ordinal 0 and nowhere else.** At 0 a record must satisfy every genesis
536:     rule; above 0 it must not claim to be a genesis at all. A key
537:     introduction (:data:`KEY_INTRODUCTION_KIND`) is held to its structural
538:     rules at any ordinal it appears: signed, and a body naming an observer
539:     and a well-formed key.
540: 
541:     Structural rules only, deliberately: whether the genesis signature
542:     verifies against ``body["key"]`` — self-certification — and whether an
543:     introduction's signer was valid at its position are cryptographic
544:     questions, answered by :func:`verify_authorship` with an injected
545:     verifier. This function stays pure so every caller (walk, append, open)
546:     can run it without carrying one.
547: 
548:     This exists as one function because the two directions were previously
549:     checked in four places with three different sets of rules, and the gaps
550:     between them were exactly the holes: a forged unsigned genesis walked
551:     clean because only :meth:`ArrivalLog.genesis` demanded a signature, and
552:     a second genesis could be appended at an interior ordinal because
553:     nothing checked the kind on the way in. Callers differ only in the error
554:     class they raise — opening a file that is not an arrival log, walking one
555:     that has been tampered with, and refusing an append are three different
556:     conversations about the same rule.
557:     """
558:     if ordinal == 0:
559:         if record["k"] != GENESIS_KIND:
560:             return f"first record has kind {record['k']!r}, not a genesis"
561:         if record["ord"] != 0:
562:             return f"first record carries ord {record['ord']!r}, not 0"
563:         if record["prev"] is not None:
564:             return "first record carries a prev — genesis opens the chain"
565:         if _SIG not in record:
566:             return (
567:                 "genesis carries no signature — genesis is the lineage's "
568:                 "attestation root"
569:             )
570:         claimed = record["body"].get("lineage")
571:         if claimed != record["lin"]:
572:             return (
573:                 f"genesis is not self-naming — body claims lineage {claimed!r} "
574:                 f"but the coordinate says {record['lin']!r}"
575:             )
576:         key_fault = _key_shape_fault(record["body"].get("key"))
577:         if key_fault is not None:
578:             return (
579:                 f"genesis carries no well-formed founding key ({key_fault}) — "
580:                 "every genesis introduces the founding public key at "
581:                 "ordinal 0; the core grammar admits no keyless genesis"
582:             )
583:         return None
584:     if record["k"] == GENESIS_KIND:
585:         return (
586:             f"a genesis at ordinal {ordinal} — the genesis kind belongs at "
587:             "ordinal 0 and nowhere else, and a lineage is opened once"
588:         )
589:     if record["k"] == KEY_INTRODUCTION_KIND:
590:         if _SIG not in record:
591:             return (
592:                 "a key introduction carries no signature — an introduction "
593:                 "is vouched for by a key that is already valid, never "
594:                 "self-certifying above ordinal 0"
595:             )
596:         named = record["body"].get("observer")
597:         if not isinstance(named, str) or not named:
598:             return "a key introduction's body must name the observer the key speaks for"
599:         key_fault = _key_shape_fault(record["body"].get("key"))
600:         if key_fault is not None:
601:             return f"a key introduction's body carries no well-formed key: {key_fault}"
602:     return None
603:

605: def _authority_fault(record: dict, lineage: str) -> str | None:
606:     """Why ``record`` may not be ADOPTED AS AN AUTHORITY, or None when it may.
607: 
608:     An authority is an existing on-disk record the code takes as a starting
609:     point and derives from: the head an append chains onto, the anchor a
610:     resume walk continues from. Reading a record forward and adopting one as
611:     a premise are different acts, and only the first was ever checked —
612:     twice, review found the same residual shape, because a rule enforced on
613:     the way past a record says nothing about a record you never walked to.
614: 
615:     So: validate at every adoption site, not at a proxy for it. A record may
616:     be adopted when it decodes and its ``rh`` recomputes (the caller's
617:     :func:`decode_record` established that), when it may sit at the ordinal
618:     it claims (:func:`_placement_fault`), and when it belongs to this log's
619:     lineage.
620: 
621:     What adoption does NOT establish, and deliberately: that the record
622:     chains back to genesis, or that its claimed ordinal matches its physical
623:     position in the file. Both need a walk of the whole prefix, which is the
624:     O(n) cost these O(1) paths exist to avoid. The residual is narrow —
625:     density is re-checked from the adopted record onward, so a hostile
626:     coordinate is refused one record later — and :meth:`ArrivalLog.walk` is
627:     the integrity statement whenever the whole file is the question.
628:     """
629:     fault = _placement_fault(record, record["ord"])
630:     if fault is not None:
631:         return fault
632:     if record["lin"] != lineage:
633:         return (
634:             f"record claims lineage {record['lin']!r}, "
635:             f"but this log's lineage is {lineage!r}"
636:         )
637:     return None

640: def content_commitment(
641:     k: str, at: float, observer: str, origin: str, body: dict
642: ) -> str:
643:     """The digest a signature covers: sha256 over JCS of the five content
644:     fields.
645: 
646:     Content only. The coordinate, ``prev``, ``rh``, and ``sig`` itself are
647:     all outside it — the first three because they are custody context rather
648:     than authored content, the last because a signature cannot sign itself.
649:     That exclusion is what makes a signature transport-stable: any store
650:     holding the record re-derives the same digest and verifies authorship
651:     against the observer registry without trusting the sender.
652:     """
653:     envelope = {"k": k, "at": at, "observer": observer, "origin": origin, "body": body}
654:     return hashlib.sha256(_canonical_bytes(envelope)).hexdigest()
655:

657: def record_hash(record: dict) -> str:
658:     """``rh``: sha256 over JCS of the whole record with ``rh`` removed.
659: 
660:     Covers the coordinate and the signature, which is how the coordinate is
661:     tamper-evident without being signed, and how ``prev`` gets something to
662:     point at.
663:     """
664:     return hashlib.sha256(
665:         _canonical_bytes({key: value for key, value in record.items() if key != _RH})
666:     ).hexdigest()

669: def _no_duplicate_keys(pairs: list[tuple[str, object]]) -> dict:
670:     """JSON's last-wins duplicate resolution would let one line carry two
671:     ordinals and a reader silently pick one. Reject instead."""
672:     seen: dict = {}
673:     for key, value in pairs:
674:         if key in seen:
675:             _bad(f"duplicate key {key!r} in one JSON object")

686:     body: dict,
687:     observer: str,
688:     origin: str = "",
689:     at: float | None = None,
690:     sig: str | None = None,
691: ) -> dict:
692:     """Assemble one validated record, ``rh`` computed and last.
693: 
694:     ``at`` is wall clock at append. It is authored content and rides inside
695:     the signed envelope — an unsigned timestamp is a claim anyone downstream
696:     can rewrite. It is never an ordering input: nothing sorts by it and no
697:     later slice's declared orders may name it.
698:     """
699:     record: dict = {
700:         "v": GRAMMAR_VERSION,
701:         "lin": lin,
702:         "ord": ordinal,
703:         "prev": prev,
704:         "at": time.time() if at is None else at,
705:         "k": k,
706:         "observer": observer,
707:         "origin": origin,
708:         "body": body,
709:     }
710:     if sig is not None:
711:         record[_SIG] = sig
712:     _validate(record, require_rh=False)
713:     record[_RH] = record_hash(record)
714:     return record
715: 
716: 
717: def _completed_candidate(record: dict) -> dict:
718:     """A carried-in record, ready to be validated against a head.
719: 
720:     :func:`build_record`'s twin on the other side of the grammar: that one
721:     ASSIGNS a coordinate, this one accepts the coordinate a record already
722:     carries. ``rh`` is computed only when it is absent; supplied, it is left
723:     exactly as it arrived so :func:`encode_record` can CHECK it rather than
724:     this function replacing it with one that trivially matches. That is where
725:     "append validates, it never assigns" lives, so it has one spelling shared
726:     by every carried-in path — a second copy is how one path quietly starts
727:     trusting a digest the other checks.
728: 
729:     A shallow copy, because the caller's dict is not this module's to mutate.
730:     """
731:     candidate = dict(record)
732:     if _RH not in candidate:
733:         _validate(candidate, require_rh=False)
734:         candidate[_RH] = record_hash(candidate)
735:     return candidate
736: 
737: 
738: def encode_record(record: dict) -> str:
739:     """One line, no trailing newline.
740: 
741:     ``ensure_ascii`` keeps the line 7-bit and free of raw U+2028/U+2029,
742:     ``allow_nan=False`` refuses the non-JSON float constants, and
743:     ``separators`` drops insignificant whitespace. Together they are why
744:     newline framing is safe rather than merely conventional: no encoded
745:     record can contain a raw newline, so a ``\\n`` in the file is
746:     unambiguously a boundary.
747:     """
748:     _validate(record, require_rh=True)
749:     if record[_RH] != record_hash(record):
750:         _bad("rh does not match the record it is attached to")
751:     ordered = {key: record[key] for key in RECORD_FIELDS}
752:     if _SIG in record:
753:         ordered[_SIG] = record[_SIG]
754:     ordered[_RH] = record[_RH]
755:     return json.dumps(ordered, ensure_ascii=True, allow_nan=False, separators=(",", ":"))
756: 
757: 
758: def decode_record(line: str | bytes) -> dict:
759:     """Decode one line and hold it to the grammar, ``rh`` included.
760: 
761:     Raises :class:`ArrivalGrammarError`. The caller decides what a failure
762:     means — the walk turns it into :class:`ArrivalCorrupt` with a coordinate
763:     attached, because at that point there is a position to name.
764:     """
765:     if isinstance(line, bytes):
766:         try:
767:             line = line.decode("utf-8")
768:         except UnicodeError as exc:
769:             _bad(f"line is not valid UTF-8: {exc}")
770:     try:
771:         record = json.loads(line, object_pairs_hook=_no_duplicate_keys)
772:     except ValueError as exc:
773:         if isinstance(exc, ArrivalGrammarError):
774:             raise
775:         _bad(f"line is not valid JSON: {exc}")
776:     _validate(record, require_rh=True)
777:     if record[_RH] != record_hash(record):
778:         _bad("rh does not recompute — the record's bytes have changed")
779:     return record

842: 
843: class ArrivalLog:
844:     """One arrival log, addressed by its path.
845: 
846:     Cheap to construct and holds no open file: every operation opens what it
847:     needs. Appends across processes are mutually exclusive, enforced by an
848:     advisory ``LOCK_EX`` on ``<name>.arrival.lock`` held for the whole
849:     read-head, validate, write, fsync sequence. Reads take no lock and are
850:     safe at any time against a concurrent append.
851: 
852:     Exclusion rather than lock-free ``O_APPEND`` is forced by the addressing
853:     scheme, not chosen for convenience: the ordinal follows from the head, so
854:     correctness requires reading the head and writing to be indivisible. Two
855:     processes that both read head ``ord=7`` would both write ``ord=8``, and a
856:     duplicate coordinate is a corrupt interior that refuses forever.
857: 
858:     ``flock`` specifically because it is released when the holding process
859:     dies. A crashed writer must not wedge the store, which is the opposite of
860:     what an ``O_EXCL`` lock file would give — that behaviour is right for the
861:     one-shot genesis gate and wrong for an operation that runs constantly.
862:     """
863: 
864:     def __init__(self, path: Path | str) -> None:
865:         self.path = Path(path)
866:         self.lock_path = lock_path_for(self.path)
867:         self.tmp_path = tmp_path_for(self.path)
868:         # Memo for :meth:`_genesis_from`, keyed on the genesis line's bytes.
869:         self._genesis_line: bytes | None = None
870:         self._genesis_record: dict | None = None

872:     # -- minting -------------------------------------------------------
873: 
874:     @classmethod
875:     def mint(
876:         cls,
877:         path: Path | str,
878:         *,
879:         observer: str,
880:         signer: Signer,
881:         key: str,
882:         origin: str = "",
883:         lineage: str | None = None,
884:         at: float | None = None,
885:     ) -> ArrivalLog:
886:         """Open a new lineage: write genesis and publish it atomically.
887: 
888:         The genesis record is self-naming — the lineage id it carries is the
889:         one its own coordinate ``(lin, 0)`` opens — and it must be signed:
890:         genesis is the lineage's attestation root, and the signature is over
891:         the ACTUAL final body, so the body is assembled first and signed
892:         second. The body is ``{protocol, lineage, key}`` and nothing else
893:         (decision:design/arrival-sliceA-authority §2.3): ``key`` is the
894:         founding PUBLIC key in the ratified wire format, self-signed at
895:         ordinal 0 — the anchor every later authorship answer resolves back
896:         to. The document set is movement 2, records at ordinal >= 1; era
897:         pins dissolve under the dense ordinal; source fingerprints and
898:         equivalence claims live in the signed migration report and bootstrap
899:         receipt, never in a containment-only genesis.
900: 
901:         ``key`` and ``signer`` must correspond — the grammar checks the
902:         key's shape here, and :func:`verify_authorship` is where the
903:         self-certification is cryptographically established. A mint whose
904:         signer does not hold ``key`` produces a log the verifier refuses at
905:         ordinal 0.
906: 
907:         Publishing — the ``O_EXCL`` staging mutex, the existence check held
908:         under it, the fsync of both file and directory — is
909:         :meth:`_publish_genesis`, shared with :meth:`adopt_genesis`.
910:         """
911:         log = cls(path)
912:         log.path.parent.mkdir(parents=True, exist_ok=True)
913: 
914:         key_fault = _key_shape_fault(key)
915:         if key_fault is not None:
916:             raise GenesisRefused(
917:                 f"genesis founding key is malformed ({key_fault}) — the "
918:                 "ratified wire format is a raw-32-byte base64 public key"
919:             )
920:         lin = mint_lineage() if lineage is None else lineage
921:         body = {"protocol": GRAMMAR_VERSION, "lineage": lin, "key": key}
922:         at = time.time() if at is None else at
923:         sig = signer(observer, content_commitment(GENESIS_KIND, at, observer, origin, body))
924:         if not sig:
925:             raise GenesisRefused(
926:                 f"genesis for lineage {lin} is unsigned — genesis is the lineage's "
927:                 f"attestation root and observer {observer!r} produced no signature"
928:             )
929:         line = encode_record(
930:             build_record(
931:                 lin=lin, ordinal=0, prev=None, k=GENESIS_KIND, body=body,
932:                 observer=observer, origin=origin, at=at, sig=sig,
933:             )
934:         )
935: 
936:         log._publish_genesis(line)
937:         return log

939:     @classmethod
940:     def adopt_genesis(cls, path: Path | str, record: dict) -> ArrivalLog:
941:         """Open a lineage on a genesis record that was minted ELSEWHERE.
942: 
943:         :meth:`mint`'s twin on the carried-in side, the way
944:         :meth:`append_record` is :meth:`append`'s: mint builds and signs a
945:         genesis, this one accepts the exact bytes of one that already exists.
946:         Nothing is re-signed and nothing is rehashed — a replica whose ordinal
947:         0 differed from the authority's by a single byte would be a different
948:         lineage wearing the same id, so the record lands exactly as offered or
949:         not at all.
950: 
951:         The record is held to the same two gates a walk holds ordinal 0 to:
952:         :func:`encode_record` (grammar, and the supplied ``rh`` CHECKED rather
953:         than trusted) and :func:`_placement_fault` at ordinal 0 (genesis kind,
954:         no ``prev``, signed, self-naming, well-formed founding key). Structural
955:         gates only — whether the genesis signature verifies against its own
956:         founding key is :func:`verify_authorship`'s question, asked with an
957:         injected verifier, and this path stays as pure as :meth:`walk` is
958:         about the same record.
959: 
960:         Publishing is :meth:`mint`'s ceremony exactly, and shared with it:
961:         staging wins ``O_EXCL`` first, so an existing log at this path refuses
962:         with the mutex held rather than being overwritten.
963:         """
964:         log = cls(path)
965:         log.path.parent.mkdir(parents=True, exist_ok=True)
966:         candidate = _completed_candidate(record)
967:         line = encode_record(candidate)
968:         fault = _placement_fault(candidate, 0)
969:         if fault is not None:
970:             raise GenesisRefused(f"cannot open a lineage on this record: {fault}")
971:         log._publish_genesis(line)
972:         return log
973: 
974:     def _publish_genesis(self, line: str) -> None:
975:         """Stage, write, fsync and publish one genesis line — atomically.
976: 
977:         The one home for the ceremony :meth:`mint` and :meth:`adopt_genesis`
978:         share. Staging wins ``O_EXCL`` on ``<name>.arrival.tmp`` FIRST and
979:         only then checks for an existing log. That order is what makes opening
980:         a lineage a race safe: the tmp path is the mutex, so the existence
981:         check happens with the mutex held. Checking before creating would
982:         leave a window where both processes see no log and the second one's
983:         rename destroys the first one's store.
984: 
985:         The directory is fsync'd once, here. The file's own fsync is not
986:         enough at creation: an fsync'd file whose directory entry never
987:         reached disk is a store that vanishes on power loss, and there is no
988:         second copy to export from again. Per-append directory fsync is not
989:         needed — the entry already exists and only the file's size and content
990:         change after this.
991:         """
992:         try:
993:             fd = os.open(str(self.tmp_path), os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o644)
994:         except FileExistsError as exc:
995:             raise GenesisRefused(
996:                 f"{self.tmp_path} exists — another process is opening this lineage, "
997:                 "or a previous attempt died mid-flight and left staging behind"
998:             ) from exc
999:         try:
1000:             with os.fdopen(fd, "w", encoding="utf-8") as fh:
1001:                 # The existence check runs with the staging mutex held, which
1002:                 # is what makes it authoritative rather than a fast path.
1003:                 if self.path.exists():
1004:                     raise GenesisRefused(
1005:                         f"{self.path} already exists — a lineage is opened once, and "
1006:                         "publishing over an existing log would destroy a store"
1007:                     )
1008:                 fh.write(line + "\n")
1009:                 fh.flush()
1010:                 os.fsync(fh.fileno())
1011:             self.tmp_path.replace(self.path)
1012:         except BaseException:
1013:             # The staging create won, but the bytes never became durable —
1014:             # leave zero residue behind the failed attempt.
1015:             with contextlib.suppress(OSError):
1016:                 self.tmp_path.unlink()
1017:             raise
1018:         _fsync_dir(self.path.parent)
1019:

1037:     def genesis(self) -> dict:
1038:         """The record at ordinal 0, validated as a genesis.
1039: 
1040:         A file whose first record is not a valid genesis is not an arrival
1041:         log and opening it refuses. There is no inference of identity from
1042:         content: the singleton heuristic that guessed a store's lineage from
1043:         its rows was a hijack vector, and here the hazard dissolves rather
1044:         than being defended against, because position 0 is structural.
1045: 
1046:         Routes through the content-keyed memo (:meth:`_genesis_from`), so a
1047:         ceremony that asks for the genesis twice pays for one validation.
1048:         The returned dict is the memoized record — treat it as immutable,
1049:         the way every record this module yields is.
1050:         """
1051:         try:
1052:             with self.path.open("rb") as fh:
1053:                 return self._genesis_from(fh)
1054:         except FileNotFoundError as exc:
1055:             raise GenesisRefused(f"{self.path} does not exist") from exc

1057:     def _genesis_from(self, fh) -> dict:
1058:         """The validated genesis record, memoized on the genesis LINE BYTES.
1059: 
1060:         The append path needs the lineage on every write, and re-deriving it
1061:         costs a JSON decode, an ``rh`` recompute and the genesis checks —
1062:         about a tenth of an append, which is worth not paying twice for the
1063:         same bytes.
1064: 
1065:         **A cache hit is not a skipped validation.** The key is the exact
1066:         line the validated answer was derived from, so a hit means "these
1067:         bytes already passed, and validating them again is a pure function
1068:         returning the same result". That is the only cache shape allowed
1069:         here: the standing lesson of this module is that an adopted
1070:         authority gets validated at the adoption site, and a cache keyed on
1071:         anything weaker than the content — a path, an inode, an mtime —
1072:         would be a way to skip that. Reading the line to compare it costs
1073:         about 0.1 µs against the 15 µs it saves, so the strong key is also
1074:         the cheap one.
1075: 
1076:         Staleness therefore cannot bite: the instance may outlive a
1077:         truncation, a re-mint at the same path, or a foreign edit, and every
1078:         one of those changes the bytes, misses, and revalidates. A
1079:         replacement whose genesis is byte-identical opens the same lineage,
1080:         so the cached answer is still the right one.
1081:         """
1082:         fh.seek(0)
1083:         raw = fh.readline()
1084:         # Snapshot both fields before comparing: a torn read across threads
1085:         # then misses and revalidates rather than pairing one entry's key
1086:         # with another's value.
1087:         cached_line, cached_record = self._genesis_line, self._genesis_record
1088:         if raw == cached_line and cached_record is not None:
1089:             return cached_record
1090:         record = self._validated_genesis(raw)
1091:         self._genesis_record = record
1092:         self._genesis_line = raw  # published last, so the key implies the value
1093:         return record
1094: 
1095:     def _lineage_from(self, fh) -> str:
1096:         """The validated lineage — the memo's hot-path projection."""
1097:         return self._genesis_from(fh)["lin"]

1099:     def _validated_genesis(self, raw: bytes) -> dict:
1100:         """Decode one raw line and hold it to every genesis rule."""
1101:         if not raw.endswith(b"\n"):
1102:             raise GenesisRefused(
1103:                 f"{self.path} holds no complete first record — not an arrival log"
1104:             )
1105:         try:
1106:             record = decode_record(raw[:-1])
1107:         except ArrivalGrammarError as exc:
1108:             raise GenesisRefused(f"{self.path}: first record is not readable: {exc}") from exc
1109:         fault = _placement_fault(record, 0)
1110:         if fault is not None:
1111:             raise GenesisRefused(f"{self.path}: {fault}")
1112:         return record

1114:     def lineage(self) -> str:
1115:         """The lineage this log's genesis opens."""
1116:         return self.genesis()["lin"]

1118:     def walk(self) -> Iterator[dict]:
1119:         """Every complete record in order, fully verified.
1120: 
1121:         Verifies, per record: that it decodes, that its ``rh`` recomputes
1122:         (inside :func:`decode_record`), that it may sit where it sits — the
1123:         full genesis rules at ordinal 0 and no genesis kind above it, via
1124:         :func:`_placement_fault` — that its lineage matches the genesis's,
1125:         that its ordinal is the next one — dense and ascending from 0, so a
1126:         gap or a repeat is caught — and that its ``prev`` names its
1127:         predecessor's ``rh``. Any failure raises :class:`ArrivalCorrupt`
1128:         naming the ordinal. Nothing is skipped and nothing is truncated.
1129: 
1130:         The genesis rules are enforced HERE and not only in
1131:         :meth:`genesis`, because a walk that trusted position 0 without
1132:         judging it would read a forged unsigned genesis clean — and
1133:         :meth:`read` and :meth:`head` both reach their answer through this
1134:         walk, so the check has to live on the path they share.
1135: 
1136:         A trailing line with no ``\\n`` ends the walk without complaint. The
1137:         reader holds no lock, so it cannot tell an interrupted append from
1138:         one that is in flight this instant, and it must assume the kinder of
1139:         the two.
1140: 
1141:         **This is a generator: calling it validates nothing.** Every check
1142:         above runs as records are pulled, so ``log.walk()`` on its own is
1143:         not an integrity test and must not be used as one — ``list(...)``
1144:         it, or iterate it to the end. The laziness is deliberate (a caller
1145:         reading the first ten records of a large log should not pay for the
1146:         rest), and it is worth naming because "I called walk and it did not
1147:         raise" is a natural and wrong way to read this code.
1148:         """
1149:         yield from _records_only(self._walk_marked_from_zero())
1150: 
1151:     def _verify_from(
1152:         self, fh, *, expected: int, prev: str | None, lineage: str | None
1153:     ) -> Iterator[tuple[dict, int]]:
1154:         """The per-record verification loop — the one home for it.
1155: 
1156:         :meth:`walk` and :meth:`_walk_tail` differ only in where they start
1157:         and how the lineage is established; every rule applied to a record
1158:         is the same, and it lives here so it cannot be added to one path and
1159:         forgotten on the other. It was duplicated for two review rounds, and
1160:         both rounds had to patch the same check twice.
1161: 
1162:         ``lineage`` is None only for a walk from ordinal 0, where the
1163:         genesis names it. A resume passes the genesis's lineage in, so the
1164:         anchor cannot decide what the rest of the file is checked against.
1165: 
1166:         Yields ``(record, end_offset)`` — the byte just past each record's
1167:         newline, tracked from the handle's position at entry so a consumer
1168:         (:meth:`walk_marked`) can persist an exact resume mark. The public
1169:         walks project the record out.
1170:         """
1171:         offset = fh.tell()
1172:         for raw in fh:
1173:             if not raw.endswith(b"\n"):
1174:                 return  # in flight, or torn — either way not the reader's to judge
1175:             offset += len(raw)
1176:             try:
1177:                 record = decode_record(raw[:-1])
1178:             except ArrivalGrammarError as exc:
1179:                 raise ArrivalCorrupt(str(exc), expected) from exc
1180:             fault = _placement_fault(record, expected)
1181:             if fault is not None:
1182:                 raise ArrivalCorrupt(fault, expected)
1183:             if lineage is None:
1184:                 lineage = record["lin"]
1185:             elif record["lin"] != lineage:
1186:                 raise ArrivalCorrupt(
1187:                     f"record claims lineage {record['lin']!r}, "
1188:                     f"but this log's lineage is {lineage!r}",
1189:                     expected,
1190:                 )
1191:             if record["ord"] != expected:
1192:                 raise ArrivalCorrupt(
1193:                     f"ordinal succession broken: found ord {record['ord']!r} "
1194:                     f"where {expected} was due",
1195:                     expected,
1196:                 )
1197:             if record["prev"] != prev:
1198:                 raise ArrivalCorrupt(
1199:                     f"prev names {record['prev']!r}, "
1200:                     f"but the preceding record's rh is {prev!r}",
1201:                     expected,
1202:                 )
1203:             yield record, offset
1204:             expected += 1
1205:             prev = record[_RH]
1206: 
1207:     def read(self, ordinal: int) -> dict:
1208:         """The record at ``ordinal``, reached by a verified walk.
1209: 
1210:         The walk is the only way in on purpose: reading a coordinate means
1211:         having established that everything before it is intact, because a
1212:         record whose predecessors do not chain is not at the coordinate it
1213:         claims.
1214:         """
1215:         if not _is_int(ordinal) or ordinal < 0:
1216:             raise ArrivalError(f"ordinal must be an integer >= 0, got {ordinal!r}")
1217:         for record in self.walk():
1218:             if record["ord"] == ordinal:
1219:                 return record
1220:         raise ArrivalError(f"{self.path} holds no record at ordinal {ordinal}")
1221: 
1222:     def head(self) -> dict:
1223:         """The record at the highest ordinal currently in the log."""
1224:         record = None
1225:         for record in self.walk():  # noqa: B007 — the last one is the answer
1226:             pass
1227:         if record is None:
1228:             raise GenesisRefused(f"{self.path} holds no complete record")
1229:         return record
1230: 
1231:     def _tail_record(self) -> dict:
1232:         """The last complete record, found by reverse scan rather than walk.
1233: 
1234:         Append needs the head and nothing else, and it needs it in constant
1235:         time: walking the whole log on every append would make ingest
1236:         quadratic. The record's own ``rh`` still recomputes here, so a tail
1237:         that has been edited is caught; whole-log density and chain
1238:         verification stay :meth:`walk`'s job.
1239: 
1240:         The head is ADOPTED AS AN AUTHORITY — the next record's coordinate
1241:         and ``prev`` are derived from it — so it is validated as one
1242:         (:func:`_authority_fault`), and the genesis is read first to
1243:         establish what this log's lineage actually is. Without that, a log
1244:         whose ordinal-0 record is an unsigned or foreign genesis kept
1245:         accepting appends and growing, which is worse than a refused write:
1246:         honest-looking records accumulating on top of a tampered one is
1247:         exactly what makes the tamper hard to see later.
1248:         """
1249:         size = self.size()
1250:         if size == 0:
1251:             raise GenesisRefused(f"{self.path} is empty — mint a genesis first")
1252:         with self.path.open("rb") as fh:
1253:             # One handle for both reads. The genesis line is read on every
1254:             # append through the handle already open for the tail, and its
1255:             # validation is memoized on those bytes (:meth:`_lineage_from`)
1256:             # — so the check still runs against what is on disk right now.
1257:             lineage = self._lineage_from(fh)
1258:             fh.seek(size - 1)
1259:             if fh.read(1) != b"\n":
1260:                 raise ArrivalTornTail(
1261:                     f"{self.path} ends mid-record — truncate the torn tail first"
1262:                 )
1263:             start = _last_newline_before(fh, size - 1)
1264:             fh.seek(start)
1265:             raw = fh.read(size - 1 - start)
1266:         try:
1267:             record = decode_record(raw)
1268:         except ArrivalGrammarError as exc:
1269:             raise ArrivalCorrupt(str(exc), -1) from exc
1270:         fault = _authority_fault(record, lineage)
1271:         if fault is not None:
1272:             raise ArrivalCorrupt(f"cannot append onto this head: {fault}", record["ord"])
1273:         return record

1277:     def resume_offset(self, mark: ResumeMark | None) -> int:
1278:         """The byte offset ``mark`` licenses, or 0 when it does not.
1279: 
1280:         The checks live in :meth:`_anchor_for`, which is where the anchor is
1281:         actually adopted; this is the offset-only view of its answer.
1282:         Rejection is not an error — the reader restarts from ordinal 0.
1283:         """
1284:         adopted = self._anchor_for(mark)
1285:         return 0 if adopted is None else adopted[0]

1287:     def anchor(self, mark: ResumeMark | None) -> dict | None:
1288:         """The validated anchor record dict for ``mark``, or None on rejection.
1289: 
1290:         Surfaces the existing :meth:`_anchor_for` validation machinery as a
1291:         public verb. The byte offset in ``mark`` is an internal verified seek
1292:         hint, trusted only after the record found there passes all anchor
1293:         checks (shape, lineage, ordinal, authority).
1294: 
1295:         Anchor validation checks self-consistency only, never chaining back to
1296:         genesis. :meth:`walk` remains the integrity statement over full history.
1297:         """
1298:         adopted = self._anchor_for(mark)
1299:         return None if adopted is None else adopted[1]
1300: 
1301:     def _anchor_for(self, mark: ResumeMark | None) -> tuple[int, dict, str] | None:
1302:         """Adopt ``mark``'s anchor, or None when the mark cannot be trusted.
1303: 
1304:         The single adoption site for the resume path: it reads the anchor
1305:         once and applies every check, and both :meth:`resume_offset` and
1306:         :meth:`walk_from` consume its answer. They used to validate and
1307:         re-read separately, which meant the place where the anchor actually
1308:         became an authority was not the place the checks lived — the proxy
1309:         shape behind two rounds of findings.
1310: 
1311:         Five checks, all of which must hold:
1312: 
1313:         1. The mark's lineage is this log's. A mark from another lineage is
1314:            not stale, it is a mark about a different file, and pointing it
1315:            here is meaningless.
1316:         2. The offset is an integer inside ``0..size``. A negative offset is
1317:            not a seek position to try, it is metadata that cannot be true.
1318:         3. It lands strictly on a record boundary.
1319:         4. The record ending there carries the ordinal the mark claims.
1320:         5. That record may be adopted as an authority
1321:            (:func:`_authority_fault`) — its own lineage is the genesis's,
1322:            and it may sit where it claims to sit. Without the first half, a
1323:            crafted interior record with a foreign lineage and a recomputing
1324:            ``rh`` becomes the anchor and the tail walk verifies the rest of
1325:            the file against the attacker's lineage. Without the second, a
1326:            record the full walk refuses becomes a legal place to resume
1327:            from, and the resume path consumes straight past corruption the
1328:            read path stops at.
1329: 
1330:         What these do NOT establish is in :func:`_authority_fault`: the
1331:         anchor is checked for self-consistency, never for chaining back to
1332:         genesis. The mark is a cheap boundary check; :meth:`walk` is the
1333:         integrity statement.
1334: 
1335:         Returns ``(offset, anchor, lineage)``, the lineage always the
1336:         genesis's and never the anchor's.
1337:         """
1338:         if mark is None:
1339:             return None
1340:         if not self.path.exists():
1341:             return None
1342:         try:
1343:             lineage = self.lineage()
1344:         except (GenesisRefused, ArrivalError):
1345:             return None
1346:         if mark.arrival_lineage != lineage:
1347:             return None
1348:         if not _is_int(mark.arrival_offset) or not _is_int(mark.arrival_ordinal):
1349:             return None
1350:         if mark.arrival_offset == 0:
1351:             return None  # nothing consumed; identical to a fresh start
1352:         if not 0 <= mark.arrival_offset <= self.size():
1353:             return None
1354:         anchor = self._record_ending_at(mark.arrival_offset)
1355:         if anchor is None or anchor["ord"] != mark.arrival_ordinal:
1356:             return None
1357:         if _authority_fault(anchor, lineage) is not None:
1358:             return None
1359:         return mark.arrival_offset, anchor, lineage
1360: 
1361:     def _record_ending_at(self, offset: int) -> dict | None:
1362:         """The record whose ``\\n`` is the byte before ``offset``, or None
1363:         when the offset is not a boundary or the record does not decode."""
1364:         with self.path.open("rb") as fh:
1365:             fh.seek(offset - 1)
1366:             if fh.read(1) != b"\n":
1367:                 return None
1368:             start = _last_newline_before(fh, offset - 1)
1369:             fh.seek(start)
1370:             raw = fh.read(offset - 1 - start)
1371:         try:
1372:             return decode_record(raw)
1373:         except ArrivalGrammarError:
1374:             return None

1376:     def walk_from(self, mark: ResumeMark | None) -> tuple[int, Iterator[dict]]:
1377:         """Resume a consumption: the ordinal resumed at, and the records after.
1378: 
1379:         :meth:`walk_marked` minus the marks — one adoption/verification
1380:         path, two projections of it. Returns ``(0, ...)`` whenever the mark
1381:         is rejected, so a caller that persists progress learns it is
1382:         starting over instead of discovering it by ordinal.
1383:         """
1384:         resumed, pairs = self.walk_marked(mark)
1385:         return resumed, _records_only(pairs)
1386: 
1387:     def walk_marked(
1388:         self, mark: ResumeMark | None
1389:     ) -> tuple[int, Iterator[tuple[dict, ResumeMark]]]:
1390:         """:meth:`walk_from`, yielding each record WITH its resume mark.
1391: 
1392:         For the consumer that indexes what it reads and must persist how far
1393:         it got: the mark beside each record is the one that licenses
1394:         resuming just past it, exact to the byte because the verification
1395:         loop tracked it. ``(0, ...)`` means the walk starts over — either no
1396:         mark was offered or the offered one was rejected, and the caller
1397:         that persists progress must treat its prior state accordingly.
1398:         """
1399:         adopted = self._anchor_for(mark)
1400:         if adopted is None:
1401:             resumed, pairs = 0, self._walk_marked_from_zero()
1402:         else:
1403:             offset, anchor, lineage = adopted
1404:             # The lineage comes from the GENESIS, never from the anchor: an
1405:             # anchor is a record found at a byte offset a caller supplied,
1406:             # and letting it name the lineage would let it decide what the
1407:             # rest of the walk is checked against.
1408:             resumed, pairs = anchor["ord"] + 1, self._walk_tail(offset, anchor, lineage)
1409: 
1410:         def marked() -> Iterator[tuple[dict, ResumeMark]]:
1411:             for record, end in pairs:
1412:                 yield record, ResumeMark(
1413:                     arrival_lineage=record["lin"],
1414:                     arrival_offset=end,
1415:                     arrival_ordinal=record["ord"],
1416:                 )
1417: 
1418:         return resumed, marked()
1419: 
1420:     def _walk_marked_from_zero(self) -> Iterator[tuple[dict, int]]:
1421:         if not self.path.exists():
1422:             return
1423:         with self.path.open("rb") as fh:
1424:             yield from self._verify_from(fh, expected=0, prev=None, lineage=None)
1425: 
1426:     def _walk_tail(
1427:         self, offset: int, anchor: dict, lineage: str
1428:     ) -> Iterator[tuple[dict, int]]:
1429:         """:meth:`walk`'s verification, started from a validated anchor.
1430: 
1431:         Literally the same rules, not merely the same list of them: both
1432:         paths run :meth:`_verify_from`, and only the starting point differs.
1433:         ``lineage`` is the log's, established from its genesis by the
1434:         caller, so the anchor never decides what the rest is checked
1435:         against.
1436:         """
1437:         with self.path.open("rb") as fh:
1438:             fh.seek(offset)
1439:             yield from self._verify_from(
1440:                 fh, expected=anchor["ord"] + 1, prev=anchor[_RH], lineage=lineage
1441:             )

1467:     def append_marked(
1468:         self,
1469:         k: str,
1470:         body: dict,
1471:         *,
1472:         observer: str,
1473:         origin: str = "",
1474:         at: float | None = None,
1475:         signer: Signer | None = None,
1476:         following: Head | None = None,
1477:     ) -> tuple[dict, ResumeMark]:
1478:         """:meth:`append`, also returning the mark that resumes just past it.
1479: 
1480:         The mark's offset is read under the append lock — the byte just past
1481:         this record's newline, exact by construction rather than a size read
1482:         that a concurrent append could land inside. A consumer that indexes
1483:         the record it just wrote persists this mark and later resumes
1484:         through :meth:`walk_from`, which re-validates the anchor at the
1485:         adoption site.
1486: 
1487:         ``following`` names the FULL head the caller expects — lineage,
1488:         ordinal and record hash. Checked under the lock, so a record that
1489:         landed since the caller last looked refuses (:class:`StaleHead`)
1490:         BEFORE any byte is written — the compare-and-swap shape for a
1491:         consumer whose staged state must not silently skip an interloper. A
1492:         bare ordinal is refused rather than accepted: see :func:`_checked_pin`.
1493:         """
1494:         pin = _checked_pin(following)
1495: 
1496:         def build(headr: dict) -> dict:
1497:             _refuse_stale_head(headr, pin)
1498:             at_ = time.time() if at is None else at
1499:             sig = (
1500:                 signer(observer, content_commitment(k, at_, observer, origin, body))
1501:                 if signer is not None
1502:                 else None
1503:             )
1504:             return build_record(
1505:                 lin=headr["lin"], ordinal=headr["ord"] + 1, prev=headr[_RH],
1506:                 k=k, body=body, observer=observer, origin=origin, at=at_,
1507:                 sig=sig or None,
1508:             )
1509: 
1510:         record, offset = self._append_under_lock(build)
1511:         return record, ResumeMark(
1512:             arrival_lineage=record["lin"],
1513:             arrival_offset=offset,
1514:             arrival_ordinal=record["ord"],
1515:         )

1517:     def append_marked_many(
1518:         self, entries: Sequence[Entry], *, following: Head | None = None
1519:     ) -> tuple[list[dict], ResumeMark]:
1520:         """Append several records under ONE lock acquisition, ONE fsync.
1521: 
1522:         Not an optimization — a correctness primitive. :meth:`append_marked`
1523:         takes the flock per call, so two processes replaying the same source
1524:         into one target would both pass their dedup pass against the
1525:         pre-merge snapshot and both append row X: the log would carry one id
1526:         twice, and every verb in this design refuses such a log. That is the
1527:         worst artifact this cut could mint, so the whole append phase holds
1528:         the lock.
1529: 
1530:         An outer lock plus per-record :meth:`append_marked` DEADLOCKS instead
1531:         of helping: ``_append_under_lock`` opens a second descriptor on the
1532:         same lock file, and ``flock`` is per open-file-description, so it
1533:         blocks within one process. Hence a primitive rather than a wrapper.
1534: 
1535:         ``following`` is the compare-and-swap pin, checked under the lock
1536:         before any byte is written: it names the FULL head — lineage, ordinal
1537:         and record hash — the caller deduped against, so a record that landed
1538:         since then, or a coordinate that now holds a different record,
1539:         REFUSES (:class:`StaleHead`) rather than letting a stale dedup decide
1540:         what to append. A caller that sees the refusal re-reads and retries.
1541:         A bare ordinal is refused rather than accepted: see
1542:         :func:`_checked_pin`.
1543: 
1544:         ONE trailing fsync is sound: a crash loses only an un-fsynced
1545:         suffix, or leaves one torn tail line that the truncate-under-lock
1546:         rule removes. Recovery is "re-run the merge" — dedup makes it
1547:         idempotent, which is the whole story to tell an operator about an
1548:         interrupted merge.
1549:         """
1550:         if not entries:
1551:             raise ArrivalError(
1552:                 "append_marked_many was handed no entries — an empty append "
1553:                 "is a caller bug, not a no-op to absorb"
1554:             )
1555: 
1556:         def assign(head: dict, index: int) -> dict:
1557:             entry = entries[index]
1558:             at_ = time.time() if entry.at is None else entry.at
1559:             return build_record(
1560:                 lin=head["lin"], ordinal=head["ord"] + 1, prev=head[_RH],
1561:                 k=entry.k, body=entry.body, observer=entry.observer,
1562:                 origin=entry.origin, at=at_, sig=entry.signature,
1563:             )
1564: 
1565:         return self._append_many_under_lock(
1566:             assign, len(entries), pin=_checked_pin(following)
1567:         )
1568: 
1569:     def append_records(
1570:         self, records: Sequence[dict], *, following: Head | None = None
1571:     ) -> tuple[list[dict], ResumeMark]:
1572:         """Append pre-coordinated records under ONE lock acquisition.
1573: 
1574:         :meth:`append_record`'s batched form, and the primitive exact
1575:         replication is composed from: every record already carries its
1576:         lineage, ordinal, ``prev``, signature and ``rh``, and this method
1577:         VALIDATES all of them rather than assigning any. Nothing here rehashes
1578:         or re-signs, so a suffix that lands is byte-identical to the suffix
1579:         that was offered — which is the whole property replication has to
1580:         keep, and the reason it cannot be expressed as a loop over
1581:         :meth:`append_marked_many` (that path assigns coordinates and builds
1582:         every record unsigned).
1583: 
1584:         Batched rather than a loop over :meth:`append_record` for the reason
1585:         :meth:`append_marked_many` gives: ``_append_under_lock`` opens its own
1586:         descriptor on the lock file, and ``flock`` is per open-file-
1587:         description, so an outer lock plus per-record appends DEADLOCKS within
1588:         one process. A partially applied suffix is also the artifact a replica
1589:         must never hold, so the whole batch lands under one lock and one
1590:         fsync or none of it does.
1591: 
1592:         ``following`` is the same full-head compare-and-swap pin every append
1593:         takes, checked inside the fence.
1594: 
1595:         Two refusals before any byte is written, and they are different
1596:         claims: a candidate at a height this log already holds DIFFERENTLY is
1597:         a fork (:class:`ForkedHeight`, and retrying cannot clear it), while a
1598:         batch that simply does not follow the current head is a stale view of
1599:         it (:class:`StaleHead`, and re-reading is exactly the fix).
1600:         """
1601:         if not records:
1602:             raise ArrivalError(
1603:                 "append_records was handed no records — an empty append is a "
1604:                 "caller bug, not a no-op to absorb"
1605:             )
1606:         # Completed before the lock: the work is pure, and a malformed batch
1607:         # should refuse without ever having held the fence.
1608:         candidates = [_completed_candidate(record) for record in records]
1609: 
1610:         def carry(head: dict, index: int) -> dict:
1611:             if index == 0:
1612:                 self._refuse_divergent_batch(candidates, head)
1613:             return candidates[index]
1614: 
1615:         return self._append_many_under_lock(
1616:             carry, len(candidates), pin=_checked_pin(following)
1617:         )
1618: 
1619:     def _refuse_divergent_batch(self, candidates: list[dict], head: dict) -> None:
1620:         """Classify a carried-in batch against the head, before any mutation.
1621: 
1622:         Called under the lock, so the head it judges against is the head the
1623:         write would land on. It names WHERE the disagreement is and refuses;
1624:         it never picks a winner.
1625: 
1626:         Order matters: the fork check runs first. A batch that overlaps the
1627:         log divergently is also, trivially, a batch that does not follow the
1628:         head — reporting that as a stale head would tell the caller to
1629:         re-read and retry, and retrying a fork spins forever. The narrower,
1630:         more informative claim goes first.
1631:         """
1632:         occupied = sorted(
1633:             {
1634:                 candidate["ord"]
1635:                 for candidate in candidates
1636:                 if _is_int(candidate.get("ord")) and candidate["ord"] <= head["ord"]
1637:             }
1638:         )
1639:         held = self._hashes_at(occupied, head)
1640:         for candidate in candidates:
1641:             ordinal = candidate.get("ord")
1642:             mine = held.get(ordinal) if _is_int(ordinal) else None
1643:             if mine is not None and mine != candidate.get(_RH):
1644:                 raise ForkedHeight(
1645:                     f"ordinal {ordinal} of {self.path} holds rh {mine}, but "
1646:                     f"the offered record carries rh {candidate.get(_RH)} — the "
1647:                     "two histories disagree at this height, so this is a fork "
1648:                     "and not a suffix. Replication stops rather than choosing"
1649:                 )
1650:         first = candidates[0]
1651:         if first.get("ord") != head["ord"] + 1 or first.get("prev") != head[_RH]:
1652:             raise StaleHead(
1653:                 f"this batch starts at ({first.get('lin')}, {first.get('ord')}) "
1654:                 f"with prev {first.get('prev')}, but the log's head is "
1655:                 f"({head['lin']}, {head['ord']}, {head[_RH]}) — records arrived "
1656:                 "since the caller computed the suffix, or it was computed "
1657:                 "against a different log. Re-read the head and recompute which "
1658:                 "records are missing"
1659:             )
1660: 
1661:     def _hashes_at(self, ordinals: Sequence[int], head: dict) -> dict[int, str]:
1662:         """The ``rh`` this log holds at each of ``ordinals``.
1663: 
1664:         The head is answered from the record already in hand; anything below
1665:         it costs one walk, which only a batch that overlaps the log pays for.
1666:         An ordinal the log does not reach is simply absent from the answer —
1667:         "there is nothing there to disagree with" is not a fork.
1668:         """
1669:         wanted = set(ordinals)
1670:         found: dict[int, str] = {}
1671:         if head["ord"] in wanted:
1672:             found[head["ord"]] = head[_RH]
1673:             wanted.discard(head["ord"])
1674:         if not wanted:
1675:             return found
1676:         for record in self.walk():
1677:             if record["ord"] in wanted:
1678:                 found[record["ord"]] = record[_RH]
1679:                 wanted.discard(record["ord"])
1680:                 if not wanted:
1681:                     break
1682:         return found

1684:     def _append_many_under_lock(
1685:         self,
1686:         build_each: Callable[[dict, int], dict],
1687:         count: int,
1688:         *,
1689:         pin: Head | None,
1690:     ) -> tuple[list[dict], ResumeMark]:
1691:         """The whole batched write sequence, indivisible.
1692: 
1693:         :meth:`_append_under_lock`'s batched twin, and the ONE place a group
1694:         of records becomes bytes. Both batched appends are this method under a
1695:         different ``build_each``: :meth:`append_marked_many` assigns each
1696:         record's coordinate from the running head, :meth:`append_records`
1697:         hands back a record that already carries one. Nothing else differs,
1698:         so nothing else is written twice — the fence, the compare-and-swap,
1699:         the per-record chain checks, the single write and the single fsync are
1700:         stated once and inherited by both.
1701: 
1702:         ``build_each(head, index)`` is called with the head the record must
1703:         follow — the log's real head for index 0, the previous record after
1704:         that — so a build that needs to judge the batch against the log can do
1705:         it at index 0, still inside the fence and still before a byte moves.
1706: 
1707:         ONE trailing fsync is sound for the reason
1708:         :meth:`append_marked_many` gives: a crash loses only an un-fsynced
1709:         suffix, or leaves one torn tail line that the truncate-under-lock rule
1710:         removes.
1711:         """
1712:         with self._locked_head() as head:
1713:             _refuse_stale_head(head, pin)
1714:             records: list[dict] = []
1715:             encoded: list[str] = []
1716:             for index in range(count):
1717:                 record = build_each(head, index)
1718:                 # Every record is held to the same head checks a single
1719:                 # append is, against the record before it — the chain is
1720:                 # built here, so it is validated here. `encode_record` is the
1721:                 # grammar gate and the supplied-`rh` check both.
1722:                 self._check_follows(record, head)
1723:                 encoded.append(encode_record(record) + "\n")
1724:                 records.append(record)
1725:                 head = record
1726:             with self.path.open("ab") as fh:
1727:                 fh.write("".join(encoded).encode("utf-8"))
1728:                 fh.flush()
1729:                 os.fsync(fh.fileno())
1730:                 offset = fh.tell()
1731:         last = records[-1]
1732:         return records, ResumeMark(
1733:             arrival_lineage=last["lin"],
1734:             arrival_offset=offset,
1735:             arrival_ordinal=last["ord"],
1736:         )

1738:     def append_record(self, record: dict) -> dict:
1739:         """Append a record that already carries its coordinate.
1740: 
1741:         The primitive the grammar describes: append VALIDATES, it never
1742:         assigns. A candidate assembled elsewhere — carried in from another
1743:         store, replayed, rebuilt — passes through exactly the same head
1744:         checks that :meth:`append` does. ``rh`` may be
1745:         omitted and is then computed over the validated object; supplied, it
1746:         must match, so a candidate carried in with its own digest is checked

1757:         the head. Truncation lives here and only here — the lock is what
1758:         makes "no append is in progress" knowable, and it is the only
1759:         condition under which an unterminated tail can be called torn
1760:         instead of in flight. The lock is per open-file-description, so it
1761:         releases exactly when this ``with`` closes the descriptor; every

1775:         validate, encode, write, fsync, release.

1778:         merely a process crash. That was strong reasoning when the line was

1976: def verify_authorship(log: ArrivalLog, verify: Verify) -> tuple[KeyResolution, ...]:
1977:     """Verify every signature in the log, resolving keys from the log alone.
1978: 
1979:     The ruled rule, implemented verbatim (decision:design/arrival-sliceA-
1980:     authority): **a key is valid at position N iff it was introduced at a
1981:     position < N, or N is the genesis position and the record is
1982:     self-certifying.** Self-certification is legal at ordinal 0 and nowhere
1983:     else — the genesis signature must verify against its own
1984:     ``body["key"]``, which is what turns a self-naming genesis into a
1985:     self-certifying one. A key introduction at ordinal >= 1
1986:     (:data:`KEY_INTRODUCTION_KIND`) is a record whose body names an
1987:     observer and a key, signed by a key already valid for the INTRODUCING
1988:     record's observer; only after that signature verifies does the named
1989:     key join the registry, at the introduction's own ordinal.
1990: 
1991:     Keys are bound to observers, not free-floating (an instantiation
1992:     decision recorded in the slice-A design fact): the genesis key speaks
1993:     for the genesis record's observer, an introduced key for the observer
1994:     its body names. A record's signature is tried against every key valid
1995:     for that record's OWN observer at its position — observer-agnostic
1996:     resolution would let one observer's record verify under another's key,
1997:     and authorship would stop being an answer.
1998:

2000:     projection, no artifact beside the log (the ruled exclusivity claim).
2001:     ``verify`` is injected, never imported, same posture as :data:`Signer`.

2004:     integrity is established for free; this is an O(n) surface where the
2005:     whole file is the question. Unsigned non-genesis records make no
2006:     authorship claim and produce no row. Returns one
2007:     :class:`KeyResolution` per verified signature, in log order; raises
2008:     :class:`AuthorshipUnverified` on the first signature that no valid key
2009:     verifies.

2012:     :func:`key_registry` — this verb is that one plus the verify-every-

1: """jsonl_codec — the line codec for the canonical JSONL store.
2: 
3: One interleaved append-only log per store (``.loops/data/<name>.jsonl``);
4: each line is a JSON object carrying a ``"t"`` discriminator (``"fact"``,
5: ``"tick"``, or the ``"batch"`` envelope for multi-row ceremonies) plus the
6: persisted row fields, in sqlite column order.
7: 
8: The load-bearing invariant (design/architecture/jsonl-canonical-store):
9: **payload rides as the VERBATIM stored TEXT string** — a JSON string value,
10: never re-serialized, never parsed by this codec. Every commitment hash in
11: ``engine.sqlite_store`` (``_fact_row_hash``, ``_fact_commitment_hash``,
12: ``_tick_envelope``/``_tick_row_hash``) embeds payload verbatim, so a line
13: round-tripped through this codec re-derives byte-identical hashes and every
14: existing signature keeps verifying.
15: 
16: Era handling mirrors the era-aware hashers exactly. All commitment fields
17: are always present (``null`` when the row holds NULL — pre-chain ticks); the
18: ``signature`` key is emitted only when the row carries one, which is precisely
19: the condition under which the hashers fold it into the envelope. A 6-tuple
20: fact row and a 7-tuple row with ``signature=None`` therefore hash identically
21: and serialize identically; deserializers return the full-arity tuple (7 fact
22: fields / 11 tick fields) with ``None`` filled in.
23: 
24: Unknown, missing, or mistyped fields are rejected loudly
25: (:class:`JsonlCodecError`) — explicit over implicit. A store's canonical log
26: is not a place for silent tolerance. So is a duplicate key: JSON's last-wins
27: resolution would let one line carry two ids and a reader silently pick one.
28: 
29: The ``t`` discriminator is THIS framing's, and it stays. A line has nowhere
30: but the object itself to say which record class it carries; an arrival
31: record says it in the envelope's ``k``, so wire v1 drops ``t`` from arrival
32: bodies and :mod:`engine.arrival_body` frames the same rows without it
33: (decision:design/arrival-wire-v1-seam-triage). This module keeps ``t``
34: because it still has live consumers that need it: the derived ``.jsonl``
35: projection every last-0.x reader reads, and the migration sidecar that
36: reads historical logs. What the two framings share is the ROW domain —
37: :func:`row_object_fault`, exported for exactly that — so they cannot drift
38: about what a well-formed row is while disagreeing about how to frame one.
39: 
40: The rules run in **both** directions from one function. ``serialize`` holds
41: its object to the same domain ``deserialize`` enforces, so ``serialize(x)``
42: is always decodable — a wrongly typed field fails at the append site, where
43: it is attributable, rather than becoming a durable line that bricks every
44: later open.

153: def _dump(obj: dict) -> str:
154:     """Encode one line. ensure_ascii keeps lines 7-bit and free of raw
155:     U+2028/U+2029; allow_nan=False refuses non-JSON floats explicitly;
156:     separators drop insignificant whitespace. Key order is the dict's
157:     insertion order — sqlite column order — not sorted: this is a transport
158:     encoding, not a canonicalization (JCS lives in the hashers)."""

163: def _encode_obj(row: tuple, spec: _Spec) -> dict:
164:     """Build the line object for a row — and hold it to the decoder's rules.
165: 
166:     ``serialize(x)`` must always be decodable. Checking arity alone let a
167:     row with a *typed*-wrong field through: a fact carrying ``ts="1.0"``
168:     (a string, which sqlite's REAL affinity accepts) got a durable JSONL
169:     receipt, and then every subsequent open failed on decode — the store
170:     bricked by a line its own serializer wrote. Running the decoder's
171:     :func:`_validate` over the object before dumping makes the two
172:     directions symmetric by construction rather than by parallel
173:     maintenance, and puts the error at the append site, where it is
174:     attributable to the caller that built the row.
175:     """
176:     n = len(spec.fields)
177:     if len(row) not in (n, n + 1):
178:         raise JsonlCodecError(
179:             f"{spec.t} row must have {n} or {n + 1} fields, got {len(row)}"
180:         )
181:     obj: dict = {"t": spec.t}
182:     obj.update(zip(spec.fields, row[:n], strict=True))
183:     if len(row) > n and row[n] is not None:
184:         obj[_SIGNATURE] = row[n]
185:     _validate(obj, spec)
186:     return obj

248: def _reject_constant(name: str) -> None:
249:     """Refuse the non-JSON literals ``NaN``/``Infinity``/``-Infinity``.
250: 
251:     Serialization uses ``allow_nan=False``, so these can never be emitted;
252:     admitting them on read would let a corrupt line past the codec gate and

259: def _no_duplicate_keys(pairs: list[tuple[str, object]]) -> dict:
260:     """Build the object, refusing a key spelled twice.
261: 
262:     ``json.loads`` resolves duplicates last-wins, so ``"id":"A","id":"B"``
263:     decodes as ``B`` — two different lines with two different meanings that
264:     a reader silently collapses to one. A canonical log admits exactly one
265:     spelling per record; ambiguity is corruption, not a parse detail.
266:     """
267:     obj: dict = {}
268:     for key, value in pairs:
269:         if key in obj:
270:             raise JsonlCodecError(f"duplicate key {key!r} in line")
271:         obj[key] = value
272:     return obj

275: def _load(line: str) -> dict:
276:     try:
277:         obj = json.loads(
278:             line, parse_constant=_reject_constant, object_pairs_hook=_no_duplicate_keys
279:         )
280:     except JsonlCodecError:
281:         # Raised by our own hooks — already the right error, with the right
282:         # message. Re-wrapping it as "not valid JSON" would bury the reason.
283:         raise
284:     except ValueError as exc:
285:         raise JsonlCodecError(f"not valid JSON: {exc}") from exc
286:     if not isinstance(obj, dict):
287:         raise JsonlCodecError(
288:             f"line must be a JSON object, got {type(obj).__name__}"
289:         )

293: def row_object_fault(
294:     obj: dict,
295:     *,
296:     t: str,
297:     frame: str,
298:     fields: tuple[str, ...],
299:     allowed: frozenset[str],
300:     nullable: frozenset[str],
301: ) -> str | None:
302:     """Why ``obj`` is not a well-formed row object, or None when it is.
303: 
304:     The half the two row framings SHARE. A ``.jsonl`` line and an arrival
305:     record's body carry the same sqlite row and differ only in where the
306:     record class is written — inside the object as ``"t"`` for a line, in
307:     the envelope's ``k`` for an arrival body — so the field domain is one
308:     rule with two callers rather than two rules that must be kept in step.
309:     ``allowed`` is exactly where they fork: the line framing admits ``t``
310:     and the arrival body does not (decision:design/arrival-wire-v1-seam-triage).
311: 
312:     Returns a fault string rather than raising, the shape
313:     :func:`engine.arrival._placement_fault` and
314:     :func:`engine.arrival._key_shape_fault` already use, so each framing
315:     raises its OWN error class over one set of rules. ``frame`` names the
316:     container in the two messages that mention it, so a body's refusal does
317:     not tell the reader to go looking for a line.
318:     """
319:     unknown = sorted(set(obj) - allowed)
320:     if unknown:
321:         return f"unknown field(s) in {t} {frame}: {unknown}"
322:     missing = [f for f in fields if f not in obj]
323:     if missing:
324:         return f"missing field(s) in {t} {frame}: {missing}"
325:     for field in fields:
326:         value = obj[field]
327:         if value is None:
328:             if field not in nullable:
329:                 return f"{t} field {field!r} must not be null"
330:             continue
331:         if field in _NUMERIC:
332:             # bool is an int subclass — reject it explicitly.
333:             if isinstance(value, bool) or not isinstance(value, (int, float)):
334:                 return (
335:                     f"{t} field {field!r} must be a number, got "
336:                     f"{type(value).__name__}"
337:                 )
338:             # Standards-valid JSON can still spell a number outside the JCS
339:             # domain (``1e999`` overflows to inf; a 400-digit integer exceeds
340:             # the safe-integer bound). Range-check here — the literal parser
341:             # only sees the NaN/Infinity spellings.
342:             if isinstance(value, float):
343:                 if not math.isfinite(value):
344:                     return (
345:                         f"{t} field {field!r} must be a finite number, got "
346:                         f"{value!r}"
347:                     )
348:             elif not (_JCS_INT_MIN <= value <= _JCS_INT_MAX):
349:                 return (
350:                     f"{t} field {field!r} is outside the JCS safe-integer "
351:                     f"domain: {value}"
352:                 )
353:         elif not isinstance(value, str):
354:             return (
355:                 f"{t} field {field!r} must be a string, got "
356:                 f"{type(value).__name__}"
357:             )
358:     if SIGNATURE_FIELD in obj and obj[SIGNATURE_FIELD] is None:
359:         # Absent IS the unsigned era; an explicit null is a second spelling
360:         # of the same state, so serialize stays the unique canonical form.
361:         return f"{t} field 'signature' must be absent, not null, when unsigned"
362:     sig = obj.get(SIGNATURE_FIELD)
363:     if sig is not None and not isinstance(sig, str):
364:         return f"{t} field 'signature' must be a string, got {type(sig).__name__}"
365:     return None
366:

368: def _validate(obj: dict, spec: _Spec) -> None:
369:     fault = row_object_fault(
370:         obj,
371:         t=spec.t,
372:         frame="line",
373:         fields=spec.fields,
374:         allowed=spec.allowed,
375:         nullable=spec.nullable,
376:     )
377:     if fault is not None:
378:         raise JsonlCodecError(fault)

381: def _validate_batch(obj: dict, *, validate_rows: bool = True) -> None:
382:     """Hold a batch envelope to the structural rules — both directions.
383: 
384:     Two halves: the STRUCTURAL rules (envelope keys, ≥ 2 rows, fact records
385:     only — no nesting, no ticks — and no duplicate id) always run; the
386:     per-row FIELD validation (``_validate``) runs only when
387:     ``validate_rows`` — the deserialize path keeps it, the serialize path
388:     skips it because its row objects were just built (and validated) by
389:     ``_encode_obj``.
390:     """
391:     unknown = sorted(set(obj) - _BATCH_KEYS)
392:     if unknown:
393:         raise JsonlCodecError(f"unknown field(s) in batch line: {unknown}")
394:     rows = obj.get(_ROWS)
395:     if not isinstance(rows, list):
396:         raise JsonlCodecError(
397:             "batch field 'rows' must be an array of fact records, got "
398:             f"{type(rows).__name__}"
399:         )
400:     if len(rows) < _MIN_BATCH_ROWS:
401:         raise JsonlCodecError(
402:             f"batch must carry at least {_MIN_BATCH_ROWS} rows, got "
403:             f"{len(rows)} — a 1-row batch is a second spelling of a plain "
404:             "fact line, and an empty one encodes nothing"
405:         )
406:     seen_ids: set = set()
407:     for i, elem in enumerate(rows):
408:         if not isinstance(elem, dict):
409:             raise JsonlCodecError(
410:                 f"batch row {i} must be a JSON object, got {type(elem).__name__}"
411:             )
412:         t = elem.get("t")
413:         if t == _BATCH:
414:             raise JsonlCodecError(f"batch row {i} is a nested batch — batches do not nest")
415:         if t == "tick":
416:             raise JsonlCodecError(
417:                 f"batch row {i} is a tick record — ticks are minted "
418:                 "one-at-a-time and chain-linked, never batched"
419:             )
420:         if t != "fact":
421:             raise JsonlCodecError(f"batch row {i} has unknown record discriminator t={t!r}")
422:         if validate_rows:
423:             _validate(elem, _SPEC["fact"])
424:         row_id = elem["id"]
425:         if row_id in seen_ids:
426:             raise JsonlCodecError(f"duplicate id {row_id!r} within one batch")
427:         seen_ids.add(row_id)

431:     """A VALIDATED record object as its full-arity row tuple."""
432:     return (*(obj[f] for f in spec.fields), obj.get(_SIGNATURE))

435: def _ordered(obj: dict, spec: _Spec) -> dict:
436:     """A VALIDATED record object, rebuilt in canonical field order.
437: 
438:     Same shape :func:`_encode_obj` builds from a row tuple — discriminator,
439:     then the spec's fields in sqlite column order, then ``signature`` when
440:     the row carries one. Rebuilt rather than re-dumped as-received because
441:     key order in a decoded object is whatever its producer happened to use,
442:     and :func:`serialize_object` promises the bytes the row serializers
443:     produce.
444:     """
445:     out: dict = {"t": spec.t}
446:     out.update((field, obj[field]) for field in spec.fields)
447:     if obj.get(_SIGNATURE) is not None:
448:         out[_SIGNATURE] = obj[_SIGNATURE]
449:     return out

452: def deserialize_row(line: str) -> tuple[str, tuple]:
453:     """Decode a SINGLE-record line, dispatching on ``"t"``. Returns
454:     ``(t, row)`` with the row at full arity (7 fact fields / 11 tick fields,
455:     signature last). A ``"t":"batch"`` line carries several records and is
456:     refused here — decode it with :func:`deserialize_records`.
457: 
458:     Defined AS :func:`deserialize_records` plus the multi-record refusal
459:     (a valid batch always expands to ≥ 2 rows), so decoding has exactly
460:     one dispatch."""
461:     records = deserialize_records(line)
462:     if len(records) > 1:
463:         raise JsonlCodecError(
464:             "batch line carries multiple records — decode with "
465:             "deserialize_records, not deserialize_row"
466:         )
467:     return records[0]
468: 
469: 
470: def deserialize_records(line: str) -> list[tuple[str, tuple]]:
471:     """Decode ANY line into its record sequence, in on-the-wire order.
472: 
473:     A plain fact/tick line yields one ``(t, row)``; a batch line yields its
474:     inner rows expanded in array order (always ≥ 2 — so ``len > 1`` is
475:     exactly "this line was a batch"). This is the decode every log consumer
476:     (replay, catch-up, rebuild, audit) reads through, so batch expansion has
477:     one spelling.
478: 
479:     Defined as the line decode composed with :func:`records_from_object`,
480:     so decoding still has exactly one dispatch.
481:     """
482:     return records_from_object(_load(line))
483: 
484: 
485: def records_from_object(obj: dict) -> list[tuple[str, tuple]]:
486:     """:func:`deserialize_records` on an ALREADY-DECODED object.
487: 
488:     An arrival record's ``body`` IS this codec's object for the row it
489:     carries, so a consumer holding one has nothing to parse — before this
490:     entry existed, ``arrival_store`` re-encoded the body just to hand a
491:     string back to the line decoder.
492: 
493:     The SAME validator runs: an object handed in is held to exactly the
494:     domain a line is held to. What a line has and an object cannot is a
495:     duplicate key, which the decode hook catches; everything downstream of
496:     that hook is shared.
497:     """
498:     if not isinstance(obj, dict):
499:         raise JsonlCodecError(
500:             f"record must be a JSON object, got {type(obj).__name__}"
501:         )
502:     t = obj.get("t")
503:     if t == _BATCH:
504:         _validate_batch(obj)
505:         fact = _SPEC["fact"]
506:         return [("fact", _row_of(elem, fact)) for elem in obj[_ROWS]]
507:     spec = _SPEC.get(t) if isinstance(t, str) else None
508:     if spec is None:
509:         raise JsonlCodecError(f"unknown record discriminator t={t!r}")
510:     _validate(obj, spec)
511:     return [(spec.t, _row_of(obj, spec))]
512: 
513: 
514: def serialize_object(obj: dict) -> str:
515:     """Encode an ALREADY-DECODED record object as one canonical line.
516: 
517:     Validated through :func:`records_from_object` — the same domain a line
518:     is held to — then dumped in canonical field order, so the result is
519:     byte-identical to the ``serialize_*`` call that produced the object.
520:     Field order is REBUILT rather than inherited: a decoded object's key
521:     order is its producer's, and this function's contract is about the
522:     codec's bytes, not the caller's dict.
523:     """
524:     records_from_object(obj)
525:     t = obj["t"]
526:     if t == _BATCH:
527:         fact = _SPEC["fact"]
528:         return _dump({"t": _BATCH, _ROWS: [_ordered(e, fact) for e in obj[_ROWS]]})
529:     return _dump(_ordered(obj, _SPEC[t]))

=== docs/scratch/arrival-break/legacy-provenance-verification-2026-09-18.md ===
1: # Audited legacy provenance verification
2: 
3: The user accepted preserved historical boundaries whose fact-window commitments
4: refer to the original archive, with a separately audited observer mapping. The
5: verifier must prove that relationship; it must not turn off historical checks.
6: 
7: ## Meaning of a passing result
8: 
9: A passing provenance result means the independently pinned original archive and
10: preparation manifest describe the supplied prepared input exactly; the pinned
11: migration prefix preserves that input's facts and ticks in order; the original
12: historical chain passes; every transformed historical window difference is
13: explained by the authorized observer mapping; and later tick-chain checks pass.
14: It is reported as **preserved boundaries with audited transformation**. The old
15: window commitments are not described as unchanged commitments over the prepared
16: facts. The ordinary deep audit retains its existing result.
17: 
18: ## Trust and scope
19: 
20: The original archive SHA-256, preparation-manifest SHA-256, and migration head S
21: (lineage, ordinal, record hash) are review inputs supplied independently of the
22: files being checked. A file's own claims are not sufficient trust anchors. For
23: the loops rehearsal these pins come from the previously reviewed snapshot,
24: preparation and verified migration evidence. Replacing all files and all trusted
25: pins together is outside the verifier's ability to detect; the cutover procedure
26: must retain the reviewed pins separately.
27: 
28: The first supported policy is `legacy-unattributed-v1`: flat legacy JSONL, no
29: migration drops, exact empty observers on unsigned facts only. IDs, payload text,
30: all other fact fields, order, and every tick field remain preserved. Rows that do
31: not change must remain byte-identical between original and prepared input. The
32: manifest must account for every changed row exactly once, with matching original
33: and prepared line hashes. Unsupported formats and policies refuse explicitly.
34: 
35: This is an offline, read-only sidecar. It does not open live runtime writers,
36: create credentials, rewrite ticks, refresh signatures, or change the standard
37: runtime audit. Its chain checks are structural/content checks, not an independent
38: authentication of all historical authors or signing keys. The preexisting signed
39: migration report remains separate evidence of migration at S.
40: 
41: ## Required refusal cases
42: 
43: - Unpinned or mismatched original/manifest bytes, incomplete or duplicate audit
44:   entries, signed-row attribution changes, or any other transformed field.
45: - Dropped, added, reordered, or mutated source rows in the migration prefix;
46:   arbitrary extra user rows cannot be treated as protocol metadata.
47: - A broken original tick predecessor, cursor continuity, or window commitment.
48: - An unexplained historical window mismatch, changed tick evidence, or any later
49:   tick-chain mismatch. Verification must inspect the whole stream rather than
50:   treating the first ten diagnostic locations as the complete failure set.
51: - Missing or forward-referencing window cursors, invalid Arrival chain, wrong S,
52:   or input mutation during verification.
53: 
54: Only window-hash differences on preserved historical ticks may be explained by
55: the checked mapping. Predecessor and cursor continuity are not waived. Results
56: contain counts, hashes and coordinates rather than payloads or private keys.
57: 
58: ## Exact verification claims
59: 
60: The receipt labels trust pins as caller-supplied and does not claim the tool can
61: prove their independent origin. It distinguishes structural tick-chain checking
62: from signature authentication: registry-forming Arrival signatures are checked,
63: but ordinary record-envelope and inner FACT/TICK signatures are not authenticated
64: by this verifier. The separate real-copy tick attestation remains evidence for
65: the new signed boundary tick.
66: 
67: Historical unchained ticks are counted explicitly. They are permitted only
68: before the chained era and only without predecessor or cursor claims. A return
69: to an unchained tick after chaining starts refuses; later post-migration ticks
70: must be chained. No unchained row can erase a previously established continuity
71: requirement.

250: ## Publication and writer handoff
251: 
252: There are two distinct publication points:
253: 
254: 1. Migration and adoption make durable records in `ARRIVAL_STORE`, while only
255:    `CANDIDATE_VERTEX` names them. This is pre-live evidence and does not change
256:    `LIVE_VERTEX`.
257: 2. The final descriptor replacement makes the candidate declaration live. It
258:    must be an atomic same-directory replace that first confirms the live vertex
259:    still equals the pinned original bytes and then fsyncs the directory. Save
260:    a separately named prepublication backup under `ARCHIVE_ORIGINAL` (for
261:    example `project.vertex.prepublish`); do not overwrite it on a retry.
262: 
263: Prepare the final replacement file in `LIVE_VERTEX.parent` so the final
264: `os.replace` is same-directory. Under the already-held quiescence lock, a small
265: one-shot publication procedure must do the following in order:
266: 
267: 1. Read `LIVE_VERTEX` and `CANDIDATE_VERTEX`; require their SHA-256 values to
268:    equal the recorded original and candidate pins. Re-parse the candidate and
269:    require its descriptor to name the final absolute `ARRIVAL_STORE`, the
270:    pinned lineage, and authority role.
271: 2. Create and `fsync` the separately named prepublication backup exclusively,
272:    or, on a retry, require its existing hash to equal the original pin. Then
273:    create `LIVE_VERTEX.parent / ".<name>.arrival-cutover"` exclusively with
274:    the candidate bytes, `fsync` it, reread/hash it, and require the candidate
275:    hash again. An existing stage path refuses.
276: 3. Reread and hash `LIVE_VERTEX` one final time. If it no longer equals the
277:    original pin, remove only the uninstalled stage file and refuse.
278: 4. Call `os.replace(stage_path, LIVE_VERTEX)`, then open and `fsync` the parent
279:    directory. Reread/hash the new live vertex and require the candidate pin.
280: 
281: This is an atomic same-directory replacement under quiescence; it is not an
282: atomic compare-and-swap. Quiescence is the guarantee against a concurrent
283: editor; the two live-byte checks detect drift observed before replacement. The
284: procedure's receipt binds the old and candidate hashes, A, stable Arrival path,
285: backup hash, and replacement result.
286: 
287: ```python
288: def publish_vertex(live: Path, candidate: Path, backup: Path) -> tuple[str, str]:
289:     old = live.read_bytes()
290:     replacement = candidate.read_bytes()
291:     old_hash = hashlib.sha256(old).hexdigest()
292:     replacement_hash = hashlib.sha256(replacement).hexdigest()
293:     if old_hash != ORIGINAL_SHA256 or replacement_hash != CANDIDATE_SHA256:
294:         raise RuntimeError("cutover input hash changed")
295:     stage = live.parent / f".{live.name}.arrival-cutover"
296:     if backup.exists():
297:         if hashlib.sha256(backup.read_bytes()).hexdigest() != old_hash:
298:             raise RuntimeError("existing prepublication backup hash mismatch")
299:     else:
300:         _write_exclusive_and_fsync(backup, old)
301:     if hashlib.sha256(backup.read_bytes()).hexdigest() != old_hash:
302:         raise RuntimeError("original backup hash mismatch")
303:     if stage.exists():
304:         raise RuntimeError("exclusive live-directory stage already exists")
305:     _write_exclusive_and_fsync(stage, replacement)
306:     if hashlib.sha256(stage.read_bytes()).hexdigest() != replacement_hash:
307:         raise RuntimeError("staged candidate changed")
308:     if live.read_bytes() != old:  # quiescence check immediately before replace
309:         stage.unlink(missing_ok=True)
310:         raise RuntimeError("live vertex changed during publication")
311:     os.replace(stage, live)
312:     _fsync_directory(live.parent)
313:     if hashlib.sha256(live.read_bytes()).hexdigest() != replacement_hash:
314:         raise RuntimeError("published vertex hash mismatch")
315:     return old_hash, replacement_hash
316: ```
317: 
318: This is illustrative algorithm text, not an implemented cutover command.
319: `_write_exclusive_and_fsync` opens with exclusive creation, writes, flushes and
320: calls `os.fsync`; `_fsync_directory` opens the directory read-only and calls
321: `os.fsync`. Before use, the cutover transcript must add the step-1 descriptor
322: reparse and receipt serialization, then record a disposable-directory rehearsal
323: of the exact fragment.
324: 
325: Keep writers stopped through the replacement, reopen the descriptor only with
326: the SDK operations above, and repeat `verify_target`, `sync_target`,
327: `inspect_declaration`, `read_summary`, and `read_state` against `LIVE_VERTEX`.
328: Run `preview_emission` for the separately approved first business payload with
329: the production mapped provider against `LIVE_VERTEX`; require
330: `read_path == "arrival"` and `captured_head == A`, with no append. Record the
331: explicit `CUTOVER_BINDINGS` and `CUTOVER_STATE` paths used by that process and
332: confirm no binding or head-journal material was resolved from the candidate or
333: from a default-state root.
334: Only then hand each inventoried writer to a descriptor-first SDK integration:
335: 
336: - interactive/API fact writer: `preview_emission` then `emit_fact` with the
337:   production mapped provider;
338: - batch ingest: `emit_batch` with the same mapped provider and captured
339:   result/commit evidence;
340: - declaration management: `edit_declaration` and its recovery API;
341: - source/cadence jobs: `run_sources` with the production collector factory,
342:   dispatcher, observer, mapped provider, and `CUTOVER_STATE` environment;
343: - CLI-only workflows: remain disabled until they have a reviewed adapter to
344:   those descriptor-first SDK calls.
345: 
346: The handoff receipt names every enabled writer and its SDK version. No legacy
347: or raw JSONL/SQLite writer is re-enabled.
348: 
349: ## Failure disposition and rollback
350: 
351: Before final descriptor replacement, any failed migration, provenance, report,
352: adoption, projection or read gate leaves `LIVE_VERTEX` and `LIVE_SOURCE`
353: unchanged. Preserve the candidate Arrival file, report, intent and receipts as
354: evidence; do not truncate, delete, reuse, or append a second anchor. Resume is
355: allowed only through the documented migration/adoption recovery paths when
356: their exact input pins still hold.
357: 
358: After descriptor replacement but before any post-A business write, legacy
359: routing may be restored only when all of the following are recorded while
360: writers and source jobs are stopped:
361: 
362: - `verify_target(LIVE_VERTEX)` captures a head exactly equal to A in lineage,
363:   ordinal, and record hash; the Arrival file SHA-256 equals the recorded
364:   post-adoption A pin.
365: - `LIVE_VERTEX` SHA-256 equals the candidate pin, and `LIVE_SOURCE` SHA-256
366:   still equals its original snapshot pin.
367: - The exclusive same-directory restore stage is written from the separately
368:   named prepublication backup, fsynced and rehashed against the original vertex
369:   pin before `os.replace`; after replace, fsync the directory and require the
370:   live vertex hash to equal that original pin.
371: 
372: If any condition fails, restoration is forbidden and disposition is forward
373: only. When it succeeds, retain the candidate Arrival lineage and adoption
374: anchor; this is a service-routing reversal, never a history deletion.
375: 
376: After the first post-A write, do not restore the legacy vertex as a writable
377: fallback. It would conceal new authoritative facts. Stop writers, retain the
378: Arrival descriptor, head journal, mapped-binding root, original archive and
379: exact export, then recover or correct forward on the Arrival lineage. If
380: service must be limited during that work, use descriptor-first read-only SDK
381: access at the recorded head; do not drop or replay new facts into the legacy
382: source.
