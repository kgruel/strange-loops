283:             raise AggregateCycle(f"composition cycle: {chain}")

121: def descriptor_for(
122:     ast: VertexFile, vertex_path: Path | None = None
123: ) -> StoreDescriptor | None:
124:     """Which adapter opens this vertex's store, or None if no adapter does.
125: 
126:     Two answers, and only the first describes a ledger:
127: 
128:     **Explicit** — the vertex declared ``backend=``. The declared name wins,
129:     and there is NO cross-check against the location's suffix. A
130:     ``backend="file"`` on a ``.duckdb``-named path is legal, because under §02
131:     the suffix carries no meaning once the backend is declared; adding a "do
132:     they agree?" check would re-admit inference through the back door on the
133:     very change that removes it. An unregistered name is not this function's
134:     to refuse — see :meth:`BackendRegistry.open`.
135: 
136:     **No descriptor** — a bare legacy store, an aggregate (``combine``), or a
137:     loops-only vertex has no explicit adapter declaration. Suffix inference
138:     ends here: a ``.arrival`` suffix does not synthesize ``backend="file"``.
139: 
140:     ``lineage`` and ``role`` are declaration claims, not grants. The registry
141:     checks them against the presented genesis and backend capabilities, then
142:     keeps enforcing them on the returned custody handle.
143:     """
144:     if ast.store is None or ast.store_backend is None:
145:         return None
146: 
147:     declaration = ast.store_backend
148:     if declaration.name == FILE_BACKEND:
149:         # This is the built-in file adapter's interpretation of a location:
150:         # relative paths are relative to the declaring vertex, never cwd.
151:         from .residence import canonical_store_path
152: 
153:         location = str(canonical_store_path(ast.store, vertex_path))
154:     else:
155:         # No pathlib round-trip for another adapter. A DSN or service URL is
156:         # opaque, including repeated slashes and query text.
157:         if ast.store_location is None:
158:             raise ValueError(
159:                 f"non-file backend {declaration.name!r} has no opaque store "
160:                 "location; construct it by parsing a descriptor declaration"
161:             )
162:         location = ast.store_location
163:

165:     return StoreDescriptor(
166:         backend=declaration.name,
167:         location=location,
168:         lineage=declaration.lineage,
169:         role=role,

377:         self._search_maintenance_openers: dict[str, SearchMaintenanceOpener] = {}
378: 
379:     @classmethod
380:     def with_builtin_backends(cls) -> BackendRegistry:
381:         """A registry holding the adapters that ship in this package."""
382:         registry = cls()
383:         registry.register(
384:             FILE_BACKEND,
385:             _open_file_backend,
386:             binding_provider=_file_binding,
387:             maintenance_opener=_open_file_maintenance,
388:             search_maintenance_opener=_open_file_search_maintenance,
389:         )
390:         return registry
391: 
392:     def register(
393:         self,
394:         name: str,
395:         opener: BackendOpener,
396:         *,
397:         binding_provider: BindingProvider | None = None,
398:         maintenance_opener: MaintenanceOpener | None = None,
399:         search_maintenance_opener: SearchMaintenanceOpener | None = None,
400:     ) -> None:
401:         """Bind a backend name to its opener.
402: 
403:         Re-registering a name refuses. A silent overwrite would let import
404:         order decide which adapter answers to "file", and the failure would
405:         surface as a store opening through the wrong codec rather than as an
406:         error at the point of the mistake. ``binding_provider`` and the
407:         private ``maintenance_opener`` are separate metadata so the opener's
408:         tuple remains stable. Binding is checked before the opener runs, and
409:         maintenance remains reachable only through the attested coordinator.
410:         """
411:         if name in self._openers:
412:             raise ValueError(f"backend {name!r} is already registered")
413:         self._openers[name] = opener
414:         self._bindings[name] = (
415:             binding_provider
416:             if binding_provider is not None
417:             else lambda descriptor, backend=name: opaque_binding(
418:                 backend, descriptor.location
419:             )
420:         )
421:         if maintenance_opener is not None:
422:             self._maintenance_openers[name] = maintenance_opener
423:         if search_maintenance_opener is not None:
424:             self._search_maintenance_openers[name] = search_maintenance_opener
425: 
426:     def registered(self) -> tuple[str, ...]:
427:         """The backend names this registry answers to, sorted."""
428:         return tuple(sorted(self._openers))
429: 
430:     def _maintenance_for(self, descriptor: StoreDescriptor) -> ProjectionMaintenance:
431:         """Private adapter factory used only after an attested registry open.
432: 
433:         The public maintenance entry point is
434:         :func:`engine.arrival_maintenance.sync_projection`. Keeping this
435:         factory private prevents a raw derived-state authority from becoming a
436:         second registry open path that omits head attestation.
437:         """
438:         opener = self._maintenance_openers.get(descriptor.backend)
439:         if opener is None:
440:             raise NotSupported(
441:                 f"backend {descriptor.backend!r} does not provide projection maintenance"
442:             )
443:         return opener(descriptor)
444: 
445:     def _search_maintenance_for(self, descriptor: StoreDescriptor) -> SearchIndexMaintenance:
446:         """Private FTS provider reached only by the attested coordinator."""
447:         opener = self._search_maintenance_openers.get(descriptor.backend)
448:         if opener is None:
449:             raise NotSupported(
450:                 f"backend {descriptor.backend!r} does not provide search maintenance"
451:             )
452:         return opener(descriptor)
453: 
454:     def open(
455:         self, descriptor: StoreDescriptor
456:     ) -> tuple[ArrivalLedger, ArrivalQuery]:
457:         """The custody half and the reads half for this descriptor.
458: 
459:         Refuses an unregistered name with
460:         :class:`~engine.arrival_contract.UnknownBackend`, naming what IS
461:         registered. Refusing is the whole point: §02 forbids inferring a
462:         backend from a suffix, so an unregistered name has no fallback to
463:         degrade into — and a registry that guessed would reintroduce
464:         inference at the one seam built to remove it.
465: 
466:         **The custody half comes back wrapped, always, with no way to ask for
467:         it unwrapped** (slice 3 §D.1). :class:`~engine.arrival_head_seam.AttestedLedger`
468:         compares the head this store presents against the one this machine last
469:         accepted, and journals every commit made through it. A configuration
470:         switch for a safety property is a shape the arrival vocabulary
471:         ratchet's own denylist rejects: custody is structural rather than
472:         configured, so there is no flag here and no second opener that skips
473:         the comparison.
474: 
475:         The comparison happens at construction, which means **this method
476:         raises what the comparison raises**: a rollback, a fork, a rewrite, a
477:         replacement or a lost store refuses the open rather than being reported
478:         afterwards by a caller who might not ask. This is also why the seam
479:         sits here rather than deeper — the opener above performs no recovery,
480:         so the comparison sees the store as it is rather than as a catch-up
481:         just repaired it.
482:         """
483:         ledger, query, binding = self._open_components(descriptor)
484:         try:
485:             attested = AttestedLedger(
486:                 ledger, location=descriptor.location, query=query, binding=binding
487:             )
488:         except BaseException:
489:             _close_handle(query)
490:             _close_handle(ledger)
491:             raise
492:         return attested, query
493: 
494: 
495:     def _open_components(
496:         self, descriptor: StoreDescriptor
497:     ) -> tuple[ArrivalLedger, ArrivalQuery, BindingIdentity]:
498:         """Private resources for attested open or the bounded restore procedure.
499: 
500:         Never returned to consumers. Both callers own comparison, witnessing,
501:         and closure; restoration exposes only its completed receipt.
502:         """
503:         opener = self._openers.get(descriptor.backend)
504:         if opener is None:
505:             known = ", ".join(self.registered()) or "(none)"
506:             raise UnknownBackend(
507:                 f"no adapter registered for backend {descriptor.backend!r} "
508:                 f"(registered: {known})"
509:             )
510:         provider = self._bindings[descriptor.backend]
511:         try:
512:             binding = provider(descriptor)
513:         except BindingProbeError as exc:
514:             raise BindingProbeUnanswered(
515:                 f"{descriptor.location} exists and storage would not establish "
516:                 f"its file identity ({exc}); nothing has been accepted or "
517:                 "written — repair the location or retry"
518:             ) from exc
519:         _validate_binding(descriptor, binding)
520:         ledger, query = opener(descriptor)
521:         try:
522:             capabilities = ledger.capabilities()
523:             if (
524:                 descriptor.role is not None
525:                 and descriptor.role not in capabilities.profiles
526:             ):
527:                 offered = ", ".join(
528:                     sorted(profile.value for profile in capabilities.profiles)
529:                 ) or "(none)"
530:                 raise NotSupported(
531:                     f"backend {descriptor.backend!r} cannot open location "
532:                     f"{descriptor.location!r} as {descriptor.role.value}: "
533:                     f"capabilities advertise {offered}"
534:                 )
535: 
536:             # Check a presented genesis before AttestedLedger can turn a
537:             # first-contact head into a bootstrap journal entry. An absent or
538:             # not-yet-minted ledger has no genesis to compare; its later mint
539:             # remains pinned by _DescriptorLedger.
540:             if descriptor.lineage is not None:
541:                 try:
542:                     presented = ledger.verify(Open())
543:                 except Exception:  # backend-specific absence/corruption
544:                     presented = None
545:                 if (
546:                     presented is not None
547:                     and presented.lineage != descriptor.lineage
548:                 ):
549:                     raise NotAuthority(
550:                         f"descriptor names lineage {descriptor.lineage}, but "
551:                         f"the genesis at {descriptor.location!r} opens "
552:                         f"{presented.lineage}; the declaration is a check, "
553:                         "not a replacement for genesis"
554:                     )
555: 
556:             return _DescriptorLedger(ledger, descriptor), query, binding
557:         except BaseException:
558:             _close_handle(query)
559:             _close_handle(ledger)
560:             raise
561: 
562:     def restore_forward(
563:         self,
564:         source: StoreDescriptor,
565:         receiver: StoreDescriptor,
566:         *,
567:         through: Head | None = None,
568:     ) -> RestoreForwardResult:
569:         """Restore an existing exact prefix without granting an open handle."""
570:         from .arrival_restore import _restore_forward
571: 
572:         return _restore_forward(self, source, receiver, through=through)
