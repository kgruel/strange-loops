510: def _check_inventory_equality(
511:     source_inv: SourceInventory,
512:     target_records: list[dict[str, Any]],
513:     exceptions: TransformExceptions,
514: ) -> None:
515:     """Assert inventory equality per §I.3 between source inventory and target projection.
516: 
517:     Compares per-kind row counts, tick count, and observer census (accounting for
518:     dropped units). Record counts are deliberately NOT compared (§I.3).
519: 
520:     Partial-batch drops are refused upstream by the transformer, so its dropped-units
521:     accounting covers whole-unit drops only.
522: 
523:     Raises:
524:         PublishPreconditionRefused: If any inventory metric diverges.
525:     """
526:     target_per_kind: dict[str, int] = defaultdict(int)
527:     target_observer_census: dict[str, int] = defaultdict(int)
528:     target_tick_count = 0

530:     for rec in target_records:
531:         k = rec.get("k")
532:         if k in {GENESIS_KIND, KEY_INTRODUCTION_KIND}:
533:             continue
534:         if k not in {FACT_KIND, BATCH_KIND, TICK_KIND}:
535:             continue
536: 
537:         expanded = rows_of_body(k, rec["body"])
538:         for row_k, row_tuple in expanded:
539:             if row_k == FACT_KIND:
540:                 fact_id, kind, ts, observer, origin, payload, sig = row_tuple
541:                 target_per_kind[kind] += 1
542:                 target_observer_census[observer] += 1
543:             elif row_k == TICK_KIND:
544:                 target_tick_count += 1
545: 
546:     expected_per_kind = dict(source_inv.per_kind_counts)
547:     expected_observer_census = dict(source_inv.observer_census)

568:         )
569:     if dict(target_observer_census) != expected_observer_census:
570:         mismatches.append(
571:             "observer census mismatch: "
572:             f"source/expected={expected_observer_census} != target={dict(target_observer_census)}"
573:         )
574:     if target_tick_count != expected_tick_count:
575:         mismatches.append(
576:             "tick count mismatch: "
577:             f"source/expected={expected_tick_count} != target={target_tick_count}"
578:         )
579: 
580:     if mismatches:
581:         raise PublishPreconditionRefused(
582:             f"Inventory equality check failed: {'; '.join(mismatches)}. "
583:             "Advisory: source and target contents diverge in census projection.",
584:             condition="inventory_equality",
585:         )

588: def verify_migration_report(
589:     report_path: Path | str,
590:     public_key: str,
591:     *,
592:     verify: Verify | None = None,
593:     target_path: Path | str | None = None,
594: ) -> bool:
595:     """Verify a migration report's signature and re-check target head claim (§I.1).
596: 
597:     Args:
598:         report_path: Path to the signed <lineage>.migration-report.json file.
599:         public_key: Public Ed25519 key of the custodian (base64 string).
600:         verify: Injected Verify function (key_b64, sig, digest) -> bool.
601:         target_path: Optional explicit path to the target arrival store.
602:             If omitted, defaults to `<store_dir>/<lineage>.arrival` beside the report.
603: 
604:     Returns:
605:         True if the signature is valid and the target head claim matches live target.
606: 
607:     Raises:
608:         ReportMalformedRefused: If report is unreadable, not JSON, or carries unknown keys.
609:         ReportBadSignatureRefused: If signature fails verification.
610:         ReportMissingTargetRefused: If target store file does not exist.
611:         ReportTargetUnopenableRefused: If target store cannot be opened or read.
612:         ReportHeadMismatchRefused: If target store head does not match report claim.
613:     """
614:     r_path = Path(report_path).resolve()
615:     if not r_path.exists():
616:         raise ReportMissingTargetRefused(f"Migration report file does not exist: {r_path}")
617: 
618:     try:
619:         report_doc = json.loads(r_path.read_text(encoding="utf-8"))
620:     except Exception as exc:
621:         raise ReportMalformedRefused(f"Migration report is not valid JSON: {exc}") from exc
622: 
623:     if not isinstance(report_doc, dict):
624:         raise ReportMalformedRefused("Migration report top-level document must be a JSON object")
625: 
626:     unknown_keys = set(report_doc.keys()) - ALLOWED_REPORT_KEYS
627:     if unknown_keys:
628:         raise ReportMalformedRefused(
629:             f"Migration report carries unknown top-level keys: {sorted(unknown_keys)}"
630:         )
631: 
632:     body = report_doc.get("body")
633:     signature = report_doc.get("signature")
634:     signer = report_doc.get("signer")
635:     if not isinstance(body, dict) or not isinstance(signature, str) or not isinstance(signer, str):
636:         raise ReportMalformedRefused(
637:             "Migration report missing required top-level fields (body, signature, signer)"
638:         )
639: 
640:     if verify is None:
641:         raise ValueError(
642:             "verify function must be provided (verify is injected per Rule 4/11)"
643:         )
644: 
645:     # Verify signature over canonical JSON bytes of report minus signature field
646:     signed_dict = {k: v for k, v in report_doc.items() if k != "signature"}
647:     canonical_bytes = _canonical_bytes(signed_dict)
648:     digest = hashlib.sha256(canonical_bytes).hexdigest()
649: 
650:     if not verify(public_key, signature, digest):
651:         raise ReportBadSignatureRefused(
652:             f"Migration report signature failed verification for custodian key {public_key!r}"
653:         )
654: 
655:     # Re-check target head claim against live target
656:     target_head_claim = body.get("target_head")
657:     if not isinstance(target_head_claim, dict):
658:         raise ReportMalformedRefused("Migration report body missing 'target_head' dictionary")
659: 
660:     lineage = target_head_claim.get("lineage")
661:     if not lineage:
662:         raise ReportMalformedRefused("Migration report 'target_head' missing 'lineage'")
663: 
664:     if target_path is None:
665:         t_path = r_path.parent / f"{lineage}.arrival"
666:     else:
667:         t_path = Path(target_path).resolve()
668: 
669:     if not t_path.exists():
670:         raise ReportMissingTargetRefused(f"Target arrival store does not exist: {t_path}")
671: 
672:     registry = BackendRegistry.with_builtin_backends()
673:     descriptor = StoreDescriptor(
674:         backend="file",
675:         location=str(t_path),
676:         lineage=str(lineage),
677:         role=Profile.AUTHORITY,
678:     )
679:     try:
680:         ledger, _ = registry.open(descriptor)
681:         live_head = ledger.head()
682:     except NotAuthority as exc:
683:         raise ReportHeadMismatchRefused(
684:             f"Target store lineage does not match report claim: {exc}"
685:         ) from exc
686:     except (StoreLost, ArrivalTornTail, ArrivalCorrupt, GenesisRefused, OSError) as exc:
687:         raise ReportTargetUnopenableRefused(
688:             f"Cannot open or read head from target arrival store {t_path}: {exc}",
689:             target_path=str(t_path),
690:             cause=exc,
691:         ) from exc
692: 
693:     if (
694:         live_head.lineage != target_head_claim.get("lineage")
695:         or live_head.ordinal != target_head_claim.get("ordinal")
696:         or live_head.record_hash != target_head_claim.get("record_hash")
697:     ):
698:         raise ReportHeadMismatchRefused(
699:             f"Live target head ({live_head.lineage}, {live_head.ordinal}, {live_head.record_hash}) "
700:             f"does not match report claim ({target_head_claim.get('lineage')}, "
701:             f"{target_head_claim.get('ordinal')}, {target_head_claim.get('record_hash')})"
702:         )
703: 
704:     return True

707: def run_migration(
708:     source_path: Path | str,
709:     vertex_path: Path | str,
710:     *,
711:     store_dir: Path | str,
712:     signer: Signer,
713:     transform_rule: Transform | None = None,
714:     resume_target: Path | str | None = None,
715:     tool_version: str = "0.1.0",
716: ) -> MigrationOutcome:
717:     """Run full migration pipeline from legacy store to Arrival store.
718: 
719:     Stages:
720:     1. Inventory pass: validates legacy store format and defects before any target touch.
721:     2. Deterministic transform: transforms source units into record drafts.
722:     3. Journal pre-flight: probes state root and heads/ directory.
723:     4. Target open & mint (fresh) / scan & diff (resume).
724:     5. Append loop: commits drafts through AttestedLedger.
725:     6. Verification: verify(Full) + equivalence re-run.
726:     7. Migration report: builds and signs migration report with custodian key.
727:     8. Preconditions check: asserts all 5 publish preconditions hold.
728:     9. Atomic publish: surgical .vertex store clause edit with verification-by-re-parse.
729: 
730:     Args:
731:         source_path: Path to legacy .jsonl or .sqlite store.
732:         vertex_path: Path to target .vertex file.
733:         store_dir: Directory where staging .arrival store and report will be created.
734:         signer: Injected Signer for custodian signatures.
735:         transform_rule: Deterministic fact transform rule (defaults to identity()).
736:         resume_target: Explicit staging target path when resuming an interrupted migration.
737:         tool_version: Tool version string for report metadata.
738: 
739:     Returns:
740:         MigrationOutcome with target path, lineage, head, report path, exceptions.
741: 
742:     Raises:
743:         MigrationRefused: On any source defect, resume mismatch, torn tail,
744:             preflight failure, or precondition violation.
745:         FileNotFoundError: If source or vertex file does not exist.
746:     """
747:     src_path = Path(source_path).resolve()
748:     v_path = Path(vertex_path).resolve()
749:     s_dir = Path(store_dir).resolve()
750: 
751:     if not src_path.exists():
752:         raise FileNotFoundError(f"Legacy source not found: {src_path}")
753:     if not v_path.exists():
754:         raise FileNotFoundError(f"Vertex file not found: {v_path}")
755: 
756:     original_vertex_bytes = v_path.read_bytes()
757:     vertex_ast = parse_vertex(original_vertex_bytes.decode("utf-8"), v_path)
758:     already_published = False
759:     if vertex_ast.store_backend is not None:
760:         existing_descriptor = descriptor_for(vertex_ast, v_path)
761:         resume_path = None if resume_target is None else Path(resume_target).resolve()
762:         if (
763:             resume_path is not None
764:             and existing_descriptor is not None
765:             and existing_descriptor.backend == "file"
766:             and existing_descriptor.role is Profile.AUTHORITY
767:             and existing_descriptor.lineage == resume_path.stem
768:             and Path(existing_descriptor.location).resolve() == resume_path
769:         ):
770:             # A process may have stopped after the atomic descriptor replace
771:             # but before returning its outcome. Re-verification of that exact
772:             # published target is safe; no descriptor rewrite is needed.
773:             already_published = True
774:         else:
775:             raise PublishPreconditionRefused(
776:                 "Migration requires a legacy descriptor snapshot; refusing to replace an "
777:                 "existing explicit backend/lineage/role declaration.",
778:                 condition="vertex_already_descriptor_backed",
779:             )
780: 
781:     # Stage 1: Inventory pass (refusals fire here, before any target creation)
782:     try:
783:         inv = inventory(src_path)
784:     except (sqlite3.OperationalError, sqlite3.DatabaseError, OSError, UnicodeDecodeError) as exc:
785:         raise LegacyStorageRefused(
786:             f"Legacy storage read failed for source {src_path!r}: {exc}. "
787:             "Advisory: inspect legacy store database schema or file integrity.",
788:             source=str(src_path),
789:         ) from exc
790: 
791:     # Stage 2: Transform pass
792:     t_rule = transform_rule if transform_rule is not None else identity()
793: 
794:     try:
795:         transform_result = transform(
796:             source=src_path,
797:             vertex=vertex_ast,
798:             rule=t_rule,
799:             signer=signer,
800:         )
801:     except (sqlite3.OperationalError, sqlite3.DatabaseError, OSError, UnicodeDecodeError) as exc:
802:         raise LegacyStorageRefused(
803:             f"Legacy storage read failed during transform for source {src_path!r}: {exc}. "
804:             "Advisory: inspect legacy store database schema or file integrity.",
805:             source=str(src_path),
806:         ) from exc
807: 
808:     drafts = transform_result.drafts
809:     exceptions = transform_result.exceptions
810:     genesis_req = transform_result.genesis
811: 
812:     # Stage 3: Journal pre-flight (§D.3)
813:     preflight_journal()
814: 
815:     # Stage 4 & 5: Target open & Append loop
816:     registry = BackendRegistry.with_builtin_backends()
817: 
818:     if resume_target is None:
819:         # Fresh migration run (§D.2: Lineage-named staging path)
820:         lineage = mint_lineage()
821:         target_path = s_dir / f"{lineage}.arrival"
822:         target_path.parent.mkdir(parents=True, exist_ok=True)
823: 
824:         descriptor = StoreDescriptor(
825:             backend="file",
826:             location=str(target_path),
827:             lineage=lineage,
828:             role=Profile.AUTHORITY,
829:         )
830:         ledger, query = registry.open(descriptor)
831: 
832:         # First touch is mint() through the wrapper (§E)
833:         mint_options: dict[str, Any] = {
834:             "observer": genesis_req.custodian,
835:             "signer": signer,
836:             "key": genesis_req.key,
837:             "lineage": lineage,
838:             "at": 0.0,
839:         }
840:         current_head = ledger.mint(mint_options)
841: 
842:         # Append drafts via ordinary contract append (F1)
843:         if drafts:
844:             current_head = _append_drafts(ledger, current_head, drafts)
845: 
846:     else:
847:         # Resume run (§D.1: Resume point derived from target head)
848:         target_path = Path(resume_target).resolve()
849:         if not target_path.exists():
850:             raise FileNotFoundError(f"Resume target does not exist: {target_path}")
851: 
852:         expected_lineage = target_path.stem
853:         descriptor = StoreDescriptor(
854:             backend="file",
855:             location=str(target_path),
856:             lineage=expected_lineage,
857:             role=Profile.AUTHORITY,
858:         )
859: 
860:         try:
861:             ledger, query = registry.open(descriptor)
862:             current_head = ledger.head()
863:         except NotAuthority as exc:
864:             raise TargetMismatchOnResumeRefused(
865:                 f"Resume target {target_path} does not match its filename lineage: {exc}",
866:                 target_path=str(target_path),
867:                 ordinal=0,
868:             ) from exc
869:         except (ArrivalTornTail, StoreLost, ArrivalCorrupt, GenesisRefused, OSError) as exc:
870:             cause = exc.__cause__
871:             if isinstance(exc, ArrivalTornTail) or isinstance(cause, ArrivalTornTail):
872:                 raise TornTailRefused(
873:                     f"Resume target {target_path} ends mid-record: {exc}. "
874:                     "Advisory: out-of-band truncate the torn tail or start a fresh migration.",
875:                     target_path=str(target_path),
876:                 ) from exc
877:             raise TargetUnopenable(
878:                 f"Resume target {target_path} cannot be opened: {exc}. "
879:                 "Advisory: inspect target store or start a fresh migration.",
880:                 target_path=str(target_path),
881:                 cause=exc,
882:             ) from exc
883: 
884:         lineage = current_head.lineage
885: 
886:         expected_filename = f"{lineage}.arrival"
887:         if target_path.name != expected_filename:
888:             raise TargetMismatchOnResumeRefused(
889:                 f"Resume target filename {target_path.name!r} does not match "
890:                 f"genesis lineage {lineage!r} (expected {expected_filename!r}). "
891:                 "Advisory: target arrival stores must be lineage-named "
892:                 "(<lineage>.arrival per M-2); rename target file or start a fresh migration.",
893:                 target_path=str(target_path),
894:             )
895: 
896:         # Scan existing records through current_head and diff prefix against expected drafts
897:         try:
898:             scanned_records = list(ledger.scan(through=current_head))
899:         except (ArrivalTornTail, StoreLost, ArrivalCorrupt, HeadMismatch, OSError) as exc:
900:             cause = exc.__cause__
901:             if isinstance(exc, ArrivalTornTail) or isinstance(cause, ArrivalTornTail):
902:                 raise TornTailRefused(
903:                     f"Resume target {target_path} ends mid-record during scan: {exc}. "
904:                     "Advisory: out-of-band truncate the torn tail or start a fresh migration.",
905:                     target_path=str(target_path),
906:                 ) from exc
907:             raise TargetUnopenable(
908:                 f"Resume target {target_path} failed scan at {current_head}: {exc}. "
909:                 "Advisory: inspect target store or start a fresh migration.",
910:                 target_path=str(target_path),
911:                 cause=exc,
912:             ) from exc
913: 
914:         if len(scanned_records) != current_head.ordinal + 1:
915:             raise TargetMismatchOnResumeRefused(
916:                 f"Target at {target_path} has {len(scanned_records)} scanned records, "
917:                 f"expected {current_head.ordinal + 1}. "
918:                 "Advisory: target store density or scan is inconsistent; start a fresh migration.",
919:                 target_path=str(target_path),
920:             )
921: 
922:         # Check genesis at ordinal 0 against genesis_req (F3)
923:         genesis_rec = scanned_records[0]
924:         if (
925:             genesis_rec.get("k") != GENESIS_KIND
926:             or genesis_rec.get("body", {}).get("lineage") != lineage
927:         ):
928:             raise TargetMismatchOnResumeRefused(
929:                 f"Target at {target_path} genesis at ordinal 0 does not match lineage {lineage}. "
930:                 "Advisory: start a fresh migration.",
931:                 target_path=str(target_path),
932:                 ordinal=0,
933:             )
934: 
935:         if genesis_rec.get("observer") != genesis_req.custodian:
936:             raise TargetMismatchOnResumeRefused(
937:                 f"Target at {target_path} genesis custodian {genesis_rec.get('observer')!r} "
938:                 f"does not match expected custodian {genesis_req.custodian!r}. "
939:                 "Advisory: start a fresh migration.",
940:                 target_path=str(target_path),
941:                 ordinal=0,
942:             )
943: 
944:         if genesis_rec.get("body", {}).get("key") != genesis_req.key:
945:             actual_key = genesis_rec.get("body", {}).get("key")
946:             raise TargetMismatchOnResumeRefused(
947:                 f"Target at {target_path} genesis public key {actual_key!r} "
948:                 f"does not match expected custodian public key {genesis_req.key!r}. "
949:                 "Advisory: start a fresh migration.",
950:                 target_path=str(target_path),
951:                 ordinal=0,
952:             )
953: 
954:         # Check ordinals 1..current_head.ordinal against expected drafts
955:         if current_head.ordinal > len(drafts):
956:             raise TargetMismatchOnResumeRefused(
957:                 f"Target at {target_path} has advanced beyond expected drafts: "
958:                 f"target ordinal={current_head.ordinal} > expected drafts={len(drafts)}. "
959:                 "Advisory: start a fresh migration.",
960:                 target_path=str(target_path),
961:                 ordinal=current_head.ordinal,
962:             )
963: 
964:         for ord_idx in range(1, current_head.ordinal + 1):
965:             target_rec = scanned_records[ord_idx]
966:             exp_draft = drafts[ord_idx - 1]
967: 
968:             if (
969:                 target_rec.get("k") != exp_draft.kind
970:                 or target_rec.get("observer") != exp_draft.observer
971:                 or target_rec.get("origin") != exp_draft.origin
972:                 or target_rec.get("body") != exp_draft.body
973:                 or target_rec.get("at") != exp_draft.authored_at
974:                 or target_rec.get("sig") != exp_draft.signature
975:             ):
976:                 raise TargetMismatchOnResumeRefused(
977:                     f"Target record at ordinal {ord_idx} diverges from "
978:                     "expected deterministic draft. "
979:                     "Advisory: target log content does not match source transform; "
980:                     "start a fresh migration.",
981:                     target_path=str(target_path),
982:                     ordinal=ord_idx,
983:                 )
984: 
985:         # Append remaining drafts beyond current_head
986:         if current_head.ordinal < len(drafts):
987:             remaining_drafts = drafts[current_head.ordinal :]
988:             current_head = _append_drafts(
989:                 ledger, current_head, remaining_drafts
990:             )
991: 
992:     # Stage 6: Verification (BOTH gates: Full verify + Equivalence re-run)
993:     verified_head = ledger.verify(Full(through=current_head))
994:     if verified_head != current_head:
995:         raise PublishPreconditionRefused(
996:             f"Target verification failed: verified head {verified_head} != "
997:             f"current head {current_head}. Advisory: internal log consistency check failed.",
998:             condition="target_verify_full",
999:         )
1000: 
1001:     # Precondition: Source quiescence check (source content hash unchanged)
1002:     post_inv = inventory(src_path)
1003:     if post_inv.content_hash != inv.content_hash:
1004:         raise SourceChangedRefused(
1005:             f"Legacy source changed during staging at {src_path}: "
1006:             f"content hash changed from {inv.content_hash} to {post_inv.content_hash} "
1007:             "(two histories, not a migration). "
1008:             "Advisory: ensure legacy store is quiesced before running migration.",
1009:             source=str(src_path),
1010:             expected_hash=inv.content_hash,
1011:             actual_hash=post_inv.content_hash,
1012:         )
1013: 
1014:     all_target_records = list(ledger.scan(through=current_head))
1015:     equiv_res = _check_equivalence(
1016:         src_path, all_target_records, t_rule
1017:     )
1018:     if not equiv_res.matched:
1019:         raise PublishPreconditionRefused(
1020:             f"Equivalence re-run failed: {'; '.join(equiv_res.mismatches)}. "
1021:             "Advisory: re-derived expected rows do not match target records.",
1022:             condition="equivalence_rerun",
1023:         )
1024: 
1025:     # Precondition: Inventory equality (§I.3)
1026:     _check_inventory_equality(inv, all_target_records, exceptions)
1027: 
1028:     # Precondition: Journal's FIRST entry for lineage is bootstrap/MINT (§E)
1029:     j_read = read_journal(lineage)
1030:     if (
1031:         not j_read.entries
1032:         or j_read.entries[0].kind != Kind.BOOTSTRAP
1033:         or j_read.entries[0].level != Level.MINT
1034:     ):
1035:         first_entry = j_read.entries[0] if j_read.entries else None
1036:         raise PublishPreconditionRefused(
1037:             f"Journal first entry for lineage {lineage} is not bootstrap/MINT "
1038:             f"(found: {first_entry}). "
1039:             "Advisory: target lineage was not minted through AttestedLedger.",
1040:             condition="journal_first_entry_mint",
1041:         )
1042: 
1043:     # Stage 7: Migration report (§I.1)
1044:     report_path = target_path.parent / f"{lineage}.migration-report.json"
1045:     report_body: dict[str, Any] = {
1046:         "source_format": inv.source_format,
1047:         "tool_version": tool_version,
1048:         "source_content_sha256": inv.source_content_sha256,
1049:         "source_file_sha256": inv.source_file_sha256,
1050:         "source_inventory": {
1051:             "total_rows": inv.total_rows,
1052:             "total_lines": inv.total_lines,
1053:             "per_kind_counts": inv.per_kind_counts,
1054:             "tick_count": inv.tick_count,
1055:             "batch_line_count": inv.batch_line_count,
1056:             "observer_census": inv.observer_census,
1057:             "id_era_census": inv.id_era_census,
1058:         },
1059:         "target_head": {
1060:             "lineage": current_head.lineage,
1061:             "ordinal": current_head.ordinal,
1062:             "record_hash": current_head.record_hash,
1063:         },
1064:         "exceptions": {
1065:             "keyless_declared_observers": list(
1066:                 exceptions.keyless_declared_observers
1067:             ),
1068:             "undeclared_row_observers": list(
1069:                 exceptions.undeclared_row_observers
1070:             ),
1071:             "dropped_units": [
1072:                 {"coordinate": u.coordinate, "kind": u.kind, "rule": u.rule}
1073:                 for u in exceptions.dropped_units
1074:             ],
1075:         },
1076:         "transform_rule": t_rule.rule,
1077:         "equivalence_diff": {
1078:             "matched": equiv_res.matched,
1079:             "source_rows": equiv_res.source_rows,
1080:             "target_rows": equiv_res.target_rows,
1081:             "mismatches": list(equiv_res.mismatches),
1082:         },
1083:         "custodian": genesis_req.custodian,
1084:     }
1085: 
1086:     report_doc = {
1087:         "body": report_body,
1088:         "signer": genesis_req.custodian,
1089:     }
1090:     canonical_bytes = _canonical_bytes(report_doc)
1091:     digest = hashlib.sha256(canonical_bytes).hexdigest()
1092:     report_signature = signer(genesis_req.custodian, digest)
1093:     if report_signature is None:
1094:         raise PublishPreconditionRefused(
1095:             f"Custodian {genesis_req.custodian!r} produced no signature for migration report. "
1096:             "Advisory: check signer configuration.",
1097:             condition="report_signature",
1098:         )
1099: 
1100:     report_document = {
1101:         **report_doc,
1102:         "signature": report_signature,
1103:     }
1104: 
1105:     # Atomic write for report temp+fsync+replace (F7(c))
1106:     report_dir = report_path.parent
1107:     temp_report = report_dir / f".tmp_{report_path.name}_{uuid4().hex}"
1108:     try:
1109:         with temp_report.open("w", encoding="utf-8") as f:
1110:             f.write(json.dumps(report_document, indent=2) + "\n")
1111:             f.flush()
1112:             os.fsync(f.fileno())
1113:         temp_report.replace(report_path)
1114:     finally:
1115:         if temp_report.exists():
1116:             with contextlib.suppress(OSError):
1117:                 temp_report.unlink()
1118: 
1119:     # Stage 9: Atomic descriptor publish (§H)
1120:     try:
1121:         rel_target = target_path.relative_to(v_path.parent)
1122:         new_store_location = (
1123:             f"./{rel_target}"
1124:             if not str(rel_target).startswith(".")
1125:             else str(rel_target)
1126:         )
1127:     except ValueError:
1128:         new_store_location = str(target_path)
1129: 
1130:     if not already_published:
1131:         edit_vertex_store_clause(
1132:             vertex_path=v_path,
1133:             target_location=new_store_location,
1134:             lineage=lineage,
1135:             backend="file",
1136:             expected_original=original_vertex_bytes,
1137:         )
1138:     elif v_path.read_bytes() != original_vertex_bytes:
1139:         raise PublishPreconditionRefused(
1140:             "Already-published descriptor changed during migration verification.",
1141:             condition="vertex_changed",
1142:         )
1143: 
1144:     return MigrationOutcome(
1145:         target_path=target_path,
1146:         lineage=lineage,
1147:         head=current_head,
1148:         report_path=report_path,
1149:         exceptions=exceptions,
1150:     )

===== libs/migrate/README.md =====
1: # `migrate` — Migration Sidecar
2: 
3: The `migrate` package transforms pre-Arrival legacy stores into a new Arrival
4: history. Legacy readers and codecs remain quarantined here: active runtime
5: packages do not import `migrate`.
6: 
7: The sidecar preserves source rows under the selected transform, verifies the
8: resulting log and row equivalence, writes a signed external report, and publishes
9: an explicit File descriptor with the minted lineage and `role="authority"`.
10: It checks that the vertex bytes captured for transformation have not changed
11: at publication entry and immediately before replacement. This is an offline operation on quiescent inputs; the check
12: is not a lock or an atomic compare-and-swap against concurrent writers.
13: 
14: **Publication does not adopt a declaration.** The descriptor can be resolved,
15: verified, and explicitly synced through the SDK. The new log still needs an
16: explicit declaration adoption before ordinary SDK reads, declaration edits, or
17: writes can proceed. Historical `_decl.*` rows are retained as migration evidence;
18: the current runtime does not infer the new store's identity from them or from
19: its locator file. The old `loops store migrate` wrapper is therefore not a
20: complete migration-to-current-runtime workflow.
21: 
22: Use isolated copies for rehearsals, with separate state/configuration and
23: credential roots. Keep the original stores and descriptors untouched. SQLite
24: copies must come from quiescent inputs or a coherent database snapshot.
25: 
26: The exported `edit_vertex_store_clause` helper now requires explicit `lineage`
27: and `expected_original` byte arguments. Direct callers must supply the exact
28: vertex snapshot used for planning; `run_migration` supplies these itself. An
29: error after replacement (for example, directory fsync failure) can leave the new
30: descriptor visible. Inspect the artifacts before retrying. An exact explicit
31: `resume_target` can re-verify an already published migration without replacing
32: the descriptor. This is not a general recovery or adoption API.
33: 
34: The migration report certifies the head at migration completion. Its current
35: verifier requires that exact live head, so verify it before a later adoption or
36: ordinary append advances the target. Structural Full verification without an
37: injected signature verifier does not establish historical authorship.
38: 
39: See the [publication report](../../docs/scratch/arrival-break/consistency-c9-migration-publication-2026-09-06.md)
40: and [adoption design](../../docs/scratch/arrival-break/consistency-c9-migration-adoption-design-2026-09-06.md).
41: The historical sidecar design is recorded in
42: [the slice 4 proposal](../../docs/scratch/arrival-break/slice4-design-proposal.md).

1: # C9 migration descriptor publication and adoption boundary
2: 
3: Status: implementation and validation complete; Fable-low review pending.
4: 
5: This slice follows the accepted, uncommitted setup/recovery work on
6: `arrival/finish` at `50e37595`. It leaves that implementation intact. Sol owns
7: sidecar publication and regressions, Luna owns the migration-to-SDK boundary
8: workload and validation, Terra owns the explicit adoption design, and root owns
9: integration, review triage, and handoff.
10: 
11: ## Finding and scope
12: 
13: The old sidecar omitted both lineage and role when publishing its descriptor.
14: More fundamentally, physical Arrival genesis does not supply a declaration
15: anchor: the current projection licenses `own_lineage` only after consuming an
16: `_decl.genesis` fact with the physical lineage as its ID. Migration preserves
17: legacy records, which do not establish that new anchor. This contradicts the
18: older slice 4 assumption that declaration genesis simply dissolves into mint.
19: 
20: The implemented slice makes descriptor publication explicit and rejects a
21: vertex changed after its transformation snapshot. It does not add an adoption
22: ceremony or silently turn the current locator file into canonical meaning.
23: The companion [adoption design](consistency-c9-migration-adoption-design-2026-09-06.md)
24: records the next append-forward operation and its identity decisions.
25: 
26: ## Implemented publication boundary
27: 
28: - `run_migration` captures and parses one exact byte snapshot before transformation.
29: - The editor requires that snapshot and the physical lineage, quotes the new
30:   location safely, and verifies the parsed backend/lineage/authority tuple while
31:   retaining all non-store AST fields.
32: - The editor compares the snapshot at entry and immediately before replacement,
33:   then fsyncs the descriptor directory after replacement.
34: - Fresh, resumed, and report-verification opens supply explicit role/lineage.
35:   Registry lineage mismatches retain migration/report-specific refusal types.
36: - Existing explicit descriptors refuse fresh migration. An explicit resume of
37:   the exact already-published authority target re-verifies the migration and
38:   skips descriptor replacement; it still checks the captured descriptor bytes.
39: 
40: The exported editor's required `lineage` and `expected_original` arguments are
41: an intentional direct-call compatibility change. `run_migration` keeps its
42: existing signature and does not create an adoption anchor.
43: 
44: ## Contract limits
45: 
46: The migration descriptor's authority role expresses operator intent and is
47: checked by the registry; it does not create observer authorization or supply a
48: historized declaration. Public target resolution, structural verification, and
49: explicit projection sync can succeed. Declaration-dependent reads and writes
50: must still refuse. Migration report verification checks its signed completion
51: head against the live target; a later adoption will advance beyond that head.
52: 
53: Input byte checks detect intervening changes at their check points. They do not
54: provide cross-process exclusion or guarantee a concurrently changing source is
55: quiescent. Publication uses the existing atomic replacement boundary plus directory fsync.
56: An error after replacement may leave the new descriptor visible and must not be
57: interpreted as an unmodified target. No new
58: recovery intent, live cutover coordinator, or arbitrary power-loss claim is
59: introduced here. Refusal after staging can leave valid target/report evidence;
60: it is not a zero-side-effects promise.
61: 
62: All exercised stores and credentials are synthetic and isolated. No user store
63: has been selected or migrated, and no original/live writer has been touched.
64: 
65: ## Validation and review
66: 
67: Baseline migration suite: 69 passed. Publication regressions: full migrate
68: 74 passed, focused sidecar 27 passed. Regression coverage includes exact
69: fresh/resumed descriptors, already-published resume without replacement, a
70: changed declaration during transformation, a change during temp-file fsync,
71: quoted/backslash location spelling, and preexisting authority/replica refusal.
72: These are in-process seam injections and seeded partial-state tests, not
73: subprocess-kill or arbitrary power-loss experiments.
74: 
75: The new repository integration test exercises both synthetic JSONL and SQLite
76: histories through the published descriptor. It keeps SDK dependencies free of
77: the quarantined migration package. Final combined migration and integration suite: **76 passed** (74 migrate +
78: 2 boundary cases); architecture **101 passed**; legacy CLI migration caller
79: compatibility **9 passed**. Scoped Ruff and whitespace checks pass. All commands
80: use isolated state/configuration roots. No SDK runtime production changed in
81: this slice, and no new quarantine dependency was added to SDK.
82: 
83: The boundary tests compare complete migrated fact/tick envelopes, verify the
84: external signed report at its completion head, assert exact SDK store/head and
85: post-sync projection evidence, and require declaration-dependent
86: summary/facts/ticks/lookup/inspection/edit/emission to refuse with the source,
87: ledger, and descriptor bytes intact. These use public in-process SDK calls and

===== docs/scratch/arrival-break/consistency-c9-migration-adoption-design-2026-09-06.md =====
1: # C9 migration to current Arrival adoption
2: 
3: ## Current gap
4: 
5: At baseline `50e37595`, before this publication slice, the offline sidecar creates a valid
6: Arrival **custody** history, then rewrites a legacy vertex store clause. It does
7: not create a current Arrival declaration history. `run_migration` mints a
8: physical lineage and appends transformed records through the registry
9: ([sidecar.py](../../../libs/migrate/src/migrate/sidecar.py#L647)), but
10: `edit_vertex_store_clause` writes only `backend="file"`
11: ([sidecar.py](../../../libs/migrate/src/migrate/sidecar.py#L209)). The current
12: SDK refuses a descriptor without an explicit role
13: ([target.py](../../../libs/sdk/src/sdk/target.py#L88)).
14: 
15: Adding `role="authority"` and the physical lineage is necessary, but it does
16: not make the target usable. Ordinary descriptor-based Arrival reads resolve the effective
17: declaration from the captured projection and refuses when no declaration is
18: adopted; it will not substitute the locator file
19: ([read.py](../../../libs/sdk/src/sdk/read.py#L194)). Projection identity is
20: licensed only by a consumed `_decl.genesis` fact whose **fact ID equals the
21: physical log lineage** ([arrival_projection.py](../../../libs/engine/src/engine/arrival_projection.py#L423)).
22: The sidecar transform creates migration key introductions and migrated rows, but
23: no such declaration fact. Therefore a role-only publication would make
24: `resolve_arrival_target`, `verify_target`, and `sync_target` reachable, while
25: inspection, normal reads, declaration editing, and writes still honestly
26: refuse.
27: 
28: The current initializer already has the exact declaration-anchor construction:
29: `_draft` makes one `_decl.genesis` row using canonical document JSON, a FACT
30: inner signature, and an ARRIVAL outer signature
31: ([arrival_initialization.py](../../../libs/engine/src/engine/arrival_initialization.py#L273)).
32: That construction must be extracted or shared; a migration-specific copy would
33: invite wire and signing drift.
34: 
35: ## Bounded publication hardening
36: 
37: First change only the sidecar publication output, not old records or source
38: inputs. Its surgical editor must write and re-parse:
39: 
40: ```kdl
41: store "<rehearsal arrival location>" backend="file" lineage="<physical lineage>" role="authority"
42: ```
43: 
44: It must retain the existing same-file atomic replace and non-store AST equality,
45: and validate all three descriptor claims. This makes the copied target an
46: explicit descriptor declaring the operational authority role; the registry still
47: checks the residence and lineage, and the declaration is still unadopted.
48: 
49: The later adoption slice should retain a hash of the exact copied vertex bytes
50: used for transform and report it. That hash is a proposal, not current sidecar
51: output. It prevents a later local edit from silently becoming the adoption
52: declaration. The sidecar already protects legacy-source content during its run;
53: the publication implementation now checks its captured declaration bytes in
54: memory, but does not persist that snapshot for a later adoption ceremony.
55: 
56: For a rehearsal, copy the legacy source and vertex before any sidecar write,
57: write all staging log/report/credentials under a disposable root, and isolate
58: XDG state/config/LOOPS_HOME. A live SQLite source needs a coherent snapshot
59: mechanism or a quiesced copy; byte copying a changing database is not an honest
60: fixture. The original source, vertex, store, keys, and witnesses remain untouched.
61: 
62: ## Explicit adoption, a later append-forward ceremony
63: 
64: A new dedicated engine operation with an SDK wrapper should adopt a **reviewed
65: snapshot** into the already-migrated copied log. It is neither `init_vertex`
66: nor `edit_declaration`: initialization mints a fresh lineage, while declaration
67: editing correctly requires an existing historized declaration.
68: 
69: Inputs are: the explicit authority descriptor; the selected migration head `S`;
70: an explicitly reviewed declaration snapshot; declared adopting observer; mapped
71: credentials; and an optional verified migration-report receipt. The operation
72: must:
73: 
74: 1. Open the descriptor through the registry, capture `H`, require `H == S`, and
75:    `Full(S)`. A changed source target refuses before an adoption append.
76: 2. Require an explicit selected document snapshot. The initial recommendation is
77:    the copied locator bytes used by transform, re-parsed and compared with a
78:    retained hash/documents only after the caller names that snapshot for
79:    adoption. A copied legacy locator is evidence to review, not automatic
80:    authority: it can be stale relative to the legacy store's then-effective
81:    declaration. Any alternative reviewed declaration needs its own named
82:    input/evidence. The ceremony never reads a later cache as authority.
83: 3. Scan the captured prefix for declaration identity conflicts before drafting:
84:    refuse an existing current-lineage `_decl.genesis`, any fact already using the
85:    physical-lineage fact ID, and any historical `_decl.*` record that claims the
86:    new physical lineage. Historical foreign or legacy declaration rows remain
87:    retained evidence; they are not selected, removed, or reinterpreted as this
88:    store's authority.
89: 4. Parse and validate the selected documents with the current Vertex grammar and
90:    runtime identity validator. Refuse duplicate document subjects and loop/vertex
91:    collisions. In the first adoption slice, every selected observer public key
92:    must match a key already valid for that observer at `S`; it neither introduces
93:    nor replaces keys. A later key-changing adoption must use the existing
94:    key-introduction/declaration path with separately designed ordering.
95: 5. Verify registry-forming migration signatures (genesis and key introductions)
96:    with an injected ARRIVAL verifier. `verify_target` alone establishes grammar,
97:    density, lineage, and hash chain, not signatures, key trust, or projection
98:    agreement. Migrated ordinary envelopes may remain unsigned under the existing
99:    migration contract; adoption must not claim their authorship.
100: 6. Resolve a pre-created mapped binding for the adopting observer after capture.
101:    Its public key must equal the key valid for that observer at `S`; no resolver
102:    mint, legacy import, or fallback occurs in adoption. Use separate FACT and
103:    ARRIVAL requests/signatures, then verify both against that captured key.
104: 7. Append exactly one initializer-compatible `_decl.genesis` draft whose fact ID
105:    is the physical lineage. Existing migration rows stay byte-for-byte intact.
106:    The commit names `before=S` and `after=A`; it is the explicit identity claim.
107: 8. Explicitly sync through `A`. Only that consumed row lets the projection stamp
108:    `own_lineage`. Publish/reconcile the copied locator cache as an operation
109:    phase, retaining the real commit if it fails after append.
110: 
111: This needs an intent/recovery record analogous to initialization/declaration
112: editing: durable planned draft, descriptor, `S`, selected document hash, author,
113: and public non-secret binding evidence. Recovery never re-signs or duplicates a
114: committed draft. It may append the exact reserved draft only after proving that
115: the captured `S` still names the pre-append target and that no planned draft is
116: present; otherwise it compares the durable draft, synchronizes, and finishes
117: cache publication. Refusal before append has no fabricated commit; post-append
118: failure retains `Commit` and an adoption-specific incomplete result.
119: 
120: ## Signer and provenance boundaries
121: 
122: The sidecar `Signer(observer, digest)` currently signs genesis, key
123: introductions, and the external report without a domain/purpose argument.
124: Mapped credentials are domain- and purpose-specific. Do not silently route that
125: callback through one mapped signer. Decide whether migration report signing stays
126: an explicitly injected sidecar/report verifier, or version the sidecar signer
127: contract with an explicit report domain. The adoption record itself uses current
128: FACT/ARRIVAL domains only.
129: 
130: `verify_migration_report` must run against target head `S` before adoption; its
131: current live-head equality is expected to fail after the adoption append. An
132: adoption outcome may retain report path/digest and verified `S`, but this is
133: external audit evidence, not a claim that the report itself is an on-ledger
134: identity record. If durable source-to-adoption provenance is required, it needs
135: a separately designed versioned declaration payload or evidence event.
136: 
137: The filename-derived sidecar custodian is quarantined historical behavior. The
138: new ceremony must take the explicit observer and prove it is authorized by the
139: captured migration key registry. Foreign `_decl.genesis` records remain inert:
140: only an ID matching the physical lineage licenses the projection marker.
141: 
142: ## Acceptance path
143: 
144: Use a temporary copied JSONL fixture first. Prove: sidecar output has explicit
145: backend/lineage/authority and `Full(S)`; report verifies at `S`; pre-adoption
146: sync can build a projection but inspection/read/write refuse for no adopted
147: declaration; adoption refuses stale `S`, wrong mapped public key, bad genesis or
148: key-introduction signature, an existing/current-lineage declaration collision,
149: conflicting physical-lineage fact ID, a historical declaration overlay claiming
150: the new lineage, changed copied declaration bytes, invalid runtime identity, and
151: new/replaced observer keys without an append. Success appends one exact anchor,
152: preserves every prefix row through `S`, and treats that synthetic anchor as
153: outside the sidecar's source-equivalence comparison. It syncs and makes
154: inspection/read basis `A` agree. Recovered after-append failure must not
155: duplicate the anchor. A later mapped write must use the captured key history and
156: report its actual post-commit projection result; ordinary mapped writes may
157: already synchronize as part of their supported operation, so no universal second
158: sync claim is made.
159: 
160: ## Finite implementation order
161: 
162: 1. Complete and review the bounded publication implementation with synthetic
163:    offline integration tests. It remains explicitly unadopted.
164: 2. Design and implement a shared initializer-compatible anchor-draft builder,
165:    captured-prefix collision/key/document validation, adoption
166:    intent/apply/recovery types, and a narrow engine append seam.
167: 3. Add an SDK adoption wrapper with mapped binding and captured key-registry
168:    checks, without importing the migration sidecar into ordinary reads/writes.
169: 4. Add a rehearsal orchestrator that snapshots/reviews input, invokes sidecar,
170:    verifies its report at `S`, then calls adoption. It owns temporary roots and
171:    report provenance presentation.
172: 
173: Open decisions: whether the report’s signature domain is versioned; whether a
174: reviewed non-identical declaration can be adopted; and whether source-to-anchor
175: provenance must be durable rather than retained in the operation receipt.
