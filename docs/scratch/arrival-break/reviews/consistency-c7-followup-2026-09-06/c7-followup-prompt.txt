62: def _arrival_definition(
63:     target: Path | str, *, allow_aggregate: bool = False
64: ) -> tuple[Path, Any, StoreDescriptor | None] | None:
65:     """Parse one Arrival-capable root without probing or opening its residence.
66:
67:     ``TargetInfo`` is a legacy filesystem classification and necessarily turns
68:     a store spelling into a ``Path``. Descriptor resolution instead parses the
69:     vertex locator directly, so another backend's DSN or service URL remains
70:     byte-for-byte opaque until that named adapter receives it. Aggregate shape
71:     remains refused by default; declaration inspection may opt in because it
72:     reports bounded root structure without claiming one basis for the aggregate.
73:
74:     A valid storeless vertex retains its parsed AST with ``descriptor=None`` so
75:     a structural caller can make its descriptor/storeless decision from these
76:     same bytes. Malformed and non-vertex targets retain the legacy ``None``
77:     classification.
78:     """
79:     path = Path(target).resolve()
80:     if path.suffix.lower() != ".vertex":
81:         return None
82:     if not path.exists():
83:         raise TargetNotFound(f"target path does not exist: {path}")
84:     try:
85:         ast = parse_vertex_file(path)
86:     except Exception:
87:         return None  # the legacy resolver owns malformed-target classification
88:     descriptor = descriptor_for(ast, path)
89:     if descriptor is None:
90:         return path, ast, None
91:     if descriptor.role is None:
92:         raise SdkValueError(
93:             f"Arrival target {path} must declare store role explicitly"
94:         )
95:     if not allow_aggregate and (ast.combine is not None or ast.discover is not None):
96:         raise TargetUnsupported(
97:             f"Arrival target {path} is an aggregate; member-basis reads are not "
98:             "implemented in this single-store stage"
99:         )
100:     return path, ast, descriptor
103: def _arrival_descriptor(
104:     target: Path | str, *, allow_aggregate: bool = False
105: ) -> tuple[Path, Any, StoreDescriptor] | None:
106:     """Resolve and require an explicitly declared Arrival descriptor."""
107:     resolved = _arrival_definition(target, allow_aggregate=allow_aggregate)
108:     if resolved is None:
109:         return None
110:     path, ast, descriptor = resolved
111:     if descriptor is None:
112:         return None
113:     return path, ast, descriptor
