out.mkdir(parents=True, exist_ok=True)
baseline = '79696a72'

def git(*args):
    return subprocess.check_output(['git', *args], cwd=root, text=True)

def nodes(text):
    return {n.name: n for n in ast.parse(text).body
            if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef))}

def span(n):
    return min([n.lineno] + [d.lineno for d in n.decorator_list]), n.end_lineno

def node_text(lines, node):
    a, b = span(node)
    return '\n'.join(lines[a - 1:b])

changed = git('diff', '--name-only').splitlines()
python_files = [f for f in changed if f.endswith('.py')]

hashes = {}

def add(file, ranges=None):
    p = root / file
    data = p.read_text()
    hashes[file] = hashlib.sha256(p.read_bytes()).hexdigest()
    lines = data.splitlines()
    ranges = ranges or [(1, len(lines))]
    indices = sorted({i for a, b in ranges for i in range(a, b + 1)})
    parts.append('\n## ' + file + '\n```\n' + '\n'.join(
        f'{i}: {lines[i - 1]}' if lines[i - 1] else f'{i}:' for i in indices
    ) + '\n```\n')

for file in python_files:
    data = (root / file).read_text()
    old = git('show', baseline + ':' + file)
    current_nodes, old_nodes = nodes(data), nodes(old)
    lines, old_lines = data.splitlines(), old.splitlines()
    selected = {name for name, node in current_nodes.items()
                if name not in old_nodes or node_text(lines, node) != node_text(old_lines, old_nodes[name])}
    if '/tests/' in file:
        pending = list(selected)
        while pending:
            name = pending.pop()
            references = {n.id for n in ast.walk(current_nodes[name]) if isinstance(n, ast.Name)}
            references.update(n.arg for n in ast.walk(current_nodes[name]) if isinstance(n, ast.arg))
            dependencies = references.intersection(current_nodes).difference(selected)
            selected.update(dependencies)
            pending.extend(dependencies)
    # Keep imports and simple module constants visible before the first function.
    prefix_end = min((span(n)[0] for n in current_nodes.values()), default=len(lines) + 1) - 1
    ranges = [(1, prefix_end)] if prefix_end else []
    ranges.extend(span(current_nodes[name]) for name in selected)
    add(file, ranges)

]:

        add(file, [span(ns[name]) for name in symbols])
